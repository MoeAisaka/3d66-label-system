def test_failed_safety_supplement_keeps_human_and_does_not_repeat_model(setup):

    from app.safety_rules import SafetyError
    from app.safety_regression import SafetyAttempt
    from app.texture_regression import retry_item, recompute, TextureStatistics
    engine, sessions, settings, *_ = setup

    class Provider:
        region = "cn-shanghai"
        fail = True
        calls = 0
        def moderate_image(self, image, request_id, service):
            self.calls += 1
            if self.fail: raise SafetyError("fixture_declined", retryable=False, submitted=True)
            return {"Code": 200, "Data": {"DataId": request_id, "RiskLevel": "medium", "Result": [{"Label": "fixture_risk", "Confidence": 90}]}}

    assert safety_tick(engine, settings=settings, provider=provider)

        assert db.scalar(select(func.count(SafetyAttempt.id))) == 2 and provider.calls == 2
        assert db.scalar(select(func.count(TextureResult.id))) >= 3
