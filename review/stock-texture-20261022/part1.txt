def _validate_snapshot(policy):
    if not isinstance(policy,dict) or policy.get('schema_version')!='texture-audit-policy-v1' or policy.get('mode')!='stock':
        raise TextureAuditError('invalid_texture_policy')
    if policy.get('policy_hash')!=_hash({k:v for k,v in policy.items() if k!='policy_hash'}):
        raise TextureAuditError('policy_snapshot_changed')
    selected=policy.get('selected_rules')
    if (not isinstance(selected,list) or not selected or any(not isinstance(k,str) or k not in RULES for k in selected)
        or len(set(selected))!=len(selected) or type(policy.get('full_scope')) is not bool
        or policy['full_scope']!=(set(selected)==set(RULES))
        or policy.get('unselected_rules')!=[k for k in RULES if k not in selected]
        or policy.get('excluded_rules')!=['similarity_lookup','upload_time_order']):
        raise TextureAuditError('invalid_texture_policy_scope')
    rules=policy.get('rules')
    if not isinstance(rules,list) or len(rules)!=len(selected):raise TextureAuditError('invalid_texture_policy_rules')
    for key,rule in zip(selected,rules):
        if (not isinstance(rule,dict) or rule.get('key')!=key or rule.get('required_inputs')!=RULES[key]['required_inputs']
            or any(not isinstance(rule.get(field),str) or not rule[field].strip() or len(rule[field])>8000
                   for field in ('title','requirement','rejection_reason','source'))
            or not isinstance(rule.get('exceptions'),list) or len(rule['exceptions'])>50
            or any(not isinstance(value,str) or len(value)>8000 for value in rule['exceptions'])):
            raise TextureAuditError('invalid_texture_policy_rules')
    if 'company_compliance' in selected:
        standard=policy.get('company_standard')
        if (not isinstance(standard,dict) or standard.get('schema_version')!='company-texture-compliance-v1'
            or not isinstance(standard.get('rows'),list) or not 0<len(standard['rows'])<=1000):
            raise TextureAuditError('invalid_company_standard')

def reduce_checks(policy, checks, *, available_inputs, safety=None, partial_scope_confirmed=False):
    _validate_snapshot(policy)
    if not isinstance(checks,list):raise TextureAuditError('invalid_texture_checks')
    if type(partial_scope_confirmed) is not bool:raise TextureAuditError('invalid_scope_confirmation')
    if not isinstance(available_inputs,(set,frozenset,list,tuple)) or any(not isinstance(x,str) for x in available_inputs):
        raise TextureAuditError('invalid_input_availability')
    groups={};guards=[]
    for row in checks:
        if not isinstance(row,dict) or not isinstance(row.get('rule_key'),str):guards.append('invalid_rule_row');continue
        if row['rule_key'] not in policy['selected_rules']:guards.append('unexpected_rule_key');continue
        groups.setdefault(row['rule_key'],[]).append(row)
    results=[];reasons=[];decisions=[]
    for rule in policy['rules']:
        key=rule['key'];rows=groups.get(key,[]);missing=sorted(set(rule['required_inputs'])-set(available_inputs))
        result={'rule_key':key,'title':rule['title'],'status':'completed','verdict':'review','reasons':[], 'missing_inputs':missing}
        if missing: result.update(status='unverifiable',error='required_input_missing')
        elif not rows: result.update(status='unverifiable',error='rule_result_missing')
        elif len(rows)!=1: result.update(status='failed',error='duplicate_rule_results')
        else:
            row=rows[0];verdict=row.get('verdict');score=row.get('confidence');why=row.get('reasons');evidence=row.get('evidence')
            valid_verdict=isinstance(verdict,str) and verdict in {'pass','reject','review','not_applicable'}
            valid_score=score is None or type(score) in (int,float) and math.isfinite(score) and 0<=score<=1
            valid_reasons=isinstance(why,list) and len(why)<=50 and all(isinstance(x,str) and x.strip() and len(x)<=1000 for x in why)
            if (set(row)!={'rule_key','verdict','reasons','evidence','confidence'} or not valid_verdict or not valid_score or not valid_reasons or not isinstance(evidence,str) or not evidence.strip()
                or len(evidence)>4000 or verdict=='reject' and not why):
                result.update(status='failed',error='invalid_rule_result')
            else:result.update(verdict=verdict,reasons=why[:],evidence=evidence,confidence=score)
        if result['status']=='completed':
            decisions.append('pass' if result['verdict']=='not_applicable' else result['verdict'])
        else: decisions.append('review')
        reasons.extend({'rule_key':key,'reason':reason} for reason in result['reasons'])
        results.append(result)
    complete=all(r['status']=='completed' for r in results) and not guards
    if safety is not None:
        risk=safety.get('verdict') if isinstance(safety,dict) else None
        if (not isinstance(safety,dict) or safety.get('status')!='completed' or not isinstance(risk,str) or risk not in {'pass','reject','review'}
            or not isinstance(safety.get('reasons'),list)):
            complete=False;decisions.append('review')
        else:
            decisions.append(risk)
            for reason in safety.get('reasons',[]):reasons.append({'rule_key':'aliyun_safety','reason':copy.deepcopy(reason)})
    scope_complete=bool(policy['full_scope'] and safety is not None and complete)
    requires_confirmation=not scope_complete and not partial_scope_confirmed
    return {'schema_version':'texture-audit-result-v1','policy_hash':policy['policy_hash'],
        'verdict':max(decisions or ['review'],key={'pass':0,'review':1,'reject':2}.get),
        'reasons':reasons,'rule_results':results,'raw_checks':copy.deepcopy(checks),'safety':copy.deepcopy(safety),
        'guards':sorted(set(guards)),'execution_complete':complete,'full_scope_evaluated':scope_complete,
        'requires_partial_scope_confirmation':requires_confirmation,
        'can_write':complete and not requires_confirmation,
        'validation_scope':{'selected_rules':policy['selected_rules'][:],'unselected_rules':policy['unselected_rules'][:],
            'excluded_rules':policy['excluded_rules'][:],'aliyun_safety_selected':safety is not None,
            'partial_scope_confirmed':partial_scope_confirmed}}

def normalize_working_content(content):
    # Editing a working copy intentionally creates a new policy hash. Runtime
    # validation below never repairs an immutable snapshot's hash.
    policy = content.get("policy") if isinstance(content, dict) else None
    if not isinstance(policy, dict): raise TextureAuditError("texture_module_invalid")
    policy["policy_hash"] = _hash({key: value for key, value in policy.items() if key != "policy_hash"})
    validate_content(content)


def validate_content(content):
    if (not isinstance(content, dict)
        or set(content) != {"format_version", "policy", "model_node_id"}
        or content.get("format_version") != "texture-audit-module-v1"
        or not isinstance(content.get("model_node_id"), str)
        or not re.fullmatch(r"[A-Za-z0-9_-]{1,120}", content["model_node_id"])):
        raise TextureAuditError("texture_module_invalid")
    _validate_snapshot(content["policy"])


def policy_for_model_node(manifest, node_id):

    for node in manifest.get("nodes", []):
        config = node.get("config", {}) if isinstance(node, Mapping) else {}
        if node.get("kind") != "rule_module" or config.get("module_type_key") != MODULE_TYPE: continue
        snapshot = config.get("component_snapshot", {})
        content = snapshot.get("content")
        if not isinstance(content, dict):
            try: content = json.loads(snapshot.get("content_json", "null"))
            except (ValueError, TypeError): raise TextureAuditError("texture_module_snapshot_invalid") from None
        validate_content(content)
        if content["model_node_id"] == node_id: matches.append(content["policy"])
    if len(matches) > 1: raise TextureAuditError("texture_policy_ambiguous")
    return matches[0] if matches else None


def bind_model_policy(manifest, node_id, system_prompt):
    """One source for the actual model request and its input/cost estimate."""
    policy = policy_for_model_node(manifest, node_id)
    if policy is None: return system_prompt, None
    return system_prompt + "\n\n【本轮冻结规则，优先于正文中的旧示例】\n" + prompt_text(policy), policy["policy_hash"]


def execute(context):
    config = context.node.get("config", {})
    snapshot = config.get("component_snapshot", {})
    content = snapshot.get("content")
    if not isinstance(content, dict):
        try:
            content = json.loads(snapshot.get("content_json", "null"))
        except (TypeError, ValueError):
            raise TextureAuditError("texture_module_snapshot_invalid") from None
    validate_content(content)
    upstream = context.input_manifest.get("upstream", {})
    result = upstream.get(content["model_node_id"], {}) if isinstance(upstream, Mapping) else {}
    output = result.get("output", {}) if isinstance(result, Mapping) else {}
    checks = output.get("checks", []) if isinstance(output, Mapping) else []
    if result.get("status") != "succeeded":
        checks = []
    runtime = context.run_snapshot.get("runtime_context", {})
    proof = runtime.get(CONTEXT_KEY, {}) if isinstance(runtime, Mapping) else {}
    if not isinstance(proof, Mapping):
        raise TextureAuditError("texture_input_proof_invalid")
    # Never fall back to prompt output / upstream / user supplied input fields.
    normalized = reduce_checks(content["policy"], checks,
        available_inputs=proof.get("available_inputs", []),
        safety=proof.get("safety"),
        partial_scope_confirmed=proof.get("partial_scope_confirmed", False))
    return {"normalized_result": normalized}

"""Create ordinary versioned native workflows for stock texture review.

No incremental binding or business connector is created here. Identical
policy/model selections reuse the immutable mechanism, not an editable copy.

CATEGORY = "texture"


def managed_mechanism(db, version_id):
    from .workflow_registry import persisted_workflow_manifest
    from .texture_workflow import validate_content
    row = db.get(WorkflowVersion, version_id)
    if not row or row.status != "active" or row.workflow_schema_version != "workflow-v3":
        raise TextureAuditError("texture_mechanism_unavailable")
    manifest = persisted_workflow_manifest(row)
    if manifest.get("category_key") != CATEGORY or "stock_regression" not in manifest.get("run_modes", []):
        raise TextureAuditError("texture_mechanism_scope_mismatch")
    prompts = [node for node in manifest.get("nodes", []) if node.get("kind") in {"prompt", "foreach_image"}]
    modules = [node for node in manifest.get("nodes", []) if node.get("kind") == "rule_module" and node.get("config", {}).get("module_type_key") == "texture_audit"]
    if len(prompts) != 1 or prompts[0]["kind"] != "prompt" or len(modules) != 1:
        raise TextureAuditError("texture_mechanism_nodes_incompatible")
    component = modules[0]["config"].get("component_snapshot", {})
    content = component.get("content")
    if not isinstance(content, dict):
        try: content = json.loads(component.get("content_json", "null"))
        except (ValueError, TypeError): raise TextureAuditError("texture_module_snapshot_invalid") from None
    validate_content(content)
    prompt = prompts[0]
    if content["model_node_id"] != prompt["node_instance_id"]:
        raise TextureAuditError("texture_model_node_mismatch")
    contract = prompt["config"].get("output_contract", {})
    if contract.get("mode") != "json_schema" or contract.get("strict") is not True or contract.get("schema") != output_schema(content["policy"]):
        raise TextureAuditError("texture_model_contract_incompatible")
    return row, content["policy"], prompt["config"]["model_registry_id"], prompt["node_instance_id"]


def ensure_category(db, actor, *, require_active=True):
    row = db.scalar(select(EvaluationCategoryProfile).where(EvaluationCategoryProfile.category_key == CATEGORY))
    if row is None:
        row = EvaluationCategoryProfile(category_key=CATEGORY, display_name="普通贴图审核", status="active",
            description="独立的贴图业务规则与安全审核；存量回归不查重。",
            allowed_mime_types_json='["image/jpeg","image/png","image/webp","image/tiff","image/bmp","image/gif"]',
            preprocess_config_json='{"preprocess":"image"}', pipeline_config_json="{}",
            rubric_version="texture-business-v4", created_by=actor)
        db.add(row); db.flush()
    if require_active and row.status != "active":
        raise TextureAuditError("texture_category_disabled")
    return row

    ensure_category(db, actor)

        raise TextureAuditError("texture_model_unavailable")

    if model.protocol != "openai_chat" or not {"vision", "structured_output"} <= set(capabilities):
        raise TextureAuditError("texture_model_incompatible")

            description="业务规则逐项检查；安全审核独立执行并由存量任务汇总。不连接生产写入。",

        "temperature", "max_tokens", "timeout_seconds", "thinking_mode", "structured_output",
        "vision_capability_profile_json", "pricing_contract_json", "capabilities_json")

    def node(node_id, kind, title, config, schema=None):
