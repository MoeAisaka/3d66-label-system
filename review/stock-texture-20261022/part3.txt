def build_texture_regression_router(current_user):
    router = APIRouter(tags=["stock-texture-review"])
    def writable(user):
        if not has_permission(user, "reviews:write"): raise HTTPException(403, "需要贴图业务纠偏权限")
        require_categories(user, {CATEGORY})
    def run_row(db, run_id):
        row = db.get(TextureRun, run_id)
        if not row: raise HTTPException(404, "贴图审核回归不存在")
        return row
    def item_row(db, run_id, item_id):
        row = db.get(TextureItem, item_id)
        if not row or row.run_id != run_id: raise HTTPException(404, "样本不属于本轮")
        return row
    def summary(run):
        return {"id": run.id, "name": run.name, "status": run.status, "category_key": run.category_key,
            "config": json.loads(run.config_json), "native_run_id": run.native_run_id,
            "safety_run_id": run.safety_run_id, "error_code": run.error_code, "created_at": run.created_at}
    def execute(db, operation):
        try: value = operation(); db.commit(); return value
        except IntegrityError:
            db.rollback(); raise HTTPException(409, "请求冲突，请读取最新版本后重试") from None
        except ValueError as exc:
            db.rollback()
            code = getattr(exc, "code", str(exc))
            if not isinstance(code, str) or not re.fullmatch(r"[a-zA-Z0-9_.-]{1,120}", code): code = "texture_configuration_invalid"
            raise HTTPException(409, {"code": code, "message": "未能完成操作，请核对输入、模型配置及当前任务状态。"}) from None

    @router.get("/api/texture-regressions/capabilities")
    def capabilities(user=Depends(current_user), db: Session = Depends(get_db)):
        from .models import ModelRegistryEntry, ModelConfig, WorkflowVersion
        models = []
        for entry in db.scalars(select(ModelRegistryEntry).where(ModelRegistryEntry.role == "main", ModelRegistryEntry.active.is_(True)).order_by(ModelRegistryEntry.id)):
            source = db.get(ModelConfig, entry.source_model_config_id) if entry.source_model_config_id else None
            compatible = entry.protocol == "openai_chat" and {"vision", "structured_output"} <= set(json.loads(entry.capabilities_json))
            models.append({"id": entry.id, "name": entry.name, "model_id": entry.model_id, "compatible": compatible,
                "ready": bool(compatible and source and source.active and source.benchmark_enabled and source.encrypted_api_key)})
        from .texture_mechanism import managed_mechanism
        workflows = []
        candidates = db.scalars(select(WorkflowVersion).where(WorkflowVersion.status == "active", WorkflowVersion.workflow_schema_version == "workflow-v3")
            .order_by(WorkflowVersion.id.desc()).limit(100)).all()
        for row in candidates:
            try: _, policy, mid, _ = managed_mechanism(db, row.id)
            except (TextureAuditError, ValueError, KeyError): continue
            workflows.append({"id": row.id, "version": row.version, "model_registry_id": mid,
                "selected_rules": policy["selected_rules"], "rules": policy["rules"]})
        return {"enabled": os.environ.get("STOCK_TEXTURE_ENABLED") == "1", "category_key": CATEGORY, "rules": list(RULES.values()),
            "default_rules": list(RULES), "default_safety_rules": DEFAULT_RULES, "safety_configured": provider_configuration()["configured"],
            "models": models, "workflows": workflows, "default_model_id": next((m["id"] for m in models if m["ready"]), None),
            "excluded_rules": ["similarity_lookup", "upload_time_order"], "actual_cost": None}

    @router.post("/api/texture-regressions")
    def launch(payload: CreateTextureRound, user=Depends(current_user), db: Session = Depends(get_db)):
        writable(user)
        if os.environ.get("STOCK_TEXTURE_ENABLED") != "1": raise HTTPException(503, "贴图存量审核后台尚未启用")
        if payload.safety_selected and not provider_configuration()["configured"]: raise HTTPException(503, "阿里安全审核凭据尚未配置")
        from .models import BaselineSet, Asset, BaselineSetItem, ModelRegistryEntry, ModelConfig
        if bool(payload.baseline_set_id) == bool(payload.samples): raise HTTPException(422, "基准集与导入素材只选一种")
        if payload.baseline_set_id:
            baseline = db.get(BaselineSet, payload.baseline_set_id)
            if not baseline or baseline.category_key != CATEGORY: raise HTTPException(409, "请选择贴图类目的基准集")
            categories = set(db.scalars(select(Asset.category_key).join(BaselineSetItem).where(BaselineSetItem.baseline_set_id == baseline.id)))
            require_categories(user, categories | {CATEGORY})
        chosen_model_id = payload.model_registry_id
        if payload.workflow_version_id:
            from .texture_mechanism import managed_mechanism
            try: _, _, mid, _ = managed_mechanism(db, payload.workflow_version_id)
            except ValueError: raise HTTPException(409, "该机制不是适配本入口的贴图审核版本") from None
            chosen_model_id = mid
        if not chosen_model_id: raise HTTPException(422, "请选择模型或已管理的贴图机制")
        model = db.get(ModelRegistryEntry, chosen_model_id)
        source = db.get(ModelConfig, model.source_model_config_id) if model and model.source_model_config_id else None
        if not source or not source.active or not source.benchmark_enabled or not source.encrypted_api_key:
            raise HTTPException(409, "所选模型尚不具备真实调用条件")
        def operation():
            baseline_id = payload.baseline_set_id
            if not baseline_id:
                baseline_id = import_samples(db, request_id=payload.request_id,
                    samples=[sample.model_dump(exclude_none=True) for sample in payload.samples], actor=user.username).id
            return summary(create_round(db, actor=user.username, baseline_set_id=baseline_id,
                **payload.model_dump(exclude={"confirmed", "baseline_set_id", "samples"})))
        return execute(db, operation)

    @router.get("/api/texture-regressions")
    def listing(offset: int = Query(0, ge=0), limit: int = Query(50, ge=1, le=100), user=Depends(current_user), db: Session = Depends(get_db)):
        rows = db.scalars(select(TextureRun).order_by(TextureRun.created_at.desc(), TextureRun.id).offset(offset).limit(limit)).all()
        return {"items": [summary(row) for row in rows], "total": db.scalar(select(func.count(TextureRun.id)))}

    @router.get("/api/texture-regressions/{run_id}")
    def detail(run_id: str, offset: int = Query(0, ge=0), limit: int = Query(24, ge=1, le=100), user=Depends(current_user), db: Session = Depends(get_db)):
        run = run_row(db, run_id)
        ids = list(db.scalars(select(TextureItem.id).where(TextureItem.run_id == run.id).order_by(TextureItem.id).offset(offset).limit(limit)))
        items = [doc for _, doc in source_rows(db, run.id, item_ids=ids)]
        for doc in items:
            prepared = doc["input_preparation"]
            doc["preview_url"] = f"/api/texture-regressions/{run.id}/items/{doc['item_id']}/input/preview" if "preview" in prepared.get("available_inputs", []) else None
            doc.pop("result_configuration", None)
            for field in ("raw", "effective"):
                if doc[field]: doc[field] = {key: value for key, value in doc[field].items()
                    if key not in {"raw_checks", "business_evidence", "safety_evidence"}}
        safety_rules = None; safety_revision = None
        if run.safety_run_id:
            from .safety_regression import SafetyRun, SafetyRuleVersion
            safety_run = db.get(SafetyRun, run.safety_run_id); safety_revision = safety_run.rule_revision
            safety_rules = json.loads(db.scalar(select(SafetyRuleVersion.rules_json).where(
                SafetyRuleVersion.run_id == safety_run.id, SafetyRuleVersion.revision == safety_revision)))
        return {**summary(run), "statistics": statistics(db, run.id), "usage": usage(db, run.id), "items": items,
            "current_safety_rules": safety_rules, "safety_rule_revision": safety_revision, "offset": offset, "limit": limit,
            "total": db.scalar(select(func.count(TextureItem.id)).where(TextureItem.run_id == run.id))}

    @router.get("/api/texture-regressions/{run_id}/items/{item_id}/input/{role}")
    def input_file(run_id: str, item_id: str, role: Literal["preview", "original_file"], user=Depends(current_user), db: Session = Depends(get_db)):
        item = item_row(db, run_id, item_id); snapshot = json.loads(item.input_snapshot_json)[role]; sha = snapshot.get("sha256", "")
        if not isinstance(sha, str) or not re.fullmatch(r"[0-9a-f]{64}", sha): raise HTTPException(409, "输入身份不完整")
        path = get_settings().data_dir / "stock-texture" / "inputs" / sha
        if not path.is_file(): raise HTTPException(409, "固定输入尚未准备好")
        with path.open("rb") as file:
            if hashlib.file_digest(file, "sha256").hexdigest() != sha: raise HTTPException(409, "固定输入校验失败")
        mime = json.loads(item.preparation_json).get(role + "_alias", {}).get("mime_type", "application/octet-stream")
        return FileResponse(path, media_type=mime, headers={"X-Content-Type-Options": "nosniff"})

    @router.post("/api/texture-regressions/{run_id}/control/{action}")
