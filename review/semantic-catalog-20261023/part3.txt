READ_PERMISSIONS = frozenset({
    "assets:read", "jobs:read", "reviews:read", "scripts:read", "workflows:read",
    "runtime:read", "releases:read", "reports:read", "prompts:read", "dimensions:read",
    "models:read", "results:export", "preferences:write",
})
# Existing coarse dependencies are not object authorization. employee_access.py
# checks the exact action and all actual objects before those dependencies run.
OPERATE_PERMISSIONS = frozenset({
    "reviews:write", "regressions:run", "production:rerun", "jobs:write", "runtime:write",
})
ROLE_PERMISSIONS: Final = {
    "admin": frozenset({"*"}),
    "category_manager": READ_PERMISSIONS | OPERATE_PERMISSIONS,
    "product_rd": READ_PERMISSIONS | OPERATE_PERMISSIONS | frozenset({
        "drafts:write", "prompts:write", "dimensions:write", "scripts:write",
        "workflows:write", "evaluation_components:write", "evaluation_components:version",
        "evaluation_mechanisms:write",
    }),
    "viewer": READ_PERMISSIONS,
}
ROLE_LABELS: Final = {
    "admin": "系统管理员", "category_manager": "类目管理员",
    "product_rd": "产研", "viewer": "只读成员",
}
ROLE_HOMES: Final = {
    "admin": "/workflow/operations",
    "category_manager": "/workflow/incremental/correction",
    "product_rd": "/workflow/governance/workflow-manager",
    "viewer": "/workflow/dashboard",
}


def effective_role(user: User) -> str:
    if user.is_admin:
        return "admin"
    role = (user.role or "viewer").strip().lower()
    return role if role in ROLE_PERMISSIONS and role != "admin" else "viewer"


def has_permission(user: User, permission: str) -> bool:
    permissions = ROLE_PERMISSIONS[effective_role(user)]
    return "*" in permissions or permission in permissions


def writable_categories(user: User) -> list[str] | None:
    role = effective_role(user)
    if role in {"admin", "product_rd"}:
        return None
    if role != "category_manager":
        return []
    try:
        values = json.loads(user.category_keys_json or "[]")
    except (ValueError, TypeError):
        return []
    return sorted(set(item for item in values if isinstance(item, str))) if isinstance(values, list) else []


def require_categories(user: User, categories: set[str | None]) -> None:
    if effective_role(user) == "admin":
        return
    if not categories or None in categories or "" in categories:
        raise HTTPException(403, "无法确认对象类目，请联系系统管理员")
    allowed = writable_categories(user)
    if allowed is not None and not categories.issubset(set(allowed)):
        raise HTTPException(403, "包含未授权类目的对象，本次操作未执行")


def lock_access_users(db: Session, user_ids: list[int]) -> None:
    reserve_sqlite_immediate_transaction(db)
    db.execute(select(User.id).where(User.id.in_(sorted(set(user_ids))))
               .order_by(User.id).with_for_update()).all()


def protect_write_transaction(db: Session, user: User) -> None:
    """Recheck after legacy service commits; serialize with account revocation.

    Revocation locks the same actor row: it either precedes this transaction
    (rejected) or follows its commit. Already approved queued work continues.
    """
    db.info["employee_write_guard"] = (user.id, user.permission_revision)
    _check_write_transaction(db)


def _check_write_transaction(db: Session) -> None:
    identity = db.info.get("employee_write_guard")
    if not identity:
        return
    if db.get_transaction() is not None and db.info.get("employee_guard_transaction") is db.get_transaction():
        return
    user_id, revision = identity
    lock_access_users(db, [user_id])
    row = db.execute(select(User.is_active, User.permission_revision).where(User.id == user_id)).one_or_none()
    if row is None or not row.is_active or row.permission_revision != revision:
        raise HTTPException(403, "权限已变更，请刷新后重试")
    from .employee_access import recheck_employee_objects
    recheck_employee_objects(db, db.get(User, user_id))
    db.info["employee_guard_transaction"] = db.get_transaction()


@event.listens_for(Session, "before_flush")
def _before_employee_flush(db: Session, _context, _instances) -> None:
    _check_write_transaction(db)


@event.listens_for(Session, "before_commit")
def _before_employee_commit(db: Session) -> None:
    _check_write_transaction(db)


def require_permission(permission: str):
    from .main import current_user

    def dependency(user: User = Depends(current_user)) -> User:
        if not has_permission(user, permission):
            raise HTTPException(status_code=403, detail=f"缺少权限：{permission}")
        return user

    return dependency

def check_origin(request: Request) -> None:
    """Cookie writes cannot originate from another site, including login CSRF."""
    if request.method in {"GET", "HEAD", "OPTIONS"}:
        return
    if request.headers.get("sec-fetch-site") == "cross-site":
        raise HTTPException(403, "不接受跨站操作")
    source = request.headers.get("origin") or request.headers.get("referer")
    if source:
        supplied, actual = urlsplit(source), urlsplit(str(request.base_url))
        from .dingtalk_identity import identity_settings
        config = identity_settings()
        canonical = urlsplit(config.redirect_uri)
        # A configured HTTPS public origin also works behind the private HTTP
        # gateway; arbitrary forwarded headers never supply this authority.
        if config.ready and actual.netloc == canonical.netloc:
            actual = canonical
        def origin(value):
            try:
                return value.scheme, value.hostname, value.port or (443 if value.scheme == "https" else 80)
            except ValueError:
                return None
        if origin(supplied) is None or origin(supplied) != origin(actual):
            raise HTTPException(403, "操作来源不匹配")


def _row(db: Session, model, identity):
    if identity is None:
        raise HTTPException(422, "缺少业务对象标识")
    # Pin ownership until this business transaction commits. A concurrent
    # administrator category change must order before or after the write.
    row = db.scalar(select(model).where(model.id == identity).with_for_update()
        .execution_options(populate_existing=True))
    if row is None:
        raise HTTPException(404, "业务对象不存在")
    return row


def _asset_categories(db: Session, ids: list[int]) -> set[str | None]:
    if not ids or len(ids) > 5000 or any(type(i) is not int or i <= 0 for i in ids):
        raise HTTPException(422, "素材范围无效")
    rows = db.execute(select(m.Asset.id, m.Asset.category_key).where(m.Asset.id.in_(ids))
        .order_by(m.Asset.id).with_for_update()).all()
    if len(rows) != len(set(ids)):
        raise HTTPException(404, "包含不存在的素材")
    return {r.category_key for r in rows}


def _texture_asset_categories(db, identifiers):
    # All roles share one ascending asset lock order, including large imports.
    ids = sorted(set(identifiers)); categories = set()
    for offset in range(0, len(ids), 5000):
        categories |= _asset_categories(db, ids[offset:offset + 5000])
    return categories


def _evaluation(db: Session, identity) -> set[str | None]:
    row = _row(db, m.EvaluationResult, identity)
    categories = {_row(db, m.Asset, row.asset_id).category_key}
    if row.production_run_id:
        categories.add(_row(db, m.ProductionRun, row.production_run_id).category_key)
    return categories

    params = {key: int(value) if key.endswith("_id") and str(value).isdigit() else value for key, value in params.items()}

        from .stock_writes import StockWritePlan,StockWriteTarget,StockWriteItem,source_run
        def source_categories(kind,run):
            from .cover_regression import CoverRegressionItem
            from .safety_regression import SafetyItem
            from .texture_regression import TextureItem
            if kind == 'texture':
                from .texture_regression import frozen_asset_ids
                return {run.category_key} | _texture_asset_categories(db, frozen_asset_ids(db, run.id))
            model=TextureItem if kind=='texture' else SafetyItem if kind=='safety' else m.BaselineRegressionItem if kind=='baseline' else CoverRegressionItem
            categories=set(db.scalars(select(m.Asset.category_key).join(model,model.asset_id==m.Asset.id)
                .where(model.run_id==run.id).order_by(m.Asset.id).with_for_update()).all())

        if params.get('item_id') is not None:
            item=db.get(StockWriteItem,params['item_id'])
            if not item:raise HTTPException(404,'请求不存在')
            target=db.get(StockWriteTarget,item.target_id)
            plan=_row(db,StockWritePlan,target.plan_id)
            _row(db,StockWriteTarget,target.id)
            item=_row(db,StockWriteItem,item.id)
            if plan.source_kind == 'texture':
                from .texture_regression import frozen_asset_ids
                return {plan.category_key} | _texture_asset_categories(db, frozen_asset_ids(db, plan.run_id, item.source_item_id))
            return {plan.category_key} | _asset_categories(db,[item.asset_id])
        if params.get('plan_id') is not None:
            plan=_row(db,StockWritePlan,str(raw_params['plan_id']))
            run=source_run(db,plan.source_kind,plan.run_id)
            if run is None:raise HTTPException(404,'回归轮次不存在')
            return source_categories(plan.source_kind,run)
        if body.get('source_kind') in {'cover','baseline','safety','texture'}:
            try:run=source_run(db,body['source_kind'],str(body.get('run_id','')))
            except ValueError:raise HTTPException(422,'轮次编号无效')
            if run is None:raise HTTPException(404,'回归轮次不存在')
            return source_categories(body['source_kind'],run)
        return {body.get('category_key')}
    if resolver == 'texture_create':
        from .texture_mechanism import CATEGORY
        if body.get('samples'):
            identifiers = [sample.get(key) for sample in body['samples'] if isinstance(sample, dict)
                for key in ('asset_id', 'original_asset_id', 'preview_asset_id') if sample.get(key) is not None]
            return {CATEGORY} | _texture_asset_categories(db, identifiers)
        baseline = _row(db, m.BaselineSet, body.get('baseline_set_id'))
        from .texture_regression import baseline_asset_ids
        return {CATEGORY, baseline.category_key} | _texture_asset_categories(db, baseline_asset_ids(db, baseline.id))
    if resolver in {'texture_run', 'texture_item'}:
        from .texture_regression import TextureRun, TextureItem
        run = _row(db, TextureRun, str(raw_params.get('run_id')))
        if resolver == 'texture_item':
            item = _row(db, TextureItem, str(raw_params.get('item_id')))
            if item.run_id != run.id: raise HTTPException(404, '样本不属于本轮')
            from .texture_regression import frozen_asset_ids
            return {run.category_key} | _texture_asset_categories(db, frozen_asset_ids(db, run.id, item.id))
        from .texture_regression import frozen_asset_ids
        return {run.category_key} | _texture_asset_categories(db, frozen_asset_ids(db, run.id))
    if resolver == 'safety_create':
        categories = {body.get('category_key')}
        if body.get('asset_ids'): categories |= _asset_categories(db, body['asset_ids'])
        if body.get('baseline_set_id'):
            baseline = _row(db, m.BaselineSet, body['baseline_set_id'])
            categories |= {baseline.category_key} | set(db.scalars(select(m.Asset.category_key).join(m.BaselineSetItem)
                .where(m.BaselineSetItem.baseline_set_id == baseline.id).order_by(m.Asset.id).with_for_update()).all())
        return categories
    if resolver in {'safety_run', 'safety_item'}:
        from .safety_regression import SafetyRun, SafetyItem
        run = _row(db, SafetyRun, str(raw_params.get('run_id')))
        if resolver == 'safety_item':
            item = _row(db, SafetyItem, str(raw_params.get('item_id')))
            if item.run_id != run.id: raise HTTPException(404, '样本不属于本轮')
            return {run.category_key} | _asset_categories(db, [item.asset_id])
        return {run.category_key} | set(db.scalars(select(m.Asset.category_key).join(SafetyItem)
            .where(SafetyItem.run_id == run.id).order_by(m.Asset.id).with_for_update()).all())
    if resolver in {"cover_run", "cover_item"}:
        from .cover_regression import CoverRegressionRun, CoverRegressionItem
        run = _row(db, CoverRegressionRun, str(params.get("run_id")))

            item = _row(db, CoverRegressionItem, str(params.get("item_id")))
            if item.run_id != run.id:
                raise HTTPException(404, "样本不属于本轮")

            raise HTTPException(404, "条目不属于指定运行")

            categories |= _evaluation(db, body.get("baseline_evaluation_id")) | _evaluation(db, body.get("candidate_evaluation_id"))
        return categories
    if resolver in {"runtime_id", "runtime_key"}:
        row = (_row(db, m.ProductionRun, params.get("run_id")) if resolver == "runtime_id" else
               db.scalar(select(m.ProductionRun).where(m.ProductionRun.run_key == params.get("run_key"))
                   .with_for_update().execution_options(populate_existing=True)))
        if row is None:
            raise HTTPException(404, "运行不存在")
        return {row.category_key}
    if resolver == "material_job":
        from .incremental_material_models import IncrementalMaterialJob
        return {_row(db, IncrementalMaterialJob, params.get("job_id")).category_key}
    if resolver == "assets":
        return _asset_categories(db, body.get("asset_ids", []))
    if resolver == "package":
        row = _row(db, m.MaterialPackage, body.get("material_package_id"))
        return {row.category_key} | set(db.scalars(select(m.Asset.category_key).join(m.MaterialPackageItem)
            .where(m.MaterialPackageItem.package_id == row.id).order_by(m.Asset.id).with_for_update()).all())
    if resolver in {"production_run", "production_evaluation"}:
        row = _row(db, m.EvaluationProductionRun, params.get("run_id"))
        categories = {row.category_key}
        if resolver == "production_evaluation":
            categories |= _evaluation(db, params.get("evaluation_id"))
        return categories
    raise HTTPException(403, "该操作仅系统管理员可执行")


def authorize_employee_write(request: Request, db: Session, user: m.User, body: dict[str, Any]) -> None:
    if request.method in {"GET", "HEAD", "OPTIONS"}:
        return
    check_origin(request)
    if effective_role(user) == "admin":
        return
    path = getattr(request.scope.get("route"), "path", "").rstrip("/")
    policy = WRITE_POLICIES.get((request.method, path))
    if policy is None or not has_permission(user, policy[0]):
        raise HTTPException(403, "当前角色不能执行此操作")
    resolver = policy[1]
    if resolver in {"self", "read"}:
        return
    if resolver == "draft":
        if body.get("status", "draft") not in {"draft", "candidate"} or body.get("active") is True:
            raise HTTPException(403, "产研只能编辑草稿，正式启用由系统管理员执行")
        return
    categories = object_categories(db, resolver, request.path_params, body)
    require_categories(user, categories)
    # A supplied category is never authoritative, but must also agree with the
    # source objects so it cannot redirect a write into another category.
    if body.get("category_key") and body["category_key"] not in categories:
        raise HTTPException(403, "请求类目与实际对象不一致")
    db.info["employee_object_guard"] = (resolver, dict(request.path_params), body)


def recheck_employee_objects(db: Session, user: m.User) -> None:
    guard = db.info.get("employee_object_guard")
    if guard:
        resolver, params, body = guard
        categories = object_categories(db, resolver, params, body)
        require_categories(user, categories)
        if body.get("category_key") and body["category_key"] not in categories:
            raise HTTPException(403, "对象类目已变化，请刷新后重试")

# 标准词表、别名与旧字段映射

状态：按已确认 Q4 范围实施；未部署。

## 范围与边界

复用已验知识图谱工作流，先补标准词表、别名和旧字段／旧值映射。不新增实体业务关系、不重跑历史模型、不自动修改已冻结机制或业务下游。三模型仍搁置。

每个词表由类目、字段路径和语言唯一确定；词条有稳定编号。原词保留，匹配键采用 NFKC、首尾去空白和 casefold，不删除词内空格或业务符号，不把搜索词的归一化规则套到标签。标准词与别名共用唯一匹配键，停用词仍占用原键，避免悄悄换指。旧字段映射明确记录来源系统、旧字段和旧值；不覆盖源数据。

管理员与产研可维护，员工只读。写入必须携带当前修订号；历史为追加记录，恢复须先看差异，再确认生成新修订。冻结导出携带词表修订和内容摘要，供后续工作流显式引用；修改词表不改变旧运行。

自动收录与运营整理分开：新词不设收录前审核，记录未整理及真实来源；已有人工标准词和停用状态不会被模型覆盖。原生收尾只登记待收录任务，独立后台事务写词表；收录失败不回滚评测结果，有限重试三次后显示需处理，可单独重试。四路流程按实际成功的提示词节点确定语言，跳过节点不能作为来源。`python -m app.semantic_catalog_worker` 为独立进程入口，先核对schema；`SEMANTIC_CATALOG_COLLECTION_ENABLED` 默认0，生产未启用，后续增量接入时须同时接线worker。

导入引用的词条、映射和目标词编号必须在批次前已经存在，不能猜测预览中新分配的临时编号；MySQL预览回滚后的自增号允许有空档，不作为连续编号承诺。历史恢复包含整理状态差异，恢复原修订状态，不把“待整理”悄悄变成“已整理”。

修改已有对象时，导入中未提供的列保留原值；显式空数组才清空别名，映射字段的显式null按可空定义清除。词条行不能混入映射列。只改名称不应意外清空别名、重新启用停用词或改变整理状态。

## 实施顺序和检查

1. 独立词表、词条、别名键、旧映射、历史与批次账本；SQLite139及MySQL138→139的升级入口同步。
2. 逐条维护、历史差异／恢复、导入预览与整批确认、全筛选导出；权限与事务冲突检查。
3. 员工界面，复用现有表格、抽屉、分页和按钮样式；构建与真实 Chrome 验证。
4. 自动收录及冻结规则适配；只处理实际字段，不把嵌套证据或路由控制字段当词。
5. 固定候选交 Poe／Gemini 集中复审；同归档隔离 Linux／MySQL验证。生产发布另按已有窗口规范，不用本地测试冒充上线。

当前已核对85：schema133/MCP9ad，知识图谱无增量绑定及专属接入登记。最新实验504的400条中26条校验失败，输出与冻结提示词的路线镜像约束冲突，不能通过放宽旧校验消除。实际增量输入及写回合同待对接资料；未切换机制、改提示词或重投。
