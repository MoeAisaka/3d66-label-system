def catalog(db, field="style", locale="zh"):
    return s.create_vocabulary(db, category_key="knowledge_graph", field_path=field, locale=locale, name="测试词表")


def term(db, identity, name="北欧简约", aliases=None, **extra):
    return s.save_term(db, identity, {"canonical_term": name, "aliases": aliases or [], **extra}, actor="user:1")


def test_frozen_lookup_aliases_inactive_identity_and_language(database):
    db = database; vocab = catalog(db)
    row = term(db, vocab.id, aliases=["北欧风", "Ｃ＋＋", "3D-Max"])
    first = s.frozen_catalog(db, vocab.id)
    assert s.resolve(first, "c++")["term"]["id"] == row.id
    assert s.resolve(first, "3d max")["status"] == "unmapped"
    assert s.resolve(first, "3d-max")["status"] == "matched"
    with pytest.raises(s.CatalogError, match="归属于其他"):
        term(db, vocab.id, "另一个词", ["c++"])
    term(db, vocab.id, aliases=["北欧风", "C++", "3D-Max"], id=row.id, expected_revision=1, status="inactive")
    assert s.resolve(s.frozen_catalog(db, vocab.id), "北欧风")["status"] == "inactive"
    assert s.resolve(first, "北欧风")["status"] == "matched"
    with pytest.raises(s.CatalogError):
        term(db, vocab.id, "北欧风")
    other = catalog(db, locale="en")
    assert term(db, other.id, "C++").id != row.id
    damaged = first | {"revision": 999}
    with pytest.raises(s.CatalogError, match="摘要"):
        s.resolve(damaged, "北欧风")


def test_restore_is_append_only_and_stale_preview_cannot_override(database):
    db = database; vocab = catalog(db); row = term(db, vocab.id)
    original = db.scalar(select(s.VocabularyHistory.id))
    term(db, vocab.id, "现代", id=row.id, expected_revision=1)
    preview = s.history_preview(db, original)
    assert preview["changes"]["canonical_term"] == {"before": "现代", "after": "北欧简约"}
    term(db, vocab.id, "中式", id=row.id, expected_revision=2)
    with pytest.raises(s.CatalogError, match="重新查看"):
        s.restore_history(db, original, preview_hash=preview["preview_hash"], actor="user:1")
    restored = s.restore_history(db, original, preview_hash=s.history_preview(db, original)["preview_hash"], actor="user:1")
    assert restored.id == row.id and restored.revision == 4 and restored.canonical_term == "北欧简约"
    assert db.scalar(select(func.count(s.VocabularyHistory.id))) == 4
    assert json.loads(db.get(s.VocabularyHistory, original).snapshot_json)["revision"] == 1


def test_legacy_field_and_value_mapping_preserves_inactive_override(database):
    db = database; vocab = catalog(db); row = term(db, vocab.id, aliases=["北欧风"])
    s.save_mapping(db, vocab.id, {"source_system": "old-db", "source_field": "old_style"}, actor="user:1")

def test_import_checks_all_rows_no_partial_write_and_confirmation_is_idempotent(database):
    db = database; vocab = catalog(db); db.commit()
    bad = s.preview_import(db, vocab.id, [{"canonical_term": "北欧"}, {"canonical_term": "北欧"}, {"canonical_term": ""}], actor="user:1")
    assert bad.state == "invalid" and len(json.loads(bad.errors_json)) == 2
    assert db.scalar(select(func.count(s.VocabularyTerm.id))) == 0
    assert db.scalar(select(func.count(s.VocabularyHistory.id))) == 0
    with pytest.raises(s.CatalogError):
        s.confirm_import(db, bad.id, payload_hash=bad.payload_hash, actor="user:1")
    good = s.preview_import(db, vocab.id, [{"canonical_term": "北欧"}, {"canonical_term": "现代"}], actor="user:1")
    assert good.state == "preview" and s.vocabulary(db, vocab.id).revision == 1
    changes = json.loads(good.changes_json)
    assert changes[0]["before"] is None and changes[0]["after"]["id"] is None
    assert changes[0]["after"]["canonical_term"] == "北欧"
    s.confirm_import(db, good.id, payload_hash=good.payload_hash, actor="user:1")
    s.confirm_import(db, good.id, payload_hash=good.payload_hash, actor="user:1")
    assert db.scalar(select(func.count(s.VocabularyTerm.id))) == 2
    assert db.scalar(select(func.count(s.VocabularyHistory.id))) == 2
    later = s.preview_import(db, vocab.id, [{"canonical_term": "中式"}], actor="user:1")
    term(db, vocab.id, "日式")
    with pytest.raises(s.CatalogError, match="重新导入"):
        s.confirm_import(db, later.id, payload_hash=later.payload_hash, actor="user:1")

function ImportChanges({ batch }: { batch: SemanticVocabularyImport }) {
  const hidden = new Set(["id", "vocabulary_id", "revision", "source", "updated_at", "updated_by"])
  return <div className="divide-y rounded-xl border px-4">{batch.changes.map(change => <details key={change.row} className="py-3">
    <summary className="cursor-pointer font-bold">第{change.row}行 · {change.before ? "修改" : "新增"} · {String(change.after.canonical_term || change.after.source_field)}</summary>
    <div className="mt-3 space-y-3 text-xs [overflow-wrap:anywhere]">{Object.entries(change.after).filter(([key, value]) => !hidden.has(key) && JSON.stringify(change.before?.[key]) !== JSON.stringify(value)).map(([key, value]) => <div key={key}><p className="font-bold">{labels[key] || key}</p>{change.before && <p className="mt-1 text-muted-foreground">原值：{show(change.before[key])}</p>}<p className="mt-1">新值：{show(value)}</p></div>)}</div>
  </details>)}</div>
}

===== frontend/src/lib/semantic-catalog-csv.ts =====
/** Excel UTF-8 CSV; numeric identifiers must never silently become a new row. */
export function parseSemanticCatalogCsv(value: string): Record<string, unknown>[] {
  const rows: string[][] = []; let row: string[] = [], field = "", quoted = false, closed = false
  for (let i = 0; i < value.length; i++) {
    const c = value[i]
    if (c === '"') {
      if (quoted && value[i + 1] === '"') { field += '"'; i++ }
      else if (quoted) { quoted = false; closed = true }
      else if (!field && !closed) quoted = true
      else throw new Error("CSV引号位置无效")
    } else if (c === "," && !quoted) { row.push(field); field = ""; closed = false }
    else if ((c === "\n" || c === "\r") && !quoted) {
      if (c === "\r" && value[i + 1] === "\n") i++
      row.push(field); rows.push(row); row = []; field = ""; closed = false
    } else {
      if (closed) throw new Error("CSV结束引号后存在多余字符")
      field += c
    }
  }
  if (quoted) throw new Error("CSV引号未闭合")
  if (field || row.length) { row.push(field); rows.push(row) }
  const headers = rows.shift()?.map(h => h.replace(/^\uFEFF/, "").trim()) || []
  if (!headers.length || headers.some(h => !h) || new Set(headers).size !== headers.length) throw new Error("表头为空或有重复")
  return rows.filter(r => r.some(v => v.trim())).map((r, i) => {
    if (r.length !== headers.length) throw new Error(`第${i + 2}行列数不符`)
    return Object.fromEntries(headers.map((h, n) => {
      if (["id", "expected_revision", "target_term_id"].includes(h)) {
        if (!r[n].trim()) return [h, null]
        const number = Number(r[n])
        if (!/^\d+$/.test(r[n]) || !Number.isSafeInteger(number) || number <= 0) throw new Error(`第${i + 2}行的${h}必须是正整数`)
        return [h, number]
      }
      if (h === "aliases") {
        const parsed: unknown = r[n].trim().startsWith("[") ? JSON.parse(r[n]) : r[n].split("|").filter(Boolean)
        if (!Array.isArray(parsed) || parsed.some(item => typeof item !== "string")) throw new Error(`第${i + 2}行别名必须是文字数组`)
        return [h, parsed]
      }
      return [h, h === "source_value" && r[n] === "" ? null : r[n]]
    }))
  })
}
