"""Task-scoped observation journal. No production state changes or network IO.

Lock order is observer state -> issue -> notification. Sending claims only a
notification and never takes the observer/issue lock after that claim.
"""

def stamp(value=None):
    value = value or datetime.now(timezone.utc)
    if value.tzinfo is None: value = value.replace(tzinfo=timezone.utc)
    return value.astimezone(timezone.utc).isoformat()


class AlertCursor(Base):
    __tablename__ = "workflow_alert_cursors"
    key: Mapped[str] = mapped_column(String(30), primary_key=True)
    workflow_id: Mapped[int] = mapped_column(Integer, default=0)
    run_after: Mapped[int] = mapped_column(Integer, default=0)
    run_ceiling: Mapped[int] = mapped_column(Integer, default=0)
    issue_after: Mapped[int] = mapped_column(Integer, default=0)
    observed_at: Mapped[str] = mapped_column(String(40), default="")


class WorkflowAlert(Base):
    __tablename__ = "workflow_alerts"
    id: Mapped[int] = mapped_column(primary_key=True)
    run_id: Mapped[int] = mapped_column(ForeignKey("production_runs.id", ondelete="RESTRICT"), unique=True)
    inventory_key: Mapped[str] = mapped_column(String(120), index=True)
    workflow_name: Mapped[str] = mapped_column(String(200))
    state: Mapped[str] = mapped_column(String(20), default="open", index=True)
    episode: Mapped[int] = mapped_column(Integer, default=1)
    owner: Mapped[str] = mapped_column(String(120), default="")
    recipient_user_id: Mapped[int | None] = mapped_column(Integer, nullable=True)
    recipient_corp: Mapped[str] = mapped_column(String(100), default="")
    recipient_employee: Mapped[str] = mapped_column(String(100), default="")
    reminder_seconds: Mapped[int] = mapped_column(Integer, default=0)
    notification_reason: Mapped[str] = mapped_column(String(80), default="")
    error_code: Mapped[str] = mapped_column(String(80), default="")
    run_status: Mapped[str] = mapped_column(String(30))
    opened_at: Mapped[str] = mapped_column(String(40), default=stamp)
    observed_at: Mapped[str] = mapped_column(String(40), default=stamp)
    closed_at: Mapped[str | None] = mapped_column(String(40), nullable=True)
    close_reason: Mapped[str] = mapped_column(String(80), default="")


class AlertEvent(Base):
    __tablename__ = "workflow_alert_events"
    id: Mapped[int] = mapped_column(primary_key=True)
    alert_id: Mapped[int] = mapped_column(ForeignKey("workflow_alerts.id", ondelete="RESTRICT"), index=True)
    episode: Mapped[int] = mapped_column(Integer)
    event: Mapped[str] = mapped_column(String(40))
    evidence: Mapped[str] = mapped_column(Text, default="")
    created_at: Mapped[str] = mapped_column(String(40), default=stamp)


class AlertNotification(Base):
    __tablename__ = "workflow_alert_notifications"
    __table_args__ = (UniqueConstraint("alert_id", "episode", "ordinal", name="uq_alert_notification_intent"),)
    id: Mapped[int] = mapped_column(primary_key=True)
    alert_id: Mapped[int] = mapped_column(ForeignKey("workflow_alerts.id", ondelete="RESTRICT"), index=True)
    episode: Mapped[int] = mapped_column(Integer)
    ordinal: Mapped[int] = mapped_column(Integer)
    state: Mapped[str] = mapped_column(String(20), default="pending", index=True)
    task_id: Mapped[str] = mapped_column(String(100), default="")
    error_code: Mapped[str] = mapped_column(String(80), default="")
    created_at: Mapped[str] = mapped_column(String(40), default=stamp)
    updated_at: Mapped[str] = mapped_column(String(40), default=stamp)
    claim_token: Mapped[str] = mapped_column(String(36), default="")

def event(db, alert, name, evidence="", now=None):
    db.add(AlertEvent(alert_id=alert.id, episode=alert.episode, event=name, evidence=evidence, created_at=stamp(now)))


def _identity(db, user_id):
    return db.execute(select(User.id, User.username, EmployeeIdentity.corp_id, EmployeeIdentity.employee_id)
        .join(EmployeeIdentity, EmployeeIdentity.user_id == User.id)
        .where(User.id == user_id, User.is_active.is_(True), EmployeeIdentity.status == "active",
               EmployeeIdentity.verified_at.is_not(None))).first()


def validate_policy(db, details):
    user_id = details.get("notification_user_id", "")
    if user_id and (not user_id.isascii() or not user_id.isdecimal() or len(user_id) > 18 or not _identity(db, int(user_id))):
        return "提醒负责人须为已绑定有效钉钉身份的现有员工"
    if details.get("reminder_seconds", "") not in ("", "0", "3600", "86400"):
        return "提醒频率请选择仅一次、每小时或每天"
    return None


def _new(db, run, inventory, now):
    payload = json.loads(inventory.payload_json); details = payload["details"]
    user_id = details.get("notification_user_id", "")
    identity = _identity(db, int(user_id)) if user_id.isascii() and user_id.isdecimal() and len(user_id) <= 18 else None
    alert = WorkflowAlert(run_id=run.id, inventory_key=inventory.key, workflow_name=inventory.name,
        owner=identity.username if identity else payload.get("owner", ""), run_status=run.status,
        recipient_user_id=identity.id if identity else None, recipient_corp=identity.corp_id if identity else "",
        recipient_employee=identity.employee_id if identity else "", notification_reason="" if identity else "owner_not_bound",
        reminder_seconds=int(details.get("reminder_seconds") or 0), opened_at=stamp(now), observed_at=stamp(now),
        error_code=run.error_code or "")
    db.add(alert); db.flush(); event(db, alert, "opened", f"run:{run.id}", now)
    return alert


def _closed_reason(db, run):
    if run.status == "succeeded": return "run_succeeded"
    if run.status == "canceled": return "run_canceled"
    from .failure_resolution_models import ManualFailureResolution
    if db.scalar(select(ManualFailureResolution.id).where(ManualFailureResolution.production_run_id == run.id,
                  ManualFailureResolution.status == "published").limit(1)):
        return "manual_resolution_published"
    if db.scalar(select(AuditEvent.id).where(AuditEvent.event_key == f"incremental-reassessment-settled:{run.id}").limit(1)):
        return "reassessment_settled"
    return ""


def reconcile_one(db, alert, run, now=None):
    alert.observed_at = stamp(now)
    if run is None or run.run_mode != "incremental" or run.model_execution_mode != "real":
        state, reason = "unknown", "runtime_unavailable"
    else:
        reason = _closed_reason(db, run)
        state = "closed" if reason else "open" if run.status in FAILURES else "recovering"
        alert.run_status = run.status; alert.error_code = run.error_code or ""
    if alert.state != state:
        if alert.state == "closed" and state == "open": alert.episode += 1
        alert.state = state; alert.close_reason = reason
        alert.closed_at = stamp(now) if state == "closed" else None
        event(db, alert, state, reason or f"run:{alert.run_id}:{alert.run_status}", now)
    if state != "open":
        # A submitted message cannot be unsent here. Preserve its eventual receipt.
        for notice in db.scalars(select(AlertNotification).where(AlertNotification.alert_id == alert.id,
                AlertNotification.state == "pending").with_for_update()):
            notice.state = "canceled"; notice.error_code = "issue_not_open"; notice.updated_at = stamp(now)
    else:
        _enqueue(db, alert, now)


def _enqueue(db, alert, now=None):
    if not alert.recipient_user_id: return
    previous = db.scalar(select(AlertNotification).where(AlertNotification.alert_id == alert.id,
        AlertNotification.episode == alert.episode).order_by(AlertNotification.ordinal.desc()).limit(1))
    if previous:
        # Unknown or rejected delivery needs explicit diagnosis, never automatic resend.
        if previous.state == "canceled" and previous.error_code == "issue_not_open":
            db.add(AlertNotification(alert_id=alert.id, episode=alert.episode, ordinal=previous.ordinal + 1,
                created_at=stamp(now), updated_at=stamp(now)))
            return
        if previous.state != "accepted" or not alert.reminder_seconds: return
        due = datetime.fromisoformat(previous.created_at) + timedelta(seconds=alert.reminder_seconds)
        if stamp(now) < stamp(due): return
    db.add(AlertNotification(alert_id=alert.id, episode=alert.episode, ordinal=previous.ordinal + 1 if previous else 1,
        created_at=stamp(now), updated_at=stamp(now)))


def observe(db, *, batch=100, now=None):
    """One bounded discovery page and one bounded open-issue page; caller commits."""
    batch = max(1, min(int(batch), 200)); reserve_sqlite_immediate_transaction(db)
    cursor = db.scalar(select(AlertCursor).where(AlertCursor.key == "observer").with_for_update()
                       .execution_options(populate_existing=True))
    if cursor is None: raise RuntimeError("workflow alert migration is required")
    # Freeze a high-water mark per flow. New arrivals cannot starve older runs
    # whose terminal state changes after an earlier cursor passed them.
    if not cursor.run_ceiling:
        next_flow = db.execute(select(WorkflowDefinition.id, InventoryObject.key)
            .join(InventoryObject, InventoryObject.platform_workflow_key == WorkflowDefinition.workflow_key)
            .where(InventoryObject.status == "active", WorkflowDefinition.id > cursor.workflow_id)
            .order_by(WorkflowDefinition.id).limit(1)).first()
        if next_flow is None:
            cursor.workflow_id = 0
        else:
            cursor.workflow_id = next_flow.id; cursor.run_after = 0
            cursor.run_ceiling = db.scalar(select(func.max(ProductionRun.id)).where(
                ProductionRun.workflow_definition_id == next_flow.id)) or 0
    discovered = 0
    if cursor.run_ceiling:
        inventory = db.scalar(select(InventoryObject).join(WorkflowDefinition,
            InventoryObject.platform_workflow_key == WorkflowDefinition.workflow_key)
            .where(WorkflowDefinition.id == cursor.workflow_id, InventoryObject.status == "active"))
        if inventory:
            candidates = []
            for status in FAILURES:
                candidates.extend(db.execute(select(*RUN_COLUMNS).where(
                    ProductionRun.workflow_definition_id == cursor.workflow_id, ProductionRun.run_mode == "incremental",
                    ProductionRun.model_execution_mode == "real", ProductionRun.status == status,
                    ProductionRun.id > cursor.run_after, ProductionRun.id <= cursor.run_ceiling)
                    .order_by(ProductionRun.id).limit(batch)).all())
            runs = sorted(candidates, key=lambda r: r.id)[:batch]
            existing = {a.run_id: a for a in db.scalars(select(WorkflowAlert).where(
                WorkflowAlert.run_id.in_([r.id for r in runs])).with_for_update())} if runs else {}
            for run in runs:
                alert = existing.get(run.id) or _new(db, run, inventory, now)
                reconcile_one(db, alert, run, now); discovered += 1
            if len(runs) == batch: cursor.run_after = runs[-1].id
            else: cursor.run_ceiling = 0
        else: cursor.run_ceiling = 0
    # SessionLocal disables autoflush. Persist discovery intents before an issue
    # can be revisited below, otherwise its first notification is queued twice.
    db.flush()
    alerts = list(db.scalars(select(WorkflowAlert).where(WorkflowAlert.state != "closed", WorkflowAlert.id > cursor.issue_after)
        .order_by(WorkflowAlert.id).limit(batch).with_for_update()))
    records = {r.id: r for r in db.execute(select(*RUN_COLUMNS).where(ProductionRun.id.in_([a.run_id for a in alerts])))} if alerts else {}
    for alert in alerts: reconcile_one(db, alert, records.get(alert.run_id), now)
    cursor.issue_after = alerts[-1].id if len(alerts) == batch else 0
    cursor.observed_at = stamp(now); db.flush()
    return {"discovered": discovered, "reconciled": len(alerts), "observed_at": cursor.observed_at}

"""Durable single-recipient DingTalk submission; no network inside DB locks."""
from __future__ import annotations

import uuid
from datetime import datetime, timedelta, timezone

import httpx
from sqlalchemy import select
from sqlalchemy.orm import Session

from .database import reserve_sqlite_immediate_transaction
from .dingtalk_identity import DingTalkClient
from .workflow_alerts import (AlertNotification, WorkflowAlert, ProductionRun, RUN_COLUMNS, FAILURES,
                              _identity, _closed_reason, stamp)


class RejectedNotice(Exception):
    pass


class UnknownNotice(Exception):
    pass


class DingTalkNotifier:
    def __init__(self, settings, agent_id, *, transport=None):
        if type(agent_id) is not int or agent_id <= 0: raise ValueError("DINGTALK_AGENT_ID is required")
        self.identity = DingTalkClient(settings, transport=transport)
        self.corp_id = settings.corp_id
        self.agent_id = agent_id
        self.transport = transport

    def send(self, employee_id, text):
        if not employee_id or len(employee_id) > 100 or any(c.isspace() or c == ',' for c in employee_id):
            raise RejectedNotice("invalid_recipient")
        # Access-token failure occurs before any message is submitted.
        try: token = self.identity.app_token()
        except Exception: raise RejectedNotice("dingtalk_auth_unavailable") from None
        try:
            with httpx.Client(transport=self.transport, timeout=8, follow_redirects=False) as client:
                response = client.post("https://oapi.dingtalk.com/topapi/message/corpconversation/asyncsend_v2",
                    params={"access_token": token}, json={"agent_id": self.agent_id, "userid_list": employee_id,
                        "to_all_user": False, "msg": {"msgtype": "text", "text": {"content": text}}})
            if response.status_code != 200: raise UnknownNotice("dingtalk_http_unknown")
            data = response.json()
            if not isinstance(data, dict) or type(data.get("errcode")) is not int:
                raise UnknownNotice("dingtalk_response_unknown")
            if data["errcode"] != 0: raise RejectedNotice("dingtalk_rejected")
            task = data.get("task_id")
            if isinstance(task, bool) or not isinstance(task, (int, str)) or not str(task).isdigit() or len(str(task)) > 100:
                raise UnknownNotice("dingtalk_task_missing")
            return str(task)
        except (httpx.HTTPError, ValueError):
            raise UnknownNotice("dingtalk_submission_unknown") from None


def submit_one(engine, notifier, *, enabled=False, notify_since=None, now=None):
    """Caller must explicitly enable sending and choose its historical cutoff.

    Unknown sends remain blocked even after restart. Accepted means provider
    task queued, not read or delivered. The sender preserves late receipts when
    an observer closes an issue during the external request.
    """
    if not enabled or notify_since is None: return False
    current = stamp(now); expired = stamp(datetime.fromisoformat(current) - timedelta(minutes=5))
    with Session(engine) as db, db.begin():
        reserve_sqlite_immediate_transaction(db)
        # Discover without locks, then lock by primary key. A locking scan of
        # the state index can deadlock with a receipt holding the primary row
        # while updating that index, even at READ COMMITTED.
        stale_id = db.scalar(select(AlertNotification.id).where(AlertNotification.state == "sending",
            AlertNotification.updated_at <= expired).order_by(AlertNotification.id).limit(1))
        stale = db.scalar(select(AlertNotification).where(AlertNotification.id == stale_id)
            .with_for_update(skip_locked=engine.dialect.name == "mysql")) if stale_id is not None else None
        if stale and (stale.state != "sending" or stale.updated_at > expired): stale = None
        if stale:
            stale.state = "unknown"; stale.error_code = "sender_interrupted"; stale.updated_at = current
        notice = db.scalar(select(AlertNotification).where(AlertNotification.state == "pending")
            .order_by(AlertNotification.id).limit(1).with_for_update(skip_locked=engine.dialect.name == "mysql"))
        if notice is None: return bool(stale)
        alert = db.get(WorkflowAlert, notice.alert_id)
        run = db.execute(select(*RUN_COLUMNS).where(ProductionRun.id == alert.run_id)).first()
        reason = ""
        if (alert.state != "open" or alert.episode != notice.episode or run is None or run.status not in FAILURES
                or run.run_mode != "incremental" or run.model_execution_mode != "real" or _closed_reason(db, run)):
            reason = "issue_not_open"
        elif stamp(run.updated_at) < stamp(notify_since): reason = "before_notification_cutoff"
        identity = _identity(db, alert.recipient_user_id)
        if not reason and (identity is None or identity.corp_id != alert.recipient_corp
                or identity.employee_id != alert.recipient_employee or notifier.corp_id != alert.recipient_corp):
            reason = "original_owner_unavailable"
        if reason:
            notice.state = "canceled" if reason in ("issue_not_open", "before_notification_cutoff") else "blocked"
            notice.error_code = reason; notice.updated_at = current
            return True
        token = str(uuid.uuid4()); identity_id, recipient = notice.id, alert.recipient_employee
        message = f"内容中台运行异常：{alert.workflow_name[:120]}，任务 #{alert.run_id}。请在流程登记与关系页的异常列表核查。"
        notice.state = "sending"; notice.claim_token = token; notice.updated_at = current
    try:
        task_id, state, error = notifier.send(recipient, message), "accepted", ""
    except RejectedNotice as exc:
        task_id, state, error = "", "failed", str(exc)[:80]
    except Exception:
        # Never include provider text, URLs, credentials or stack traces.
        task_id, state, error = "", "unknown", "submission_unknown"
    with Session(engine) as db, db.begin():
        reserve_sqlite_immediate_transaction(db)
        row = db.scalar(select(AlertNotification).where(AlertNotification.id == identity_id).with_for_update())
        if row.claim_token == token and row.state in ("sending", "unknown"):
            row.task_id = task_id; row.state = state; row.error_code = error; row.updated_at = stamp(now)
    return True

"""Opt-in observer and sender. Disabled unless an operator configures deployment."""
import argparse
import logging
import os
import time
from datetime import datetime

from sqlalchemy import text
from sqlalchemy.orm import Session

from .workflow_alerts import observe
from .workflow_alert_notifications import DingTalkNotifier, submit_one

logger = logging.getLogger(__name__)


def main():
    parser = argparse.ArgumentParser(); parser.add_argument("--once", action="store_true")
    args = parser.parse_args()
    if os.getenv("WORKFLOW_ALERTS_OBSERVE_ENABLED") != "1": return
    from .database import engine
    from .migrations.runner import MIGRATIONS
    with engine.connect() as connection:
        if connection.scalar(text("SELECT MAX(version) FROM schema_migrations")) != max(m.version for m in MIGRATIONS):
            raise RuntimeError("workflow alert schema is not ready")
    notifier, since = None, None
    if os.getenv("WORKFLOW_ALERTS_SEND_ENABLED") == "1":
        from .dingtalk_identity import identity_settings
        settings = identity_settings()
        if not settings.enabled: raise RuntimeError("DingTalk application is not enabled")
        since = datetime.fromisoformat(os.environ["WORKFLOW_ALERTS_NOTIFY_SINCE"])
        if since.tzinfo is None: raise RuntimeError("notification cutoff requires timezone")
        notifier = DingTalkNotifier(settings, int(os.environ["DINGTALK_AGENT_ID"]))
    while True:
        try:
            with Session(engine) as db, db.begin(): observe(db)
            if notifier: submit_one(engine, notifier, enabled=True, notify_since=since)
        except Exception:
            # Do not leak database URLs or notification credentials.
            logger.error("workflow alert iteration failed; inspect platform records")
            if args.once: raise RuntimeError("workflow alert iteration failed") from None
        if args.once: break
        time.sleep(5)


if __name__ == "__main__": main()

def migrate(connection):
    from ..workflow_alerts import MODELS, AlertCursor, DISCOVERY_INDEX
    metadata = MODELS[0].metadata
    if connection.dialect.name == "mysql":
        from .mysql_baseline import mysql_metadata
        metadata = mysql_metadata()
    for model in MODELS: metadata.tables[model.__tablename__].create(connection, checkfirst=True)
    DISCOVERY_INDEX.create(connection, checkfirst=True)
    table = metadata.tables[AlertCursor.__tablename__]
    if not connection.execute(table.select().where(table.c.key == "observer")).first():
        connection.execute(table.insert().values(key="observer", workflow_id=0, run_after=0, run_ceiling=0, issue_after=0))
