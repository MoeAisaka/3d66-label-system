KINDS = {"designer", "developer", "unknown_entity"}
REGIONS = {"CN_TW", "CN_HK", "CA", "ID", "SG", "JP", "TH", "AU", "US", "GB"}
SCOPES = {"all", "3D", "SU", "inspiration"}
EDIT_FIELDS = {"name", "alias_names", "entity_type", "res_type", "region", "status"}
NORMALIZATION_VERSION = "search-name-v1"


class DictionaryError(ValueError):
    def __init__(self, code, message):
        self.code = code
        super().__init__(message)


class SearchDictionaryState(Base):
    __tablename__ = "search_dictionary_states"
    key: Mapped[str] = mapped_column(String(40), primary_key=True)
    revision: Mapped[int] = mapped_column(Integer, default=0)

    res_type: Mapped[str] = mapped_column(String(40), index=True)
    tags: Mapped[str] = mapped_column(Text, default="[]")
    status: Mapped[str] = mapped_column(String(20), default="active", index=True)

    finished_at: Mapped[str | None] = mapped_column(String(40), nullable=True)


def normalize_name(value):
    if not isinstance(value, str) or not value.strip() or len(value) > 200:
        raise DictionaryError("name_invalid", "词面需为1至40个有效字符")
    raw = unicodedata.normalize("NFKC", value).strip()
    # Detect risky original input BEFORE punctuation removal.
    risk = "".join(c for c in raw if not c.isspace() and not unicodedata.category(c).startswith("C"))
    # Explicit URL/email markers cannot be hidden with whitespace. For bare
    # domains, preserve word boundaries: "St. John" is a name, not "St.John".
    if re.search(r"(?i)(?:https?://|www\.|[^\s@]+@[^\s@]+\.)", risk) or re.search(
        r"(?i)[\w-]+\.[a-z]{2,}(?:\b|/)", raw
    ):
        raise DictionaryError("name_url", "网址、域名和邮箱不能作为词面")
    if any(c in raw for c in ",，、;；\n\r"):
        raise DictionaryError("multiple_names", "每个词面只能填写一个词；别名请分项填写")
    result = "".join(c.lower() if "A" <= c <= "Z" else c for c in raw
                     if c.isalnum() or c in "+-#")
    result = result.strip("-#")
    if not result or not any(c.isalpha() for c in result) or len(result) > 40 or re.fullmatch(r"[a-z]{2,3}\d+", result):
        raise DictionaryError("name_invalid", "词面不能为数字、符号、短字母编号或超过40个有效字符")
    return result


def state(db):
    reserve_sqlite_immediate_transaction(db)
    row = db.scalar(select(SearchDictionaryState).where(SearchDictionaryState.key == "master_brand")
        .with_for_update().execution_options(populate_existing=True))
    if row is None:
        raise DictionaryError("not_initialized", "词典尚未完成数据库升级")
    return row


def entity_payload(row):
    return {"master_brand_id": row.master_brand_id, "revision": row.revision, "version": row.version,
            "name": row.name, "normalized_name": row.normalized_name, "alias_names": json.loads(row.alias_names),
            "entity_type": row.entity_type, "res_type": row.res_type, "region": (json.loads(row.tags) or [None])[0],
            "status": row.status, "source_system": row.source_system, "updated_by": row.updated_by,
            "updated_by_name": row.updated_by_name, "updated_at": row.updated_at, "dt": row.dt}


def clean_entity(value):
    name = value.get("name")
    normalized = normalize_name(name)
    aliases = value.get("alias_names", [])
    if not isinstance(aliases, list) or len(aliases) > 50:
        raise DictionaryError("aliases_invalid", "别名需为不超过50项的数组")
    keys = [normalized] + [normalize_name(a) for a in aliases]
    if len(keys) != len(set(keys)):
        raise DictionaryError("duplicate_alias", "主名与别名不能归一成重复词")
    if value.get("entity_type", "unknown_entity") not in KINDS or value.get("res_type", "all") not in SCOPES:
        raise DictionaryError("classification_invalid", "请选择有效实体类型和素材类型")
    if value.get("region") is not None and value["region"] not in REGIONS:
        raise DictionaryError("region_invalid", "请选择已登记地区或留空")
    if value.get("status", "active") not in {"active", "inactive"}:
        raise DictionaryError("status_invalid", "仅支持启用或停用")
    return {"name": name.strip(), "normalized_name": normalized, "alias_names": [a.strip() for a in aliases],
            "entity_type": value.get("entity_type", "unknown_entity"), "res_type": value.get("res_type", "all"),
            "region": value.get("region"), "status": value.get("status", "active")}


def preview(db, rows, *, actor, actor_name="", action="edit"):
    if not isinstance(rows, list) or not 1 <= len(rows) <= 500 or len(canonical(rows).encode()) > 2 * 1024 * 1024:
        raise DictionaryError("import_limit", "每批1至500行，内容不得超过2MiB")
    root = state(db)
    changes, errors, seen_ids, seen_keys, texts = [], [], set(), {}, set()
    for index, row in enumerate(rows, 1):
        try:
            if not isinstance(row, dict) or set(row) - EDIT_FIELDS - {"master_brand_id", "expected_revision"}:
                raise DictionaryError("columns_invalid", "含未知列或只读列")
            identity = row.get("master_brand_id")
            before = None
            if identity is not None:
                if type(identity) is not int or identity <= 0 or identity in seen_ids:
                    raise DictionaryError("identity_invalid", "实体编号无效或同一实体重复出现")
                existing = db.scalar(select(SearchEntity).where(SearchEntity.master_brand_id == identity)
                    .with_for_update().execution_options(populate_existing=True))
                if existing is None:
                    raise DictionaryError("not_found", "实体不存在，不能使用本批次临时编号")
                before = entity_payload(existing)
                if type(row.get("expected_revision")) is not int or row["expected_revision"] != existing.revision:
                    raise DictionaryError("revision_conflict", "实体已被修改，请重新读取")
                seen_ids.add(identity)
            elif row.get("expected_revision") is not None:
                raise DictionaryError("identity_invalid", "新增实体不能填写旧修订号")
            after = clean_entity(({k: before[k] for k in EDIT_FIELDS} if before else {}) | {k: v for k, v in row.items() if k in EDIT_FIELDS})
            keys = [after["normalized_name"]] + [normalize_name(a) for a in after["alias_names"]]
            for key in keys:
                pair = (after["res_type"], digest(key))
                other = db.scalar(select(SearchEntityKey.entity_id).where(SearchEntityKey.res_type == pair[0], SearchEntityKey.key_hash == pair[1]).with_for_update())
                if other is not None and other != identity or pair in seen_keys:
                    raise DictionaryError("name_conflict", "主名或别名已被其他实体使用（包含停用项）")
            for key in keys:
                seen_keys[(after["res_type"], digest(key))] = index
            changed_text = before is None or before["name"] != after["name"] or before["alias_names"] != after["alias_names"]
            reenable = before is not None and before["status"] == "inactive" and after["status"] == "active"
            if changed_text or reenable:
                texts.update([after["name"], *after["alias_names"]])
            changes.append({"row": index, "before": before, "after": after, "action": action})
        except DictionaryError as exc:
            errors.append({"row": index, "code": exc.code, "message": str(exc)})
    if len(texts) > 5000:
        errors.append({"row": 0, "code": "safety_limit", "message": "待安全校验的不同词面超过5000个，请拆分批次"})
    payload_hash = digest({"revision": root.revision, "changes": changes, "normalization": NORMALIZATION_VERSION})
    batch = SearchDictionaryBatch(id=uuid.uuid4().hex, catalog_revision=root.revision,
        changes_json=canonical(changes), payload_hash=payload_hash, errors_json=canonical(errors),
        state="invalid" if errors else "checking" if texts else "ready", actor=actor, actor_name=actor_name[:100],
        expires_at=(datetime.now(timezone.utc) + timedelta(hours=24)).isoformat())
    db.add(batch); db.flush()
    if not errors:
        for text in sorted(texts):
            db.add(SearchDictionarySafety(batch_id=batch.id, content=text, content_hash=digest(text),
                request_id="search-" + uuid.uuid4().hex, state="queued"))
    db.flush()
    return batch


def confirm(db, batch_id, *, payload_hash, actor):
    root = state(db)
    batch = db.scalar(select(SearchDictionaryBatch).where(SearchDictionaryBatch.id == batch_id)
        .with_for_update().execution_options(populate_existing=True))
    if batch is None:
        raise DictionaryError("not_found", "批次不存在")
    if batch.actor != actor or batch.payload_hash != payload_hash:
        raise DictionaryError("confirmation_conflict", "请由原操作人确认当前预览")
    if batch.state == "completed":
        return batch
    if batch.expires_at <= now() or batch.catalog_revision != root.revision:
        raise DictionaryError("revision_conflict", "词典已变化或预览已过期，请重新预览")
    checks = db.scalars(select(SearchDictionarySafety).where(SearchDictionarySafety.batch_id == batch.id)
        .with_for_update().execution_options(populate_existing=True)).all()
    if batch.state != "ready" or any(row.state != "passed" for row in checks):
        raise DictionaryError("safety_incomplete", "全部校验通过后才能整批写入")
    changes = json.loads(batch.changes_json)
    if digest({"revision": batch.catalog_revision, "changes": changes, "normalization": NORMALIZATION_VERSION}) != batch.payload_hash:
        raise DictionaryError("payload_corrupt", "预览内容摘要不一致")
    checked = {r.content_hash for r in checks}
    # Revalidate complete content before changing any application row.
    for change in changes:
        after = clean_entity(change["after"])
        before = change["before"]
        if before is None or before["name"] != after["name"] or before["alias_names"] != after["alias_names"] or (before["status"] == "inactive" and after["status"] == "active"):
            if not {digest(x) for x in [after["name"], *after["alias_names"]]}.issubset(checked):
                raise DictionaryError("safety_incomplete", "内容未完成安全校验")
        if before:
            existing = db.scalar(select(SearchEntity).where(SearchEntity.master_brand_id == before["master_brand_id"])
                .with_for_update().execution_options(populate_existing=True))
            if existing is None or entity_payload(existing) != before:
                raise DictionaryError("revision_conflict", "原实体已变化，请重新预览")
    root.revision += 1
    version = datetime.now(timezone.utc).strftime("%Y%m%d") + "-" + str(root.revision).zfill(2)
    for change in changes:
        before, after = change["before"], change["after"]
        row = db.get(SearchEntity, before["master_brand_id"]) if before else SearchEntity(revision=0, source_system="mid_new")
        db.add(row)
        row.name = after["name"]; row.normalized_name = after["normalized_name"]
        row.alias_names = canonical(after["alias_names"]); row.entity_type = after["entity_type"]
        row.res_type = after["res_type"]; row.tags = canonical([after["region"]] if after["region"] else [])
        row.status = after["status"]; row.updated_by = actor; row.updated_at = now()
        row.updated_by_name = batch.actor_name
        row.dt = row.updated_at[:10]; row.version = version; row.revision += 1
        db.flush()
        db.execute(delete(SearchEntityKey).where(SearchEntityKey.entity_id == row.master_brand_id))
        for key in [after["normalized_name"], *[normalize_name(a) for a in after["alias_names"]]]:
            db.add(SearchEntityKey(entity_id=row.master_brand_id, res_type=row.res_type, normalized_value=key, key_hash=digest(key)))
        db.add(SearchEntityHistory(entity_id=row.master_brand_id, revision=row.revision, batch_id=batch.id,
            snapshot_json=canonical(entity_payload(row)), actor=actor))
    batch.state = "completed"; batch.completed_at = now(); db.flush()
    return batch


def restore_preview(db, history_id, *, actor, actor_name=""):
    state(db)
    history = db.get(SearchEntityHistory, history_id)
    if history is None:
        raise DictionaryError("not_found", "历史记录不存在")
    row = db.scalar(select(SearchEntity).where(SearchEntity.master_brand_id == history.entity_id)
        .with_for_update().execution_options(populate_existing=True))
    if row is None:
        raise DictionaryError("not_found", "实体不存在")
    old = json.loads(history.snapshot_json)
    return preview(db, [{k: old[k] for k in EDIT_FIELDS} | {"master_brand_id": row.master_brand_id,
        "expected_revision": row.revision}], actor=actor, actor_name=actor_name, action="restore")

===== backend/app/search_dictionary_api.py =====
"""Authenticated employee API for application-side search dictionaries."""
from __future__ import annotations

import csv
import io
import json
import os
from typing import Literal

from fastapi import APIRouter, Depends, HTTPException, Query
from fastapi.responses import Response
from pydantic import BaseModel, ConfigDict, Field
from sqlalchemy import func, select, true
from sqlalchemy.exc import IntegrityError
from sqlalchemy.orm import Session, aliased

from .authz import has_permission
from .database import get_db
from .safety_provider import provider_configuration
from . import search_dictionary as s


class EditRow(BaseModel):
    model_config = ConfigDict(extra="forbid", strict=True)
    master_brand_id: int | None = Field(default=None, gt=0)
    expected_revision: int | None = Field(default=None, gt=0)
    name: str | None = Field(default=None, max_length=200)
    alias_names: list[str] | None = Field(default=None, max_length=50)
    entity_type: Literal["designer", "developer", "unknown_entity"] | None = None
    res_type: Literal["all", "3D", "SU", "inspiration"] | None = None
    region: str | None = Field(default=None, max_length=20)
    status: Literal["active", "inactive"] | None = None


class Preview(BaseModel):
    model_config = ConfigDict(extra="forbid")
    rows: list[EditRow] = Field(min_length=1, max_length=500)


class Confirm(BaseModel):
    model_config = ConfigDict(extra="forbid")
    payload_hash: str = Field(pattern=r"^[0-9a-f]{64}$")
    confirmed: Literal[True]


def build_search_dictionary_router(current_user):
    router = APIRouter(tags=["search-dictionaries"])

    def writer(user):
        if not has_permission(user, "drafts:write"):
            raise HTTPException(403, "需要系统管理员或产研权限")
        return "user:" + str(user.id)

    def execute(db, operation):
        try:
            result = operation(); db.commit(); return result
        except s.DictionaryError as exc:
            db.rollback()
            raise HTTPException(404 if exc.code == "not_found" else 409, {"code": exc.code, "message": str(exc)}) from None
        except IntegrityError:
            db.rollback()
            raise HTTPException(409, "词典已变化或主名/别名重复，请重新预览") from None

    def summary(db, batch):
        checks = db.execute(select(s.SearchDictionarySafety.state, func.count()).where(
            s.SearchDictionarySafety.batch_id == batch.id).group_by(s.SearchDictionarySafety.state)).all()
        return {"id": batch.id, "state": batch.state, "actor": batch.actor, "payload_hash": batch.payload_hash,
                "changes": json.loads(batch.changes_json), "errors": json.loads(batch.errors_json),
                "safety_counts": dict(checks), "expires_at": batch.expires_at, "completed_at": batch.completed_at}

    def filtered(q, match, status, res_type, entity_type):
        query = select(s.SearchEntity)
        if status: query = query.where(s.SearchEntity.status == status)
        if res_type: query = query.where(s.SearchEntity.res_type == res_type)
        if entity_type: query = query.where(s.SearchEntity.entity_type == entity_type)
        if q:
            # Literal keyword matching, including aliases; never SQL wildcards.
            try: normalized = s.normalize_name(q)
            except s.DictionaryError: return query.where(s.SearchEntity.master_brand_id < 0)
            key = s.SearchEntityKey.normalized_value
            sub = select(s.SearchEntityKey.entity_id).where(key == normalized if match == "exact" else key.contains(normalized, autoescape=True))
            query = query.where(s.SearchEntity.master_brand_id.in_(sub))
        return query

    @router.get("/api/search-dictionary/config")
    def configuration(user=Depends(current_user)):
        provider = provider_configuration()
        return {"can_edit": has_permission(user, "drafts:write"), "safety_configured": provider["configured"],
                "safety_enabled": os.getenv("SEARCH_DICTIONARY_SAFETY_ENABLED", "0") == "1",
                "synonym_status": "pending_direction_contract", "external_sync_status": "not_connected",
                "normalization_version": s.NORMALIZATION_VERSION}

    @router.get("/api/search-dictionary/entities")
    def listing(q: str = Query(default="", max_length=200), match: Literal["contains", "exact"] = "contains",
                status: Literal["active", "inactive", ""] = "", res_type: str = "", entity_type: str = "",
                offset: int = Query(default=0, ge=0), limit: int = Query(default=24, ge=1, le=100),
                user=Depends(current_user), db: Session = Depends(get_db)):
        query = filtered(q, match, status, res_type, entity_type)
        total = db.scalar(select(func.count()).select_from(query.subquery()))
        return {"total": total, "items": [s.entity_payload(r) for r in db.scalars(query.order_by(s.SearchEntity.master_brand_id.desc()).offset(offset).limit(limit))]}

    @router.get("/api/search-dictionary/export")
    def export(q: str = Query(default="", max_length=200), match: Literal["contains", "exact"] = "contains",
               status: Literal["active", "inactive", ""] = "", res_type: str = "", entity_type: str = "",
               format: Literal["json", "csv"] = "json", user=Depends(current_user), db: Session = Depends(get_db)):
        def snapshot():
            # One read statement gives the revision and rows the same snapshot,
            # even under READ COMMITTED, without blocking dictionary writers.
            entity = aliased(s.SearchEntity, filtered(q, match, status, res_type, entity_type).subquery())
            records = db.execute(select(s.SearchDictionaryState.revision, entity)
                .select_from(s.SearchDictionaryState).outerjoin(entity, true())
                .where(s.SearchDictionaryState.key == "master_brand")
                .order_by(entity.master_brand_id).execution_options(populate_existing=True)).all()
            if not records:
                raise s.DictionaryError("not_initialized", "词典尚未完成数据库升级")
            return records[0][0], [s.entity_payload(row) for _, row in records if row is not None]
        revision, rows = execute(db, snapshot)  # release transaction before serialization
        if format == "json":
            return {"schema_version": "search-entities-v1", "catalog_revision": revision, "normalization_version": s.NORMALIZATION_VERSION,
                    "row_count": len(rows), "items": rows, "external_sync_status": "not_connected"}
        cols = ["master_brand_id", "expected_revision", "name", "alias_names", "entity_type", "res_type", "region", "status", "_text_encoding"]
        stream = io.StringIO(newline=""); output = csv.DictWriter(stream, fieldnames=cols); output.writeheader()
        for row in rows:
            row["expected_revision"] = row["revision"]
            values = {k: row.get(k) for k in cols}; values["alias_names"] = json.dumps(values["alias_names"], ensure_ascii=False)
            # Explicit format marker keeps exported escapes reversible without
            # stripping genuine apostrophes from manually authored CSV files.
            values["_text_encoding"] = "apostrophe-v1"
            for key, value in values.items():
                if isinstance(value, str) and value.startswith(("'", "=", "+", "-", "@", "\t", "\r")):
                    values[key] = "'" + value
            output.writerow(values)
        return Response("\ufeff" + stream.getvalue(), media_type="text/csv; charset=utf-8",
            headers={"Content-Disposition": 'attachment; filename="search-entities.csv"', "X-Catalog-Revision": str(revision)})

    @router.post("/api/search-dictionary/batches/preview")
    def preview(body: Preview, user=Depends(current_user), db: Session = Depends(get_db)):
        actor = writer(user)
        return execute(db, lambda: summary(db, s.preview(db, [r.model_dump(exclude_unset=True) for r in body.rows], actor=actor, actor_name=user.display_name or user.username)))

    @router.get("/api/search-dictionary/batches")

"""Opt-in moderation worker. No DB locks survive a provider call."""
from __future__ import annotations

import json
import os
import time
from datetime import datetime, timedelta, timezone

from sqlalchemy import select
from sqlalchemy.orm import Session, aliased

from .database import reserve_sqlite_immediate_transaction
from .search_dictionary import SearchDictionaryBatch as Batch, SearchDictionarySafety as Check, now, canonical
from .safety_provider import AliSafetyProvider, provider_configuration
from .safety_rules import DEFAULT_RULES, SafetyError, interpret


def summarize(db, batch):
    states = list(db.scalars(select(Check.state).where(Check.batch_id == batch.id).with_for_update()))
    if states and all(s == "passed" for s in states):
        batch.state = "ready"
    elif "rejected" in states:
        batch.state = "rejected"
    elif "unknown" in states:
        batch.state = "unknown"
    elif "failed" in states:
        batch.state = "failed"
    else:
        batch.state = "checking"


def tick(engine, *, provider=None):
    if provider is None:
        if os.getenv("SEARCH_DICTIONARY_SAFETY_ENABLED", "0") != "1" or not provider_configuration()["configured"]:
            return False
        provider = AliSafetyProvider()
    with Session(engine, expire_on_commit=False) as db:
        reserve_sqlite_immediate_transaction(db)
        # A crashed submission cannot be billed again silently. Known results
        # may finish; expired 5-minute attempts are explicitly uncertain.
        cutoff = (datetime.now(timezone.utc) - timedelta(minutes=5)).isoformat()
        stale_id = db.execute(select(Check.id, Check.batch_id).where(Check.state == "running", Check.started_at < cutoff)
            .order_by(Check.id).limit(1)).first()
        if stale_id:
            batch = db.scalar(select(Batch).where(Batch.id == stale_id.batch_id).with_for_update(skip_locked=True))
            if batch is None: db.rollback(); return False
            stale = db.scalar(select(Check).where(Check.id == stale_id.id).with_for_update().execution_options(populate_existing=True))
            if stale.state == "running" and stale.started_at < cutoff:
                stale.state = "unknown"; stale.error_code = "worker_result_unknown"; stale.finished_at = now()
                db.flush(); summarize(db, batch)
            db.commit(); return True
        running = aliased(Check)
        candidate = db.execute(select(Check.id, Check.batch_id).join(Batch, Batch.id == Check.batch_id).where(
            Check.state == "queued", Batch.state == "checking", Batch.expires_at > now(),
            ~select(running.id).where(running.batch_id == Batch.id, running.state == "running").exists())
            .order_by(Check.id).limit(1)).first()
        if candidate is None:
            db.rollback(); return False
        batch = db.scalar(select(Batch).where(Batch.id == candidate.batch_id).with_for_update(skip_locked=True).execution_options(populate_existing=True))
        if batch is None or batch.state != "checking" or batch.expires_at <= now(): db.rollback(); return False
        # Only one paid request per batch may be in flight. A crashed/uncertain
        # request blocks that batch's remaining spend, not other batches.
        if db.scalar(select(Check.id).where(Check.batch_id == batch.id, Check.state == "running").limit(1).with_for_update()) is not None:
            db.rollback(); return False
        check = db.scalar(select(Check).where(Check.id == candidate.id).with_for_update().execution_options(populate_existing=True))
        if check.state != "queued": db.rollback(); return False
        check.state = "running"; check.started_at = now()
        identity, batch_id, request_id, content = check.id, check.batch_id, check.request_id, check.content
        db.commit()
    result, code, result_state = {}, "", "failed"
    try:
        raw = provider.moderate_text(content, request_id, "comment_detection_pro")
        decision = interpret(raw, DEFAULT_RULES, data_id=request_id)
        result = {"interpretation": decision, "service": "comment_detection_pro", "rules": DEFAULT_RULES}
        result_state = "passed" if decision["verdict"] == "pass" else "rejected"
    except SafetyError as exc:
        code = exc.code
        result_state = "unknown" if exc.submitted else "failed"
    except Exception:
        # Never persist SDK exception strings (credentials/URLs/user text).
        code = "provider_result_unknown"; result_state = "unknown"
    with Session(engine) as db:
        reserve_sqlite_immediate_transaction(db)
        batch = db.scalar(select(Batch).where(Batch.id == batch_id).with_for_update())
        check = db.scalar(select(Check).where(Check.id == identity).with_for_update())
        if check.state != "running" or check.request_id != request_id:
            db.rollback(); return True
        check.state = result_state; check.result_json = canonical(result)
        check.error_code = code; check.finished_at = now()
        db.flush(); summarize(db, batch); db.commit()
    return True


def main():
    from .database import engine
    from .migrations.runner import MIGRATIONS
    from sqlalchemy import text
    if os.getenv("SEARCH_DICTIONARY_SAFETY_ENABLED", "0") != "1":
        raise SystemExit("search dictionary moderation is disabled")
    with engine.connect() as connection:
        if connection.scalar(text("select max(version) from schema_migrations")) != max(m.version for m in MIGRATIONS):
            raise SystemExit("schema upgrade required")
    while True:
        if not tick(engine):
            time.sleep(2)


if __name__ == "__main__":
    main()

def migrate(connection):
    from ..search_dictionary import (SearchDictionaryState, SearchEntity, SearchEntityKey,
                                    SearchEntityHistory, SearchDictionaryBatch, SearchDictionarySafety)
    models = (SearchDictionaryState, SearchEntity, SearchEntityKey, SearchEntityHistory,
              SearchDictionaryBatch, SearchDictionarySafety)
    metadata = models[0].metadata
    if connection.dialect.name == "mysql":
        from .mysql_baseline import mysql_metadata
        metadata = mysql_metadata()
    for model in models:
        table = metadata.tables[model.__tablename__]
        if inspect(connection).has_table(table.name):
            actual = {c["name"] for c in inspect(connection).get_columns(table.name)}
            if not set(table.columns.keys()).issubset(actual):
                raise RuntimeError("existing search dictionary requires explicit migration: " + table.name)
        table.create(connection, checkfirst=True)
    table = metadata.tables[SearchDictionaryState.__tablename__]
    if connection.execute(table.select().where(table.c.key == "master_brand")).first() is None:
        connection.execute(table.insert().values(key="master_brand", revision=0))

===== frontend/src/lib/search-dictionary-import.ts =====
const editable = ["master_brand_id", "expected_revision", "name", "alias_names", "entity_type", "res_type", "region", "status"]

export function parseSearchDictionaryImport(text: string): Record<string, unknown>[] {
  const trimmed = text.replace(/^\uFEFF/, "").trim()
  if (trimmed.startsWith("[") || trimmed.startsWith("{")) {
    const data: unknown = JSON.parse(trimmed)
    if (Array.isArray(data)) return data
    if (data && typeof data === "object" && "schema_version" in data && data.schema_version === "search-entities-v1" && "items" in data && Array.isArray(data.items)) {
      return data.items.map(row => Object.fromEntries(editable.filter(k => k !== "expected_revision").map(k => [k, row[k]]).concat([["expected_revision", row.revision]])))
    }
    throw new Error("请导入行数组或本页导出的JSON文件")
  }
  const rows: string[][] = []; let row: string[] = [], field = "", quoted = false, closed = false
  for (let i = 0; i < trimmed.length; i++) {
    const c = trimmed[i]
    if (c === '"') {
      if (quoted && trimmed[i + 1] === '"') { field += '"'; i++ }
      else if (quoted) { quoted = false; closed = true }
      else if (!field && !closed) quoted = true
      else throw new Error("CSV引号位置无效")
    } else if (c === "," && !quoted) { row.push(field); field = ""; closed = false }
    else if ((c === "\n" || c === "\r") && !quoted) {
      if (c === "\r" && trimmed[i + 1] === "\n") i++
      row.push(field); rows.push(row); row = []; field = ""; closed = false
    } else { if (closed) throw new Error("CSV引号结束后有多余字符"); field += c }
  }
  if (quoted) throw new Error("CSV引号未闭合")
  if (field || row.length) { row.push(field); rows.push(row) }
  const headers = rows.shift()?.map(x => x.trim()) || []
  if (!headers.length || new Set(headers).size !== headers.length || headers.some(h => !editable.includes(h) && h !== "_text_encoding")) throw new Error("表头为空、重复或包含不支持的列")
  return rows.filter(r => r.some(v => v.trim())).map((r, i) => {
    if (r.length !== headers.length) throw new Error(`第${i + 2}行列数不一致`)
    const encoding = headers.indexOf("_text_encoding")
    if (encoding >= 0 && r[encoding] !== "apostrophe-v1") throw new Error(`第${i + 2}行导出转义格式无效`)
    return Object.fromEntries(headers.flatMap<[string, unknown]>((h, n) => {
      if (h === "_text_encoding") return []
      let value = r[n]
      if (encoding >= 0 && /^'['=+@\-\t\r]/.test(value)) value = value.slice(1)
      if (["master_brand_id", "expected_revision"].includes(h)) {
        if (!value.trim()) return [[h, null]]
        const id = Number(value)
        if (!/^\d+$/.test(value) || !Number.isSafeInteger(id) || id <= 0) throw new Error(`第${i + 2}行${h}必须是正整数`)
        return [[h, id]]
      }
      if (h === "alias_names") {
        const aliases: unknown = value.trim().startsWith("[") ? JSON.parse(value) : value.split("|").filter(Boolean)
        if (!Array.isArray(aliases) || aliases.some(x => typeof x !== "string")) throw new Error(`第${i + 2}行别名需为文字数组或用竖线分隔`)
        return [[h, aliases]]
      }
      return [[h, h === "region" && !value ? null : value]]
    }))
  })
}

def effective_role(user: User) -> str:
    if user.is_admin:
        return "admin"
    role = (user.role or "viewer").strip().lower()
    return role if role in ROLE_PERMISSIONS and role != "admin" else "viewer"


def has_permission(user: User, permission: str) -> bool:
    permissions = ROLE_PERMISSIONS[effective_role(user)]
    return "*" in permissions or permission in permissions


def writable_categories(user: User) -> list[str] | None:
    role = effective_role(user)
    if role in {"admin", "product_rd"}:
        return None
    if role != "category_manager":
        return []
    try:
        values = json.loads(user.category_keys_json or "[]")
    except (ValueError, TypeError):
        return []
    return sorted(set(item for item in values if isinstance(item, str))) if isinstance(values, list) else []


def require_categories(user: User, categories: set[str | None]) -> None:
    if effective_role(user) == "admin":
        return
    if not categories or None in categories or "" in categories:
        raise HTTPException(403, "无法确认对象类目，请联系系统管理员")
    allowed = writable_categories(user)
    if allowed is not None and not categories.issubset(set(allowed)):
        raise HTTPException(403, "包含未授权类目的对象，本次操作未执行")


def lock_access_users(db: Session, user_ids: list[int]) -> None:
    reserve_sqlite_immediate_transaction(db)
    db.execute(select(User.id).where(User.id.in_(sorted(set(user_ids))))
               .order_by(User.id).with_for_update()).all()


def protect_write_transaction(db: Session, user: User) -> None:
    """Recheck after legacy service commits; serialize with account revocation.

    Revocation locks the same actor row: it either precedes this transaction
    (rejected) or follows its commit. Already approved queued work continues.
    """
    db.info["employee_write_guard"] = (user.id, user.permission_revision)
    _check_write_transaction(db)


def _check_write_transaction(db: Session) -> None:
    identity = db.info.get("employee_write_guard")
    if not identity:
        return
    if db.get_transaction() is not None and db.info.get("employee_guard_transaction") is db.get_transaction():
        return
    user_id, revision = identity
    lock_access_users(db, [user_id])
    row = db.execute(select(User.is_active, User.permission_revision).where(User.id == user_id)).one_or_none()
    if row is None or not row.is_active or row.permission_revision != revision:
        raise HTTPException(403, "权限已变更，请刷新后重试")
    from .employee_access import recheck_employee_objects
    recheck_employee_objects(db, db.get(User, user_id))
    db.info["employee_guard_transaction"] = db.get_transaction()


@event.listens_for(Session, "before_flush")
def _before_employee_flush(db: Session, _context, _instances) -> None:
    _check_write_transaction(db)


@event.listens_for(Session, "before_commit")
def _before_employee_commit(db: Session) -> None:
    _check_write_transaction(db)


def require_permission(permission: str):
    from .main import current_user

    def dependency(user: User = Depends(current_user)) -> User:
        if not has_permission(user, permission):
            raise HTTPException(status_code=403, detail=f"缺少权限：{permission}")
        return user

    return dependency

def check_origin(request: Request) -> None:
    """Cookie writes cannot originate from another site, including login CSRF."""
    if request.method in {"GET", "HEAD", "OPTIONS"}:
        return
    if request.headers.get("sec-fetch-site") == "cross-site":
        raise HTTPException(403, "不接受跨站操作")
    source = request.headers.get("origin") or request.headers.get("referer")
    if source:
        supplied, actual = urlsplit(source), urlsplit(str(request.base_url))
        from .dingtalk_identity import identity_settings
        config = identity_settings()
        canonical = urlsplit(config.redirect_uri)
        # A configured HTTPS public origin also works behind the private HTTP
        # gateway; arbitrary forwarded headers never supply this authority.
        if config.ready and actual.netloc == canonical.netloc:
            actual = canonical
        def origin(value):
            try:
                return value.scheme, value.hostname, value.port or (443 if value.scheme == "https" else 80)
            except ValueError:
                return None
        if origin(supplied) is None or origin(supplied) != origin(actual):
            raise HTTPException(403, "操作来源不匹配")


def _row(db: Session, model, identity):
    if identity is None:
        raise HTTPException(422, "缺少业务对象标识")
    # Pin ownership until this business transaction commits. A concurrent
    # administrator category change must order before or after the write.
    row = db.scalar(select(model).where(model.id == identity).with_for_update()
        .execution_options(populate_existing=True))
    if row is None:
        raise HTTPException(404, "业务对象不存在")
    return row


def _asset_categories(db: Session, ids: list[int]) -> set[str | None]:
    if not ids or len(ids) > 5000 or any(type(i) is not int or i <= 0 for i in ids):
        raise HTTPException(422, "素材范围无效")
    rows = db.execute(select(m.Asset.id, m.Asset.category_key).where(m.Asset.id.in_(ids))
        .order_by(m.Asset.id).with_for_update()).all()
    if len(rows) != len(set(ids)):
        raise HTTPException(404, "包含不存在的素材")
    return {r.category_key for r in rows}


def _texture_asset_categories(db, identifiers):
    # All roles share one ascending asset lock order, including large imports.
    ids = sorted(set(identifiers)); categories = set()
    for offset in range(0, len(ids), 5000):
        categories |= _asset_categories(db, ids[offset:offset + 5000])
    return categories


def _evaluation(db: Session, identity) -> set[str | None]:
    row = _row(db, m.EvaluationResult, identity)
    categories = {_row(db, m.Asset, row.asset_id).category_key}
    if row.production_run_id:
        categories.add(_row(db, m.ProductionRun, row.production_run_id).category_key)
    return categories

        row = _row(db, m.MaterialPackage, body.get("material_package_id"))

            .where(m.MaterialPackageItem.package_id == row.id).order_by(m.Asset.id).with_for_update()).all())
    if resolver in {"production_run", "production_evaluation"}:
        row = _row(db, m.EvaluationProductionRun, params.get("run_id"))
        categories = {row.category_key}
        if resolver == "production_evaluation":
            categories |= _evaluation(db, params.get("evaluation_id"))
        return categories
    raise HTTPException(403, "该操作仅系统管理员可执行")


def authorize_employee_write(request: Request, db: Session, user: m.User, body: dict[str, Any]) -> None:
    if request.method in {"GET", "HEAD", "OPTIONS"}:
        return
    check_origin(request)
    if effective_role(user) == "admin":
        return
    path = getattr(request.scope.get("route"), "path", "").rstrip("/")
    policy = WRITE_POLICIES.get((request.method, path))
    if policy is None or not has_permission(user, policy[0]):
        raise HTTPException(403, "当前角色不能执行此操作")
    resolver = policy[1]
    if resolver in {"self", "read"}:
        return
    if resolver == "draft":
        if body.get("status", "draft") not in {"draft", "candidate"} or body.get("active") is True:
            raise HTTPException(403, "产研只能编辑草稿，正式启用由系统管理员执行")
        return
    categories = object_categories(db, resolver, request.path_params, body)
    require_categories(user, categories)
    # A supplied category is never authoritative, but must also agree with the
    # source objects so it cannot redirect a write into another category.
    if body.get("category_key") and body["category_key"] not in categories:
        raise HTTPException(403, "请求类目与实际对象不一致")
    db.info["employee_object_guard"] = (resolver, dict(request.path_params), body)


def recheck_employee_objects(db: Session, user: m.User) -> None:
    guard = db.info.get("employee_object_guard")
    if guard:
        resolver, params, body = guard
        categories = object_categories(db, resolver, params, body)
        require_categories(user, categories)
        if body.get("category_key") and body["category_key"] not in categories:
            raise HTTPException(403, "对象类目已变化，请刷新后重试")

# 搜索词管理：名师名司先落地

范围依据：TPWL-21395、0818后台需求（revision1600）、0817字段说明，以及用户后续确认。先实施名师名司词典和安全校验、批量确认、版本历史；同义词/错别词方向与编号口径未核对的部分保留待接，不猜测现有数仓DDL。不改旧名师名司内容分发后台，不做AI提炼。

应用表使用search_candidate_master_brand_dictionary；稳定master_brand_id改名不变，当前表及时更新，完整历史另存。管理员和产研可维护，其他员工只读；只停用、可恢复，不提供删除。安全服务通过不等于人工审批，仍由发起人确认本批次写入；导入整批校验、差异预览、整批生效，没有逐条部分成功。

清洗与知识图谱词表分开。原名保留，生成NFKC、ASCII小写、去词内空白和装饰字符的匹配名；先识别URL/域名/邮箱再去标点，防止网址洗成普通词。业务符号C++、3D-Max保留。名师名司、房企地产、未细分实体对应designer、developer、unknown_entity。地区是可空单选，映射为已列地区代码。跨素材类型允许独立维护，同范围主名/别名不得重复，停用仍占用身份。

每次新增、改名、改别名、恢复启用都先通过阿里comment_detection_pro校验，只有明确pass可写入。校验在独立worker执行，网络等待不持有应用表事务；没有凭据时明确待配置，不绕过。校验请求已发但结果不明时标记unknown，不静默重发。编辑结果和审核证据绑定摘要，提交前重新校验词典版本、原修订、重名和操作者。单纯停用或更改分类/地区不改文字，不重复付费审核；历史恢复同样走差异预览和必要安全校验。

每批最多500条、最多5000个不同待审核文本、正文2MiB。核心数据及审计长期保留，分页24条；导出当前筛选全部页，包含版本及停用状态。后台更新应用MySQL表，数仓每日同步由外部流程执行；未收到外部回执不显示“已同步”或承诺搜索生效时间。

首期生产开关默认关闭。安全供应商凭据及同步合同待对接，全部本地验证使用明确的合成文本和模拟供应商，不调用真实付费接口。

复审后的实现约束：导出使用一个只读查询同时获取修订与全部匹配行，不锁住词典写入；序列化前结束事务。CSV使用显式_text_encoding=apostrophe-v1列标明公式转义，原有单引号也转义一层，只有带此标记的文件才解码，手写CSV不改变真实前缀。导入记录默认看本人、可切全部并分页，原来的员工只读权限不变。一个批次最多一条安全请求在途；在途未结不再扩散该批次费用，其他批次继续。
