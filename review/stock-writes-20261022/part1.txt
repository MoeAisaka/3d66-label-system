def lock_plan(db,id):
    reserve_sqlite_immediate_transaction(db)
    row=db.scalar(select(StockWritePlan).where(StockWritePlan.id==id).with_for_update().execution_options(populate_existing=True))
    if row is None:raise WriteError('plan_not_found')
    return row


def source_run(db,kind,run_id):
    from .cover_regression import CoverRegressionRun
    from .models import BaselineRegressionRun
    return db.get(CoverRegressionRun if kind=='cover' else BaselineRegressionRun,run_id if kind=='cover' else int(run_id))


def source_rank(run,doc,mode):
    human=doc.get('human') if mode=='effective' else None
    stamp=str(getattr(run,'created_at',''))
    if human:
        stamp=human.get('created_at') or max((str(e.get('corrected_at','')) for e in human.get('events',[])),default=stamp)
    try:stamp=datetime.fromisoformat(str(stamp).replace('Z','+00:00')).astimezone(timezone.utc).isoformat()
    except (ValueError,TypeError):stamp=''
    return f'{int(bool(human))}:{stamp}:{str(run.id).zfill(20)}:{int(human.get("revision",0)) if human else 0:010}'


def activate_head(db,row,config):
    if config['write_mode']!='overwrite':return
    head=db.scalar(select(StockWriteHead).where(StockWriteHead.id==row.head_id).with_for_update().execution_options(populate_existing=True))
    previous=db.get(StockWriteItem,head.latest_item_id) if head.latest_item_id else None
    equivalent=bool(previous and previous.source_hash==row.source_hash and previous.id!=row.id and previous.status not in {'failed','skipped','cancelled'}
        and json.loads(db.get(StockWriteTarget,previous.target_id).config_json)==config)
    if row.source_rank<head.latest_rank or equivalent:
        row.status='skipped';return
    head.latest_rank=row.source_rank;head.latest_item_id=row.id


def selected(doc,selection):
    return not (selection.get('status') and selection['status']!=doc['status'] or
        'reviewed' in selection and selection['reviewed']!=(doc.get('human') is not None) or
        selection.get('q','').casefold() not in (str(doc.get('name',''))+' '+str(doc['item_id'])+' '+str(doc['asset_id'])).casefold())


def complete(plan,doc):
    result=doc.get('raw') if plan.result_mode=='raw' else doc.get('effective')
    if result is None:return False
    if plan.source_kind=='cover':return doc['status']=='succeeded' or bool(doc.get('human') and plan.result_mode=='effective')
    return doc['status']=='completed'


def in_original_scope(db,plan,item_id):
    key=('stock_write_scope',plan.snapshot_id)
    if key not in db.info:
        value=func.json_extract(StockExportItem.document_json,'$.item_id')
        if db.get_bind().dialect.name=='mysql':value=func.json_unquote(value)
        db.info[key]=set(str(v) for v in db.scalars(select(value).where(StockExportItem.snapshot_id==plan.snapshot_id)))
    return str(item_id) in db.info[key]


def enqueue_document(db,plan,target,run,doc,*,initial=False):
    if not complete(plan,doc):return None
    config=json.loads(target.config_json)
    if plan.result_mode=='raw':doc={**doc,'effective':doc['raw'],'human':None}
    from .workflow_node_executors import redact_workflow_secrets
    doc={**doc,'input':redact_workflow_secrets(doc.get('input',{}))}
    if 'read_model_snapshot' in doc:doc['read_model_snapshot']=redact_workflow_secrets(doc['read_model_snapshot'])
    # Execution state/error are facts, not new content. Model reruns that retain
    # a human image cannot silently mint another effective version.
    version=digest({k:doc.get(k) for k in ('item_id','effective','human')})
    previous=db.scalar(select(StockWriteItem).where(StockWriteItem.target_id==target.id,
        StockWriteItem.source_item_id==str(doc['item_id']),StockWriteItem.source_hash==version))
    if previous:return previous
    head_id=digest([target.destination_key,doc['asset_id']])
    head=db.scalar(select(StockWriteHead).where(StockWriteHead.id==head_id).with_for_update().execution_options(populate_existing=True))
    if head is None:
        head=StockWriteHead(id=head_id,latest_rank='');db.add(head);db.flush()
    rank=source_rank(run,doc,plan.result_mode)
    if plan.source_kind=='cover':
        from .cover_attempts import CoverGenerationAttempt
        attempt_id=(doc.get('effective') or {}).get('attempt_id')
        attempt=db.get(CoverGenerationAttempt,attempt_id) if attempt_id else None
        rank+=f':{attempt.sequence_no if attempt else 0:010}'
    old=db.scalar(select(StockWriteItem).where(StockWriteItem.target_id==target.id,StockWriteItem.source_item_id==str(doc['item_id']))
        .order_by(StockWriteItem.id.desc()).limit(1))
    human_revision=int((doc.get('human') or {}).get('revision',0))
    automatic_human=bool(plan.status!='draft' and old and human_revision>old.human_revision)
    first_complete=bool(plan.status!='draft' and old is None and in_original_scope(db,plan,doc['item_id']))
    status='queued' if plan.status!='draft' and (automatic_human or plan.automatic or first_complete) else 'awaiting_confirmation'
    any_sent=db.scalar(select(StockWriteItem.id).where(StockWriteItem.head_id==head.id,StockWriteItem.attempts>0).limit(1))
    if plan.source_kind=='cover' and config['write_mode']=='overwrite' and (head.delivered_item_id or any_sent) and not automatic_human:
        status='awaiting_confirmation'
    if config['write_mode']=='overwrite' and rank<head.latest_rank:status='skipped'
    row=StockWriteItem(target_id=target.id,source_item_id=str(doc['item_id']),asset_id=doc['asset_id'],head_id=head.id,
        source_hash=version,source_rank=rank,human_revision=human_revision,request_id=uuid.uuid4().hex,
        document_json=canonical(doc),status=status,created_at=now(),updated_at=now())
    db.add(row);db.flush()
    if status=='queued':activate_head(db,row,config)
    audit(db,row,'prepare',status,plan.created_by,source_hash=version,human_revision=human_revision)
    plan.revision+=1
    return row


def create_plan(db,*,source_kind,run_id,selection,result_mode,actor,targets,automatic=False):
    if not isinstance(targets,list) or not 1<=len(targets)<=10:raise WriteError('invalid_target_count')
    if type(automatic) is not bool:raise WriteError('invalid_schedule')
    snap=freeze_results(db,source_kind=source_kind,run_id=run_id,selection=selection,result_mode=result_mode,actor=actor)
    plan=StockWritePlan(id=uuid.uuid4().hex,snapshot_id=snap.id,source_kind=snap.source_kind,run_id=snap.run_id,
        category_key=snap.category_key,selection_json=canonical(selection),result_mode=result_mode,automatic=automatic,
        status='draft',revision=1,created_by=actor,authorized_by=actor,created_at=now())
    db.add(plan);db.flush();destinations=set()
    for options in targets:
        endpoint_id=options.get('endpoint_id')
        if endpoint_id:
            if set(options)!={'endpoint_id'}:raise WriteError('registered_target_is_frozen')
            endpoint=db.get(StockWriteEndpoint,endpoint_id)
            if not endpoint or endpoint.category_key!=snap.category_key:raise WriteError('endpoint_category_mismatch')
            config=json.loads(endpoint.config_json)
        else:
            try:config=normalize_config(options)
            except WriteHttpError as exc:raise WriteError(str(exc)) from exc
        destination_key=digest(config['url'])
        if destination_key in destinations:raise WriteError('duplicate_destination')
        destinations.add(destination_key)
        target=StockWriteTarget(id=uuid.uuid4().hex,plan_id=plan.id,endpoint_id=endpoint_id,
            destination_key=destination_key,config_json=canonical(config))
        db.add(target);db.flush()
        run=source_run(db,plan.source_kind,plan.run_id)
        for stored in db.scalars(select(StockExportItem).where(StockExportItem.snapshot_id==snap.id).order_by(StockExportItem.ordinal)):
            enqueue_document(db,plan,target,run,json.loads(stored.document_json),initial=True)
    db.flush();return plan


def confirm_plan(db,id,*,expected_revision,actor):
    plan=lock_plan(db,id)
    if plan.revision!=expected_revision:raise WriteError('preview_changed')
    if plan.status in {'cancelled','paused'}:raise WriteError('plan_not_active')
    rows=db.scalars(select(StockWriteItem).join(StockWriteTarget).where(StockWriteTarget.plan_id==id,
        StockWriteItem.status=='awaiting_confirmation').order_by(StockWriteItem.id).with_for_update()).all()
    waiting={row.source_item_id:row.source_hash for row in rows}
    with source_session(db) as source:
        documents=_cover_rows(source,plan.run_id) if plan.source_kind=='cover' else _baseline_rows(source,plan.run_id)
        for _,doc in documents:
            if str(doc['item_id']) not in waiting:continue
            effective=doc.get('raw') if plan.result_mode=='raw' else doc.get('effective')
            human=None if plan.result_mode=='raw' else doc.get('human')
            current=digest({'item_id':doc['item_id'],'effective':effective,'human':human})
            if waiting[str(doc['item_id'])]!=current:raise WriteError('preview_outdated')
    for row in rows:
        row.status='queued';row.revision+=1;row.updated_at=now();audit(db,row,'confirm','queued',actor)
        activate_head(db,row,json.loads(db.get(StockWriteTarget,row.target_id).config_json))
    plan.status='active';plan.authorized_by=actor;plan.error_code=None;plan.revision+=1;db.flush();return plan


def control_plan(db,id,action,*,actor):
    plan=lock_plan(db,id)
    if plan.status=='cancelled' or plan.status=='draft' and action!='cancel':raise WriteError('plan_not_active')
    if action not in {'pause','resume','cancel'}:raise WriteError('invalid_control')
    plan.status={'pause':'paused','resume':'active','cancel':'cancelled'}[action]
    if action=='resume':plan.authorized_by=actor;plan.error_code=None
    if action=='cancel':
        for row in db.scalars(select(StockWriteItem).join(StockWriteTarget).where(StockWriteTarget.plan_id==id,
            StockWriteItem.status.in_(['queued','retry','awaiting_confirmation']))):
            row.status='cancelled';row.revision+=1;audit(db,row,'cancel','cancelled',actor)
    plan.revision+=1
    from .audit import append_audit_event
    append_audit_event(db,category='baseline',action='stock_write_'+action,subject_type='stock_write_plan',subject_id=id,actor=actor,payload={})
    db.flush();return plan


def scan_plan(db,id,*,force=False):
    plan=lock_plan(db,id)
    if plan.status=='cancelled' or plan.status=='draft' and not force:return
    targets=db.scalars(select(StockWriteTarget).where(StockWriteTarget.plan_id==id)).all()
    selection=json.loads(plan.selection_json)
    with source_session(db) as source:
        rows=_cover_rows(source,plan.run_id) if plan.source_kind=='cover' else _baseline_rows(source,plan.run_id)
        for run,doc in rows:
            if selected(doc,selection):
                for target in targets:enqueue_document(db,plan,target,run,doc)
                # Only an unsent preview can be superseded here; uncertain or
                # delivered rows remain permanent evidence and ordering barriers.
                effective=doc.get('raw') if plan.result_mode=='raw' else doc.get('effective')
                human=None if plan.result_mode=='raw' else doc.get('human')
                current=digest({'item_id':doc['item_id'],'effective':effective,'human':human})
                for old in db.scalars(select(StockWriteItem).join(StockWriteTarget).where(StockWriteTarget.plan_id==plan.id,
                    StockWriteItem.source_item_id==str(doc['item_id']),StockWriteItem.status=='awaiting_confirmation',StockWriteItem.source_hash!=current)):
                    old.status='skipped';old.revision+=1;audit(db,old,'supersede','skipped','worker')
    plan.last_scanned_at=now();db.flush()


def artifact_path(settings,descriptor):
    sha=descriptor.get('sha256','')
    if not re.fullmatch('[0-9a-f]{64}',sha):raise WriteError('invalid_artifact')
    path=Path(settings.data_dir)/'stock-writes'/'files'/(sha+'.bin')
    if not path.is_file():raise WriteError('artifact_missing')
    hasher=hashlib.sha256()
    with path.open('rb') as file:
        for chunk in iter(lambda:file.read(1024*1024),b''):hasher.update(chunk)
    if hasher.hexdigest()!=sha:raise WriteError('artifact_hash_mismatch')
    return path


def freeze_artifact(settings,plan,config,doc):
    if config['file_source']=='generated':
        if plan.source_kind!='cover':raise WriteError('generated_file_unavailable')
        result=doc['effective'];sha=result.get('sha256','')
        if not re.fullmatch('[0-9a-f]{64}',sha):raise WriteError('invalid_artifact')
        source=Path(settings.data_dir)/'stock-images'/'outputs'/(sha+'.png');mime='image/png'
    else:
        from types import SimpleNamespace
        from .nas_storage import resolve_asset_path
        frozen=doc['input'];sha=frozen.get('sha256','')
        if not re.fullmatch('[0-9a-f]{64}',sha):raise WriteError('original_file_identity_unavailable')
        source=resolve_asset_path(SimpleNamespace(**frozen),settings=settings);mime=frozen.get('mime_type','application/octet-stream')
        if frozen.get('storage_backend','local')=='local' and not source.resolve().is_relative_to(Path(settings.upload_dir).resolve()):
            raise WriteError('original_file_outside_storage')
    root=Path(settings.data_dir)/'stock-writes'/'files';root.mkdir(parents=True,exist_ok=True)
    import shutil
    if shutil.disk_usage(root).free<source.stat().st_size+512*1024**2:raise WriteError('artifact_disk_space_low')
    destination=root/(sha+'.bin');part=root/(uuid.uuid4().hex+'.part');hasher=hashlib.sha256()
    try:
        with source.open('rb') as src,part.open('xb') as dst:
            for chunk in iter(lambda:src.read(1024*1024),b''):
                hasher.update(chunk);dst.write(chunk)
            dst.flush();os.fsync(dst.fileno())
        if hasher.hexdigest()!=sha:raise WriteError('artifact_hash_mismatch')
        os.replace(part,destination)
    finally:part.unlink(missing_ok=True)
    return {'sha256':sha,'size_bytes':destination.stat().st_size,'mime_type':mime}


def prepare_body(db,row,plan,config,settings):
    if row.body_json:return json.loads(row.body_json)
    doc=json.loads(row.document_json)
    mapped={}
    for key,path in config['field_mapping'].items():
        value=doc
        for part in path.split('.'):
            if not isinstance(value,dict) or part not in value:raise WriteError('field_mapping_missing')
            value=value[part]
        mapped[key]=value
    body={'protocol':'stock-result-v1','request_id':row.request_id,'business_key':str(row.asset_id),
        'source_kind':plan.source_kind,'run_id':plan.run_id,'source_item_id':row.source_item_id,
        'source_version':row.source_hash,'write_mode':config['write_mode'],'result_mode':plan.result_mode,
        'result':doc['effective'],'human':doc.get('human'),
        'validation_scope':{'execution_complete':True,'scope':'frozen_workflow_only',
            'configuration':json.loads(db.get(StockExportSnapshot,plan.snapshot_id).config_json),
            'node_results':doc.get('read_model_snapshot',{}).get('workflow_node_outputs',{})},
        'mapped_fields':mapped}
    if config['file_mode']!='none':
        descriptor=freeze_artifact(settings,plan,config,doc);row.file_json=canonical(descriptor)
        body['file']=descriptor.copy()
        if config['file_mode']=='url':
            base=os.environ.get('STOCK_WRITE_FILE_BASE_URL','').rstrip('/')
            if not base.startswith(('http://','https://')):raise WriteError('file_url_not_configured')
            token=secrets.token_urlsafe(32);row.file_token_hash=hashlib.sha256(token.encode()).hexdigest()
            row.file_expires_at=(datetime.now(timezone.utc)+timedelta(days=30)).isoformat()
            body['file']['url']=base+'/api/stock-writes/files/'+token
            body['file']['expires_at']=row.file_expires_at
    row.payload_hash=digest(body);body['payload_hash']=row.payload_hash;row.body_json=canonical(body)
    db.flush();return body


def resolve_item(db,id,*,expected_revision,outcome,evidence,actor):
    if outcome not in {'applied','not_applied','retry'} or not isinstance(evidence,str) or not evidence.strip() or len(evidence)>2000:
        raise WriteError('resolution_evidence_required')
    reserve_sqlite_immediate_transaction(db)
    row=db.scalar(select(StockWriteItem).where(StockWriteItem.id==id).with_for_update().execution_options(populate_existing=True))
    if not row:raise WriteError('item_not_found')
    if row.revision!=expected_revision:raise WriteError('item_changed')
    if row.status not in {'uncertain','failed'}:raise WriteError('item_not_resolvable')
    head=db.scalar(select(StockWriteHead).where(StockWriteHead.id==row.head_id).with_for_update().execution_options(populate_existing=True))
    if outcome=='applied':
        row.status='succeeded';head.delivered_item_id=row.id
    elif outcome=='not_applied':row.status='skipped'
    else:
        target=db.get(StockWriteTarget,row.target_id)
        if db.get(StockWritePlan,target.plan_id).status=='cancelled':raise WriteError('plan_not_active')
        row.status='queued';row.available_at='';row.attempts=0
    row.error_code=None;row.revision+=1;row.updated_at=now()
    audit(db,row,'manual_reconcile',outcome,actor,evidence=evidence)
    db.flush();return row


def source_is_paused(db,plan):
    run=source_run(db,plan.source_kind,plan.run_id)
    return not run or run.status in {'paused','cancelled','canceled'}


def candidate_ids(db):
    cursor=0
    while True:
        ids=db.scalars(select(StockWriteItem.id).where(StockWriteItem.id>cursor,
            StockWriteItem.status.in_(['queued','retry','sending','uncertain']),StockWriteItem.available_at<=now())
            .order_by(StockWriteItem.id).limit(100)).all()
        if not ids:return
        for id in ids:cursor=id;yield id


def dispatch_one(db,*,settings,send=http_send,checkpoint=lambda:None,authorize=None):
    """One bounded network action, always after a committed immutable intent.

    The worker holds a global named lock. Sending rows left by a lost worker
    are reconciliation-only; old unconfirmed requests are an overwrite barrier.
    """
    if authorize is None:
        from .stock_write_worker import authorized
        authorize=authorized
    for candidate_id in candidate_ids(db):
        candidate=db.get(StockWriteItem,candidate_id)
        target=db.get(StockWriteTarget,candidate.target_id);plan=lock_plan(db,target.plan_id)
        row=db.scalar(select(StockWriteItem).where(StockWriteItem.id==candidate.id).with_for_update().execution_options(populate_existing=True))
        config=json.loads(target.config_json)
        if row.status not in {'queued','retry','sending','uncertain'}:continue
        is_uncertain=row.status in {'sending','uncertain'}
        query=is_uncertain and bool(config['query_path'])
        if is_uncertain and not query and not config['idempotent']:continue
        # Querying an old request is allowed while paused. A new/repeated POST
        # remains paused, including retries, and cannot follow cancellation.
        if not query and (plan.status!='active' or source_is_paused(db,plan) or not authorize(db,plan)):continue
        if not is_uncertain:
            from .models import Asset
            asset=db.get(Asset,row.asset_id)
            if not asset or asset.category_key!=plan.category_key:
                row.status='failed';row.error_code='source_category_changed';db.commit();return True
            # Check the one current source, not the whole run. A person's new
            # correction must win even between the periodic scan and this POST.
            with source_session(db) as source:
                fresh=next(_cover_rows(source,plan.run_id,row.source_item_id) if plan.source_kind=='cover' else
                    _baseline_rows(source,plan.run_id,row.source_item_id),None)
            if fresh:
                run,doc=fresh
                enqueue_document(db,plan,target,run,doc)
            else:
                row.status='failed';row.error_code='source_item_unavailable';db.commit();return True
        head=db.scalar(select(StockWriteHead).where(StockWriteHead.id==row.head_id).with_for_update().execution_options(populate_existing=True))
        if config['write_mode']=='overwrite':
            barrier=db.scalar(select(StockWriteItem.id).where(StockWriteItem.head_id==row.head_id,
                StockWriteItem.id<row.id,StockWriteItem.status.in_(['sending','uncertain'])).limit(1))
            if barrier is not None:continue
            if not is_uncertain and head.latest_item_id!=row.id:
                row.status='skipped';row.revision+=1;row.updated_at=now();audit(db,row,'supersede','skipped','worker');db.commit();return True
        if row.attempts>=3 and not query:
            row.status='uncertain' if is_uncertain or row.body_json else 'failed'
            row.error_code='automatic_attempt_limit';row.available_at=(datetime.now(timezone.utc)+timedelta(days=36500)).isoformat()
            db.commit();return True
        if query and row.attempts>=12:
            row.status='uncertain';row.error_code='automatic_query_limit'
            row.available_at=(datetime.now(timezone.utc)+timedelta(days=36500)).isoformat();db.commit();return True
        checkpoint()
        try:body=prepare_body(db,row,plan,config,settings)
        except (WriteError,OSError,ValueError) as exc:
            row.status='failed';row.error_code=str(exc) if isinstance(exc,WriteError) else 'artifact_unavailable'
            row.revision+=1;audit(db,row,'prepare','failed','worker',error_code=row.error_code);db.commit();return True
        row.status='sending';row.attempts+=1;row.revision+=1;row.updated_at=now()
        attempt=row.attempts;revision=row.revision
        audit(db,row,'query' if query else 'post','intent_committed','worker',attempt=attempt,payload_hash=row.payload_hash)
        db.commit();checkpoint()
        try:
            file_path=artifact_path(settings,json.loads(row.file_json)) if row.file_json and config['file_mode']=='upload' and not query else None
            receipt=send(config,body,query=query,file_path=file_path)
            if (receipt.get('request_id')!=row.request_id or receipt.get('payload_hash')!=row.payload_hash
                or receipt.get('status') not in {'succeeded','rejected','not_applied','pending','not_found'}):
                raise WriteError('receipt_mismatch')
            outcome=receipt['status']
        except WriteError:
            outcome='receipt_mismatch'
        except Exception:
            # Never store exception strings: receiver/network errors may carry
            # credentials, URLs or response bodies. The immutable intent stays.
            outcome='transport_unconfirmed'
        checkpoint()
        db.expire_all()
        current=db.scalar(select(StockWriteItem).where(StockWriteItem.id==row.id).with_for_update().execution_options(populate_existing=True))
        if current.attempts!=attempt or current.revision!=revision or current.status!='sending':
            db.rollback();return True
        head=db.scalar(select(StockWriteHead).where(StockWriteHead.id==current.head_id).with_for_update().execution_options(populate_existing=True))
        if outcome=='succeeded':current.status='succeeded';head.delivered_item_id=current.id;current.error_code=None
        elif outcome=='rejected':current.status='failed';current.error_code='receiver_rejected'
        elif outcome=='not_applied' and query:current.status='retry';current.error_code='receiver_not_applied'
        else:
            current.status='uncertain';current.error_code=outcome if outcome in {'receipt_mismatch','transport_unconfirmed'} else 'receiver_'+outcome
        current.revision+=1;current.updated_at=now()
        current.available_at=(datetime.now(timezone.utc)+timedelta(seconds=min(300,10*2**min(attempt,5)))).isoformat()
        audit(db,current,'query' if query else 'post',outcome,'worker',attempt=attempt)
        db.commit();return True
    db.rollback();return False

"""Small, explicit HTTP receiver protocol for stock results (never a proxy).

Requests do not use environment proxies, redirects, platform cookies or arbitrary
headers. DNS results are checked and the actual connection uses a checked IP.
The receiver acknowledges the exact immutable request ID and content digest.
"""
from __future__ import annotations
import http.client
import ipaddress
import json
import os
import re
import socket
import ssl
from urllib.parse import urlsplit, urlunsplit, quote


class WriteHttpError(ValueError):pass


def validate_url(url, *, resolve=False, resolver=socket.getaddrinfo):
    if not isinstance(url,str) or len(url)>2000 or any(ord(c)<33 for c in url):
        raise WriteHttpError('invalid_endpoint')
    try:
        parts=urlsplit(url);port=parts.port or (443 if parts.scheme=='https' else 80)
    except ValueError:raise WriteHttpError('invalid_endpoint')
    if parts.scheme not in {'http','https'} or not parts.hostname or parts.username or parts.password or parts.query or parts.fragment:
        raise WriteHttpError('invalid_endpoint')
    host=parts.hostname.lower().rstrip('.')
    if host in {'localhost','metadata.google.internal'} or host.endswith(('.localhost','.localdomain')):
        raise WriteHttpError('unsafe_endpoint')
    forbidden={('192.168.0.85',80),('192.168.0.85',443),('192.168.1.137',8082),('192.168.1.205',8190)}
    # Deployment may add other platform/control surfaces without changing code.
    for value in os.environ.get('STOCK_WRITE_DENIED_HOST_PORTS','').split(','):
        if value.strip():
            h,sep,p=value.strip().rpartition(':')
            if sep and p.isdigit():forbidden.add((h,int(p)))
    def address(value):
        ip=ipaddress.ip_address(value)
        mapped=getattr(ip,'ipv4_mapped',None)
        if mapped:ip=mapped
        if (ip.is_loopback or ip.is_link_local or ip.is_multicast or ip.is_unspecified or ip.is_reserved
            or str(ip) in {'100.100.100.200','168.63.129.16'}
            or (str(ip),port) in forbidden or (host,port) in forbidden):raise WriteHttpError('unsafe_endpoint')
        return str(ip)
    try:ips=[address(host)]
    except ValueError as exc:
        if isinstance(exc,WriteHttpError):raise
        ips=[]
    if resolve:
        try:ips=sorted({address(row[4][0]) for row in resolver(host,port,type=socket.SOCK_STREAM)})
        except OSError:raise WriteHttpError('endpoint_dns_unavailable')
        if not ips:raise WriteHttpError('endpoint_dns_unavailable')
    authority=('['+host+']') if ':' in host else host.encode('idna').decode('ascii')
    if port!=(443 if parts.scheme=='https' else 80):authority+=':'+str(port)
    return urlunsplit((parts.scheme,authority,parts.path or '/','','')),host,port,ips


def normalize_config(config):
    if not isinstance(config,dict):raise WriteHttpError('invalid_target')
    allowed={'name','url','query_path','idempotent','write_mode','file_mode','file_source','field_mapping','timeout_seconds','bearer','secret_reference'}
    if set(config)-allowed:raise WriteHttpError('unknown_target_option')
    value={'name':str(config.get('name','')).strip(),'url':validate_url(config.get('url',''))[0],
        'idempotent':config.get('idempotent',False),'query_path':config.get('query_path',''),
        'write_mode':config.get('write_mode','overwrite'),'file_mode':config.get('file_mode','none'),
        'file_source':config.get('file_source','generated'),'field_mapping':config.get('field_mapping',{}),
        'timeout_seconds':config.get('timeout_seconds',30)}
    if not value['name'] or len(value['name'])>100 or type(value['idempotent']) is not bool:
        raise WriteHttpError('invalid_target')
    if value['write_mode'] not in {'overwrite','append'} or value['file_mode'] not in {'none','url','upload'} or value['file_source'] not in {'generated','original'}:
        raise WriteHttpError('invalid_target')
    if type(value['timeout_seconds']) is not int or not 1<=value['timeout_seconds']<=60:raise WriteHttpError('invalid_timeout')
    path=value['query_path']
    if path and (not isinstance(path,str) or not path.startswith('/') or path.startswith('//') or
        path.count('{request_id}')!=1 or re.search(r'[?#\s\\]',path) or '{' in path.replace('{request_id}','') or '}' in path.replace('{request_id}','')):
        raise WriteHttpError('invalid_query_path')
    mapping=value['field_mapping']
    if not isinstance(mapping,dict) or len(mapping)>100 or any(not re.fullmatch(r'[A-Za-z][A-Za-z0-9_]{0,79}',k) or
        not isinstance(v,str) or not re.fullmatch(r'(?:effective|raw|input|human)(?:\.[A-Za-z0-9_-]+)*',v) for k,v in mapping.items()):
        raise WriteHttpError('invalid_field_mapping')
    if 'bearer' in config:
        secret=config['bearer']
        if not isinstance(secret,str) or len(secret)>8000 or any(c in secret for c in '\r\n'):raise WriteHttpError('invalid_authentication')
        if secret:
            import uuid
            from .security import protect_secret
            value['secret_reference']=protect_secret(secret,account='stock-write-'+uuid.uuid4().hex)
    elif config.get('secret_reference'):value['secret_reference']=config['secret_reference']
    return value


def send(config,body,*,query=False,file_path=None):
    url,host,port,ips=validate_url(config['url'],resolve=True)
    parts=urlsplit(url);path=parts.path or '/'
    if query:
        if not config.get('query_path'):raise WriteHttpError('query_not_configured')
        path=config['query_path'].replace('{request_id}',quote(body['request_id'],safe=''))
    connection=http.client.HTTPConnection(host,port,timeout=config['timeout_seconds'])
    # Fixed destination IP, but retain DNS hostname for TLS certificate checks
    # and HTTP Host. Do not follow a response Location header.
    def connect():
        sock=socket.create_connection((ips[0],port),timeout=config['timeout_seconds'])
        if parts.scheme=='https':
            try:sock=ssl.create_default_context().wrap_socket(sock,server_hostname=host)
            except BaseException:sock.close();raise
        connection.sock=sock
    connection.connect=connect
    headers={'Accept':'application/json','Idempotency-Key':body['request_id'],'X-Payload-SHA256':body['payload_hash']}
    if config.get('secret_reference'):
        from .security import unprotect_secret
        headers['Authorization']='Bearer '+unprotect_secret(config['secret_reference'])
    data=None
    try:
        if not query:
            metadata=json.dumps(body,ensure_ascii=True,separators=(',',':')).encode()
            if file_path is None:
                headers['Content-Type']='application/json';data=metadata
                connection.request('POST',path,body=data,headers=headers)
            else:
                import uuid
                boundary='stock'+uuid.uuid4().hex
                prefix=(f'--{boundary}\r\nContent-Disposition: form-data; name="metadata"\r\nContent-Type: application/json\r\n\r\n').encode()+metadata
                prefix+=(f'\r\n--{boundary}\r\nContent-Disposition: form-data; name="file"; filename="artifact.bin"\r\nContent-Type: application/octet-stream\r\n\r\n').encode()
                suffix=f'\r\n--{boundary}--\r\n'.encode()
                headers['Content-Type']='multipart/form-data; boundary='+boundary
                headers['Content-Length']=str(len(prefix)+file_path.stat().st_size+len(suffix))
                connection.putrequest('POST',path)
                for key,value in headers.items():connection.putheader(key,value)
                connection.endheaders();connection.send(prefix)
                with file_path.open('rb') as file:
                    for chunk in iter(lambda:file.read(1024*1024),b''):connection.send(chunk)
                connection.send(suffix)
        else:connection.request('GET',path,headers=headers)
        response=connection.getresponse()
        raw=response.read(1024*1024+1)
        if len(raw)>1024*1024 or response.status<200 or response.status>=300:raise WriteHttpError('receiver_response_unconfirmed')
        try:value=json.loads(raw)
        except ValueError:raise WriteHttpError('receiver_response_unconfirmed')
        if not isinstance(value,dict):raise WriteHttpError('receiver_response_unconfirmed')
        return value
    finally:connection.close()

"""Separate stock writer. Recovery queries continue while new posts are paused."""
from contextlib import contextmanager
from pathlib import Path
import logging,os,time
from sqlalchemy import create_engine,select,text
from sqlalchemy.orm import Session
from sqlalchemy.pool import NullPool
from .stock_writes import StockWritePlan,scan_plan,dispatch_one,now


@contextmanager
def write_lock(engine,root):
    if engine.dialect.name=='mysql':
        lock_engine=create_engine(engine.url,poolclass=NullPool)
        try:
            with lock_engine.connect() as c:
                acquired=c.execute(text("SELECT GET_LOCK('stock-write-worker',0)")).scalar()==1
                def checkpoint():
                    if c.execute(text("SELECT IS_USED_LOCK('stock-write-worker')=CONNECTION_ID()")).scalar()!=1:
                        raise RuntimeError('stock writer lost lock')
                try:yield checkpoint if acquired else None
                finally:
                    if acquired and not c.invalidated:
                        try:c.execute(text("SELECT RELEASE_LOCK('stock-write-worker')"))
                        except Exception:pass
        finally:lock_engine.dispose()
    elif engine.dialect.name=='sqlite':
        from .cover_worker import scheduler_lock
        with scheduler_lock(engine,Path(root)/'stock-write-worker') as acquired:yield (lambda:None) if acquired else None
    else:raise RuntimeError('unverified stock write locking backend')


def authorized(db,plan):
    from .models import User
    from .authz import has_permission,writable_categories
    user=db.scalar(select(User).where(User.username==plan.authorized_by))
    allowed=writable_categories(user) if user else []
    return bool(user and user.is_active and has_permission(user,'reviews:write') and (allowed is None or plan.category_key in allowed))


def tick(engine,*,settings,send=None):
    from datetime import datetime,timezone,timedelta
    with write_lock(engine,settings.data_dir) as checkpoint:
        if checkpoint is None:return False
        with Session(engine) as db:
            threshold=(datetime.now(timezone.utc)-timedelta(seconds=15)).isoformat()
            plan=db.scalar(select(StockWritePlan).where(StockWritePlan.status.in_(['active','paused']),StockWritePlan.last_scanned_at<threshold)
                .order_by(StockWritePlan.last_scanned_at,StockWritePlan.id).limit(1))
            if plan:
                checkpoint()
                if authorized(db,plan):scan_plan(db,plan.id)
                else:plan.error_code='owner_permission_changed';plan.status='paused';plan.last_scanned_at=now()
                db.commit()
            options={'send':send} if send else {}
            return dispatch_one(db,settings=settings,checkpoint=checkpoint,authorize=authorized,**options)


if __name__=='__main__':
