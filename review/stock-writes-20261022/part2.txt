def plan_payload(db,plan):
    counts=dict(db.execute(select(StockWriteItem.status,func.count()).join(StockWriteTarget)
        .where(StockWriteTarget.plan_id==plan.id).group_by(StockWriteItem.status)).all())
    targets=db.scalars(select(StockWriteTarget).where(StockWriteTarget.plan_id==plan.id)).all()
    return {'id':plan.id,'source_kind':plan.source_kind,'run_id':plan.run_id,'category_key':plan.category_key,
        'source_total':db.get(StockExportSnapshot,plan.snapshot_id).total,

        'targets':[{'id':t.id,'endpoint_id':t.endpoint_id,**public_config(json.loads(t.config_json))} for t in targets]}

  return wire

class Receiver:
    def __init__(self):self.calls=[];self.fail=set();self.receipts={}
    def __call__(self, config, body, *, query=False, file_path=None):
        self.calls.append((config['name'],body['request_id'],query,body))
