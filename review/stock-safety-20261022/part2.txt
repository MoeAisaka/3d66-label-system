from __future__ import annotations

import json
from typing import Literal
from fastapi import APIRouter, Depends, HTTPException, Query
from pydantic import BaseModel, ConfigDict, Field
from sqlalchemy import select, func
from sqlalchemy.orm import Session
from sqlalchemy.exc import IntegrityError
from .authz import has_permission, require_categories
from .config import get_settings
from .database import get_db
from .safety_rules import DEFAULT_RULES, SafetyError
from .safety_provider import provider_configuration
from .safety_regression import (SafetyRun, SafetyRuleVersion, SafetyItem, SafetyAttempt, SafetyReview, SafetyResult, SafetyReference, SafetyStatistics,
    create_round, control, review_item, retry_item, recompute, item_payload, item_payloads, statistics, update_reference)


class TextSample(BaseModel):
    model_config = ConfigDict(extra="forbid")
    content: str = Field(min_length=1, max_length=2000)
    name: str = Field(default="", max_length=500)
    reference: Literal["pass", "reject", "review"] | None = None


class CreateSafetyRound(BaseModel):
    model_config = ConfigDict(extra="forbid")
    request_id: str = Field(min_length=1, max_length=64, pattern=r"^[A-Za-z0-9_-]+$")
    name: str = Field(min_length=1, max_length=160)
    category_key: str = Field(min_length=1, max_length=40)
    kind: Literal["image", "text"]
    service: str = Field(min_length=1, max_length=100)
    rules: dict = Field(default_factory=lambda: json.loads(json.dumps(DEFAULT_RULES)))
    baseline_set_id: int | None = Field(default=None, ge=1)
    asset_ids: list[int] = Field(default_factory=list, max_length=10000)
    texts: list[TextSample] = Field(default_factory=list, max_length=10000)


class ReviewSafetyItem(BaseModel):
    model_config = ConfigDict(extra="forbid")
    result_id: str
    expected_revision: int = Field(ge=0)
    patch: dict
    reason: str = Field(min_length=1, max_length=4000)


class ReferenceRequest(BaseModel):
    model_config = ConfigDict(extra="forbid")
    verdict: Literal["pass", "reject", "review"]
    expected_revision: int = Field(ge=0)
    confirmed: Literal[True]


class RecomputeRequest(BaseModel):
    model_config = ConfigDict(extra="forbid")
    expected_revision: int = Field(ge=1)
    rules: dict
    confirmed: Literal[True]


def build_safety_regression_router(current_user):
    router = APIRouter(tags=["stock-safety-regression"])
    def writable(user, category):
        if not has_permission(user, "reviews:write"): raise HTTPException(403, "需要该业务纠偏权限")
        require_categories(user, {category})

    def run_row(db, run_id):
        row = db.get(SafetyRun, run_id)
        if not row: raise HTTPException(404, "安全审核回归不存在")
        return row

    def item_row(db, run_id, item_id):
        row = db.get(SafetyItem, item_id)
        if not row or row.run_id != run_id: raise HTTPException(404, "样本不属于本轮")
        return row

    def summary(run):
        return {"id": run.id, "name": run.name, "category_key": run.category_key, "status": run.status,
            "config": json.loads(run.config_json), "rule_revision": run.rule_revision, "created_at": run.created_at}

    def execute(db, operation):
        try: value = operation(); db.commit(); return value
        except (SafetyError, IntegrityError) as exc:
            db.rollback(); raise HTTPException(409, exc.code if isinstance(exc, SafetyError) else "request_conflict") from None

    @router.get("/api/safety-regressions/capabilities")
    def capabilities(user=Depends(current_user)):
        return {**provider_configuration(), "default_rules": DEFAULT_RULES, "mixed_inputs": False,
            "human_review_required": False, "actual_cost": None}

    @router.post("/api/safety-regressions")
    def create(payload: CreateSafetyRound, user=Depends(current_user), db: Session = Depends(get_db)):
        writable(user, payload.category_key)
        if not provider_configuration()["configured"]: raise HTTPException(503, "阿里安全审核凭据尚未配置")
        asset_ids = payload.asset_ids
        if payload.baseline_set_id:
            if asset_ids or payload.texts: raise HTTPException(422, "基准集与手工输入只选一种")
            from .models import BaselineSet, BaselineSetItem
            baseline = db.get(BaselineSet, payload.baseline_set_id)
            if not baseline or baseline.category_key != payload.category_key: raise HTTPException(409, "基准集类目不一致")
            asset_ids = list(db.scalars(select(BaselineSetItem.asset_id).where(BaselineSetItem.baseline_set_id == baseline.id).order_by(BaselineSetItem.id)))
        return execute(db, lambda: summary(create_round(db, request_id=payload.request_id, name=payload.name,
            category=payload.category_key, kind=payload.kind, service=payload.service, rules=payload.rules,
            asset_ids=asset_ids, texts=[s.model_dump() for s in payload.texts], actor=user.username, settings=get_settings())))

    @router.get("/api/safety-regressions")
    def listing(category_key: str | None = None, offset: int = Query(0, ge=0), limit: int = Query(50, ge=1, le=100),
                user=Depends(current_user), db: Session = Depends(get_db)):
        q = select(SafetyRun)
        if category_key: q = q.where(SafetyRun.category_key == category_key)
        total = db.scalar(select(func.count()).select_from(q.subquery()))
        return {"total": total, "items": [summary(r) for r in db.scalars(q.order_by(SafetyRun.created_at.desc()).offset(offset).limit(limit))]}

    @router.get("/api/safety-regressions/{run_id}")
    def detail(run_id: str, offset: int = Query(0, ge=0), limit: int = Query(24, ge=1, le=100),
               user=Depends(current_user), db: Session = Depends(get_db)):
        run = run_row(db, run_id)
        items = db.scalars(select(SafetyItem).where(SafetyItem.run_id == run_id).order_by(SafetyItem.id).offset(offset).limit(limit)).all()
        rules = db.scalar(select(SafetyRuleVersion).where(SafetyRuleVersion.run_id == run_id, SafetyRuleVersion.revision == run.rule_revision))
        return {**summary(run), "current_rules": json.loads(rules.rules_json),
            "statistics": statistics(db, run_id), "items": item_payloads(db, items, run)}

    @router.post("/api/safety-regressions/{run_id}/control/{action}")
    def change_status(run_id: str, action: Literal["pause", "resume", "cancel"], user=Depends(current_user), db: Session = Depends(get_db)):
        writable(user, run_row(db, run_id).category_key)
        return execute(db, lambda: summary(control(db, run_id, action)))

    @router.post("/api/safety-regressions/{run_id}/recompute")
    def recalculate(run_id: str, payload: RecomputeRequest, user=Depends(current_user), db: Session = Depends(get_db)):
        writable(user, run_row(db, run_id).category_key)
        return execute(db, lambda: summary(recompute(db, run_id, rules=payload.rules, expected_revision=payload.expected_revision, actor=user.username)))

    @router.post("/api/safety-regressions/{run_id}/items/{item_id}/review")
    def review(run_id: str, item_id: str, payload: ReviewSafetyItem, user=Depends(current_user), db: Session = Depends(get_db)):
        writable(user, run_row(db, run_id).category_key); item = item_row(db, run_id, item_id)
        def operation():
            review_item(db, item.id, actor=user.username, **payload.model_dump()); return item_payload(db, item)
        return execute(db, operation)

    @router.post("/api/safety-regressions/{run_id}/items/{item_id}/reference")
    def reference(run_id: str, item_id: str, payload: ReferenceRequest, user=Depends(current_user), db: Session = Depends(get_db)):
        writable(user, run_row(db, run_id).category_key); item = item_row(db, run_id, item_id)
        def operation():
            update_reference(db, item.id, actor=user.username, **payload.model_dump()); return item_payload(db, item)
        return execute(db, operation)

    @router.post("/api/safety-regressions/{run_id}/items/{item_id}/retry")
    def retry(run_id: str, item_id: str, user=Depends(current_user), db: Session = Depends(get_db)):
        writable(user, run_row(db, run_id).category_key); item = item_row(db, run_id, item_id)
        def operation(): retry_item(db, item.id); return item_payload(db, item)
        return execute(db, operation)

    @router.get("/api/safety-regressions/{run_id}/items/{item_id}/history")
    def history(run_id: str, item_id: str, offset: int = Query(0, ge=0), limit: int = Query(20, ge=1, le=100),
                user=Depends(current_user), db: Session = Depends(get_db)):
        item_row(db, run_id, item_id)
        attempts = db.scalars(select(SafetyAttempt).where(SafetyAttempt.item_id == item_id)
            .order_by(SafetyAttempt.sequence_no.desc()).offset(offset).limit(limit)).all()
        reviews = db.scalars(select(SafetyReview).where(SafetyReview.item_id == item_id)
            .order_by(SafetyReview.revision.desc()).offset(offset).limit(limit)).all()
        results = db.scalars(select(SafetyResult).where(SafetyResult.item_id == item_id)
            .order_by(SafetyResult.created_at.desc()).offset(offset).limit(limit)).all()
        references = db.scalars(select(SafetyReference).where(SafetyReference.item_id == item_id)
            .order_by(SafetyReference.revision.desc()).offset(offset).limit(limit)).all()
        return {"attempts": [{"id": a.id, "sequence": a.sequence_no, "status": a.status, "raw": json.loads(a.raw_json) if a.raw_json else None,
            "error": a.error_code, "usage_state": a.usage_state, "created_at": a.created_at} for a in attempts],
            "reviews": [{"revision": r.revision, "result_id": r.result_id, "patch": json.loads(r.patch_json),
                "reason": r.reason, "actor": r.actor, "created_at": r.created_at} for r in reviews],
            "results": [{"id": r.id, "attempt_id": r.attempt_id, "rule_revision": r.rule_revision,
                "result": json.loads(r.result_json), "created_at": r.created_at} for r in results],
            "references": [{"revision": r.revision, "verdict": r.verdict, "actor": r.actor, "created_at": r.created_at} for r in references]}

    @router.get("/api/safety-regressions/{run_id}/versions")
    def versions(run_id: str, offset: int = Query(0, ge=0), limit: int = Query(20, ge=1, le=100),
                 user=Depends(current_user), db: Session = Depends(get_db)):
        run_row(db, run_id)
        rules = db.scalars(select(SafetyRuleVersion).where(SafetyRuleVersion.run_id == run_id)
            .order_by(SafetyRuleVersion.revision.desc()).offset(offset).limit(limit)).all()
        saved = db.scalars(select(SafetyStatistics).where(SafetyStatistics.run_id == run_id)
            .order_by(SafetyStatistics.created_at.desc()).offset(offset).limit(limit)).all()
        return {"rules": [{"revision": r.revision, "rules": json.loads(r.rules_json), "actor": r.actor, "created_at": r.created_at} for r in rules],
            "statistics": [{"id": r.id, "statistics": json.loads(r.statistics_json), "created_at": r.created_at} for r in saved]}

    return router

def migrate(connection):

        SafetyResult, SafetyReview, SafetyReference, SafetyStatistics)
    models = (SafetyRun, SafetyRuleVersion, SafetyItem, SafetyAttempt, SafetyResult, SafetyReview, SafetyReference, SafetyStatistics)
    if connection.dialect.name == 'mysql':
        from .mysql_baseline import mysql_metadata
        metadata = mysql_metadata()
        for model in models: metadata.tables[model.__tablename__].create(connection, checkfirst=True)
    else:
        for model in models: model.__table__.create(connection, checkfirst=True)

    region = "cn-shanghai"

def test_four_hundred_sample_statistics_use_batched_reads(setup):
    import time
    from sqlalchemy import event
    engine, db, settings = setup
    run = make(db, settings, texts=[{'content': f'sample {i}'} for i in range(400)]); db.commit()
    statements = []
    def record(*args): statements.append(args[2])
    event.listen(engine, 'before_cursor_execute', record)
    start = time.perf_counter()
    try: result = statistics(db, run.id)
    finally: event.remove(engine, 'before_cursor_execute', record)
    assert result['total'] == 400 and result['accuracy'] is None
    assert len(statements) < 40
    print({'samples': 400, 'statistics_sql_count': len(statements), 'elapsed_seconds': round(time.perf_counter()-start, 3)})
