"""Personal MCP credentials. Only hashes are persisted; no shared machine token."""
from datetime import datetime, timezone
import hashlib
import hmac
import re
import secrets

from fastapi import HTTPException
from sqlalchemy import Boolean, DateTime, ForeignKey, String, select

from .database import Base


class PersonalMcpCredential(Base):
    __tablename__ = 'personal_mcp_credentials'
    user_id: Mapped[int] = mapped_column(ForeignKey('users.id', ondelete='CASCADE'), primary_key=True)
    token_hash: Mapped[str] = mapped_column(String(64))
    enabled: Mapped[bool] = mapped_column(Boolean, default=True)
    created_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc))
    rotated_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc))
    last_used_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True), nullable=True)
    last_tool: Mapped[str | None] = mapped_column(String(100), nullable=True)


def digest(token):
    return hashlib.sha256(token.encode()).hexdigest()


def authenticate(db, token, *, lock=False):
    from .models import User
    match = re.fullmatch(r'mcp_([1-9][0-9]{0,15})_([A-Za-z0-9_-]{43})', token or '')
    if not match:
        raise HTTPException(401, 'MCP令牌无效', headers={'WWW-Authenticate': 'Bearer'})
    user_id = int(match[1])
    query = select(User).where(User.id == user_id)
    if lock:
        query = query.with_for_update()
    user = db.scalar(query.execution_options(populate_existing=True))
    query = select(PersonalMcpCredential).where(PersonalMcpCredential.user_id == user_id)
    if lock:
        query = query.with_for_update()
    credential = db.scalar(query.execution_options(populate_existing=True))
    if not user or not user.is_active or not credential or not credential.enabled or not hmac.compare_digest(credential.token_hash, digest(token)):
        raise HTTPException(401, 'MCP已停用、令牌失效或账号不可用', headers={'WWW-Authenticate': 'Bearer'})
    return user, credential


def rotate(db, user):
    from .authz import lock_access_users
    from .models import User
    lock_access_users(db, [user.id])
    live = db.get(User, user.id, populate_existing=True)
    if not live or not live.is_active:
        raise HTTPException(403, '账号不可用')
    row = db.get(PersonalMcpCredential, user.id)
    token = f'mcp_{user.id}_{secrets.token_urlsafe(32)}'
    if row is None:
        row = PersonalMcpCredential(user_id=user.id)
        db.add(row)
    row.token_hash = digest(token)
    row.enabled = True
    row.rotated_at = datetime.now(timezone.utc)
    db.commit()
    return token


def disable(db, user):
    from .authz import lock_access_users
    lock_access_users(db, [user.id])
    row = db.get(PersonalMcpCredential, user.id, populate_existing=True)
    if row:
        row.enabled = False
        db.commit()


def status(db, user):
    row = db.get(PersonalMcpCredential, user.id)
    return {'enabled': bool(row and row.enabled), 'rotated_at': row.rotated_at if row else None,
            'last_used_at': row.last_used_at if row else None, 'last_tool': row.last_tool if row else None,
            'transport': 'streamable-http', 'path': '/api/mcp',
            'production_write': False, 'incremental_switch_write': False}

def rpc_error(ident,code,message):
    return {'jsonrpc':'2.0','id':ident,'error':{'code':code,'message':message}}


def dispatch(db,token,message):
    from .employee_lifecycle import check_employee_access
    from .models import AuditEvent
    user,_ = authenticate(db,token)
    check_employee_access(db,user)
    db.rollback()  # no old snapshot before LabWriteLock
    method=message['method']; ident=message.get('id'); params=message.get('params',{})
    if not isinstance(params,dict): return rpc_error(ident,-32602,'params必须是对象')
    if method=='initialize':
        return {'jsonrpc':'2.0','id':ident,'result':{'protocolVersion':PROTOCOL,'capabilities':{'tools':{'listChanged':False}},'serverInfo':{'name':'label-platform','version':'1.0.0'},'instructions':'仅存量实验操作；继承个人权限。人工确认必须明确，不能操作增量开关、生产绑定或业务投递。'}}
    if method.startswith('notifications/'):
        return None
    if method=='ping': return {'jsonrpc':'2.0','id':ident,'result':{}}
    if method=='tools/list':
        if params.get('cursor'): return rpc_error(ident,-32602,'无效分页游标')
        user,_=authenticate(db,token)
        tools=[{'name':name,'description':t.description,'inputSchema':schema(t),'annotations':{'readOnlyHint':not t.write,'destructiveHint':t.write,'openWorldHint':name=='start_regression'}} for name,t in TOOLS.items() if has_permission(user,t.permission)]
        return {'jsonrpc':'2.0','id':ident,'result':{'tools':tools}}
    if method!='tools/call': return rpc_error(ident,-32601,'不支持的方法')
    name=params.get('name'); arguments=params.get('arguments',{})
    if not isinstance(name,str) or name not in TOOLS or not isinstance(arguments,dict):
        return rpc_error(ident,-32602,'工具名称或参数无效')
    tool=TOOLS[name]
    try:
        with LabWriteLock(db) if tool.write else nullcontext():
            user,credential=authenticate(db,token,lock=True)
            result=invoke(db,user,name,arguments)
            credential.last_used_at=datetime.now(timezone.utc)
            credential.last_tool=name
            if tool.write:
                db.add(AuditEvent(event_key='mcp:'+uuid.uuid4().hex,category='personal_mcp',action=name,subject_type='user',subject_id=str(user.id),actor='mcp:'+str(user.id),payload_json=json.dumps({'arguments_sha256':hashlib.sha256(json.dumps(arguments,sort_keys=True,ensure_ascii=False).encode()).hexdigest(),'permission_revision':user.permission_revision})))
            db.commit()
        result=jsonable_encoder(result)
        if name=='read_asset_image':
            return {'jsonrpc':'2.0','id':ident,'result':{'content':[{'type':'image',**result}],'isError':False}}
        return {'jsonrpc':'2.0','id':ident,'result':{'content':[{'type':'text','text':json.dumps(result,ensure_ascii=False)}],'structuredContent':result if isinstance(result,dict) else {'items':result},'isError':False}}
    except (ValidationError,PromptLabError,BaselineTruthError,HTTPException) as exc:
        db.rollback()
        if isinstance(exc,HTTPException) and exc.status_code==401: raise
        if isinstance(exc,ValidationError): message='参数校验失败，请按工具Schema检查字段、类型及长度'
        elif isinstance(exc,HTTPException): message=str(exc.detail)
        else: message=f'{exc.code}: {exc}'
        return {'jsonrpc':'2.0','id':ident,'result':{'content':[{'type':'text','text':message}],'isError':True}}
    except Exception:
        db.rollback()
        # No exception bodies, prompts, endpoints or credentials returned to clients.
        return {'jsonrpc':'2.0','id':ident,'result':{'content':[{'type':'text','text':'操作未完成，请先查询原请求状态；不要更换幂等编号重复提交。'}],'isError':True}}


def build_personal_mcp_router(current_user):
    router=APIRouter()

    @router.get('/api/me/mcp')
    def get_status(user=Depends(current_user),db:Session=Depends(get_db)):
        return status(db,user)

    @router.post('/api/me/mcp/rotate')
    def issue(user=Depends(current_user),db:Session=Depends(get_db)):
        token=rotate(db,user)
        return JSONResponse({'token':token,'path':'/api/mcp','transport':'streamable-http'},headers={'Cache-Control':'no-store'})

    @router.delete('/api/me/mcp')
    def revoke(user=Depends(current_user),db:Session=Depends(get_db)):
        disable(db,user)
        return {'enabled':False}

    @router.get('/api/mcp')
    def no_sse():
        return Response(status_code=405,headers={'Allow':'POST'})

    @router.post('/api/mcp')
    async def mcp(request:Request,db:Session=Depends(get_db)):
        # Native clients do not send Origin. Browser callers are deliberately not supported.
        if request.headers.get('origin'):
            raise HTTPException(403,'MCP仅接受原生客户端连接')
        authorization=request.headers.get('authorization','')
        if not authorization.startswith('Bearer '): raise HTTPException(401,'需要个人MCP令牌',headers={'WWW-Authenticate':'Bearer'})
        if request.headers.get('mcp-protocol-version',PROTOCOL)!=PROTOCOL:
            raise HTTPException(400,'不支持的MCP协议版本')
        if 'application/json' not in request.headers.get('content-type',''):
            raise HTTPException(415,'需要application/json')
        accept=request.headers.get('accept','')
        if 'application/json' not in accept or 'text/event-stream' not in accept:
            raise HTTPException(406,'Accept必须包含application/json和text/event-stream')
        chunks=[]; size=0
        async for chunk in request.stream():
            size+=len(chunk)
            if size>MAX_BODY: raise HTTPException(413,'请求过大')
            chunks.append(chunk)
        try: message=json.loads(b''.join(chunks))
        except (ValueError,UnicodeError): return JSONResponse(rpc_error(None,-32700,'JSON解析失败'),status_code=400)
        if not isinstance(message,dict) or message.get('jsonrpc')!='2.0' or not isinstance(message.get('method'),str) or ('id' in message and (isinstance(message['id'],bool) or not isinstance(message['id'],(int,str)))):
            return JSONResponse(rpc_error(None,-32600,'无效JSON-RPC请求'),status_code=400)
        if 'id' not in message and not message['method'].startswith('notifications/'):
            return JSONResponse(rpc_error(None,-32600,'工具调用必须带请求编号'),status_code=400)
        result=await run_in_threadpool(dispatch,db,authorization[7:],message)
        if result is None: return Response(status_code=202)
        return JSONResponse(result,headers={'Cache-Control':'no-store'})

    return router

def require_row(db, model, ident):
    row = db.get(model, ident)

def categories(db, name, a):
    from .models import Asset,BaselineSet,BaselineRegressionRun,WorkflowNodeAsset

        _,manifest=lab._require_version(db,draft['source_version_id'])

        a['edits']={k:lab.PromptEdit(**v) for k,v in a.pop('prompts').items()}
        return lab.create_candidate(db,**a)

    if name=='get_reference': return stock.truth_payload(db,**a)
    if name=='confirm_reference': return stock.write_truth(db,**a)

def record(db, action, key, payload):
    append_audit_event(db, category="stock_lab", action=action,
                      subject_type="stock_experiment", subject_id=0,
                      actor=LAB_ACTOR, payload=payload, event_key=key)
    db.flush()


def create_component(db, *, source_asset_id, content, request_id):
    from .evaluation_components import (require_component_asset, create_component_asset,
        save_component_working_copy, generate_component_version, EvaluationComponentError)
    key = f"stock-lab:component:{request_id}"
    digest = _digest({"source_asset_id": source_asset_id, "content": content})
    prior = event(db, key)
    if prior:
        if prior["request_hash"] != digest:
            raise PromptLabError("request_conflict", "幂等键已用于其他组件", status_code=409)
        return prior
    try:
        source = require_component_asset(db, source_asset_id)
        target = create_component_asset(db, category_key=source.category_key,
            asset_key="stock_" + digest[:24], kind=source.kind,
            display_name=source.display_name[:80] + " · 存量候选",
            module_type_key=source.module_type_key,
            prompt_stage=content.get("stage") if source.kind == "prompt" else None,
            created_by=LAB_ACTOR)
        target.copied_from_asset_id = source.id
        save_component_working_copy(db, asset_id=target.id, expected_lock_version=1,
                                    content=content, updated_by=LAB_ACTOR)
        ref = generate_component_version(db, asset_id=target.id, created_by=LAB_ACTOR)
    except EvaluationComponentError as exc:
        raise PromptLabError(exc.code, str(exc), status_code=exc.status_code) from exc
    receipt = {"request_hash": digest, "asset_key": target.asset_key, **asdict(ref)}
    record(db, "component_created", key, receipt)
    return receipt


def event(db, key):
    row = db.scalar(select(AuditEvent).where(AuditEvent.event_key == key))
    return _json(row.payload_json) if row else None


def manifest_payload(db, version_id):
    version, manifest = _require_version(db, version_id)
    return {"source_version_id": version.id, "source_hash": version.canonical_hash,
            "manifest": manifest}


def save_draft(db, *, draft_key, source_version_id, expected_source_hash,
               expected_revision, manifest, note):
    source, original = _require_version(db, source_version_id)
    if source.canonical_hash != expected_source_hash:
        raise PromptLabError("source_hash_mismatch", "源机制已变化", status_code=409)
    if len(str(manifest)) > 512_000:
        raise PromptLabError("manifest_too_large", "工作流草稿超过大小限制")
    if manifest.get("category_key") != original.get("category_key") or manifest.get("schema_version") != "workflow-v3":
        raise PromptLabError("draft_identity_changed", "不能改变草稿所属类目或工作流协议")
    # Full validation occurs when freezing; an incomplete draft can be saved.
    payload = {"draft_key": draft_key, "source_version_id": source.id,
               "source_hash": source.canonical_hash, "revision": expected_revision + 1,
               "manifest": deepcopy(manifest), "note": note}
    key = f"stock-lab:draft:{draft_key}:{expected_revision + 1}"
    prior = event(db, key)
    if prior:
        if prior == payload:
            return prior
        raise PromptLabError("draft_conflict", "草稿已被更新，请读取最新修订", status_code=409)
    if expected_revision:
        previous = event(db, f"stock-lab:draft:{draft_key}:{expected_revision}")
        if not previous or previous["source_version_id"] != source.id:
            raise PromptLabError("draft_conflict", "草稿修订或源机制不匹配", status_code=409)
    record(db, "draft_saved", key, payload)
    return payload


def read_draft(db, draft_key):
    # Exact prefix plus numeric revisions: user keys cannot contain ':' or '%'.
    rows = db.scalars(select(AuditEvent).where(
        AuditEvent.category == "stock_lab",
        AuditEvent.event_key.like(f"stock-lab:draft:{draft_key}:%")))
    values = [_json(row.payload_json) for row in rows]
    if not values:
        raise PromptLabError("draft_not_found", "草稿不存在", status_code=404)
    return max(values, key=lambda p: p["revision"])


def freeze_draft(db, draft_key, revision):
    from .workflow_working_copies import create_immutable_evaluation_mechanism_version, WorkflowWorkingCopyError
    draft = read_draft(db, draft_key)
    if draft["revision"] != revision:
        raise PromptLabError("draft_conflict", "请核对最新草稿后生成版本", status_code=409)
    key = f"stock-lab:freeze:{draft_key}:{revision}"
    prior = event(db, key)
    if prior:
        return prior
    source, _ = _require_version(db, draft["source_version_id"])
    definition = db.get(WorkflowDefinition, source.workflow_definition_id)
    try:
        row = create_immutable_evaluation_mechanism_version(
            db, definition=definition, version="stock-" + _digest(draft)[:32],
            manifest=draft["manifest"], created_by=LAB_ACTOR,
            enforce_authoring_policy=True)
    except WorkflowWorkingCopyError as exc:
        raise PromptLabError(exc.code, str(exc), status_code=exc.status_code) from exc
    receipt = {"workflow_version_id": row.id, "canonical_hash": row.canonical_hash,
               "draft_key": draft_key, "revision": revision,
               "production_binding_changed": False}
    record(db, "draft_frozen", key, receipt)
    return receipt


def baseline_item(db, set_id, item_id):
    row = db.scalar(select(BaselineSetItem).where(
        BaselineSetItem.id == item_id, BaselineSetItem.baseline_set_id == set_id).with_for_update())
    if row is None:
        raise PromptLabError("baseline_item_not_found", "样本条目不存在", status_code=404)
    return row


def truth_payload(db, set_id, item_id):
    from .baseline_truth import effective_truth_from_latest
    from .models import BaselineTruthRevision
    row = baseline_item(db, set_id, item_id)
    # Locking/current read is essential on MySQL REPEATABLE READ: the idempotency
    # lookup may have established a snapshot before the item lock was acquired.
    latest = db.scalar(select(BaselineTruthRevision).where(
        BaselineTruthRevision.baseline_set_item_id == row.id)
        .order_by(BaselineTruthRevision.revision.desc()).limit(1).with_for_update())
    truth = effective_truth_from_latest(row, latest)
    return {"item_id": row.id, "asset_id": row.asset_id, **asdict(truth)}


def write_truth(db, *, set_id, item_id, expected_hash, expected_result,
                expected_level, reason, request_id, confirmed):
    from .baseline_truth import append_baseline_truth_revision, BaselineTruthError
    if not confirmed or not reason.strip():
        raise PromptLabError("confirmation_required", "核对差异并明确确认后才能更新参考答案")
    if (expected_level is None) == (expected_result is None):
        raise PromptLabError("truth_shape_invalid", "只能选择等级或结构化参考答案其中一种")
    body = {"expected_hash": expected_hash, "expected_result": expected_result,
            "expected_level": expected_level, "reason": reason}
    key = f"stock-lab:truth:{set_id}:{item_id}:{request_id}"
    prior = event(db, key)
    if prior:
        if prior["request_hash"] != _digest(body):
            raise PromptLabError("request_conflict", "幂等键已用于其他修改", status_code=409)
        return prior
    current = truth_payload(db, set_id, item_id)
    if current["canonical_hash"] != expected_hash:
        raise PromptLabError("truth_conflict", "参考答案已有新版本，请重新核对差异", status_code=409)
    row = baseline_item(db, set_id, item_id)
    result = {"level": expected_level} if expected_level else expected_result
    try:
        revision = append_baseline_truth_revision(
            db, item=row, expected_level=expected_level, expected_result=result or {},
            truth_contract_key=current["truth_contract_key"], workflow_input=current["workflow_input"],
            source_baseline_run_id=None, source_production_run_id=None,
            source_correction_key=key, reason=reason, actor=LAB_ACTOR)
    except BaselineTruthError as exc:
        raise PromptLabError(exc.code, str(exc), status_code=exc.status_code) from exc
    receipt = {"request_hash": _digest(body), "revision_id": revision.id,
               "canonical_hash": revision.canonical_hash, "previous_hash": expected_hash,
               "confirmation_source": "api_client_assertion", "actor": LAB_ACTOR}
    # Never attribute a machine write to a named employee or call it independent human review.
    record(db, "reference_confirmed_by_client", key, receipt)
    return receipt


def clone_baseline(db, *, source_set_id, item_ids, name, request_id):
    from .audit import canonical_json
    from .baseline_truth import resolve_effective_baseline_truth
    key = f"stock-lab:sample:{request_id}"
    digest = _digest({"source": source_set_id, "items": sorted(set(item_ids)), "name": name})
    prior = event(db, key)
    if prior:
        if prior["request_hash"] != digest:
            raise PromptLabError("request_conflict", "幂等键已用于其他样本集", status_code=409)
        return prior
    source = db.get(BaselineSet, source_set_id)
    if source is None:
        raise PromptLabError("baseline_set_not_found", "源样本集不存在", status_code=404)
    items = db.scalars(select(BaselineSetItem).where(
        BaselineSetItem.baseline_set_id == source_set_id, BaselineSetItem.id.in_(item_ids))).all()
    if len(items) != len(set(item_ids)):
        raise PromptLabError("sample_scope_mismatch", "包含不属于源样本集的条目")
    if db.scalar(select(BaselineSet.id).where(BaselineSet.name == name)):
        raise PromptLabError("sample_name_exists", "样本集名称已存在", status_code=409)
    target = BaselineSet(category_key=source.category_key, name=name,
        description=f"由样本集 {source.id} 复制；冻结复制时参考答案", default_expected_level=source.default_expected_level,
        fingerprint=digest, created_by=LAB_ACTOR)
    db.add(target); db.flush()
    for item in items:
        truth = resolve_effective_baseline_truth(db, item)
        snapshot = _json(item.asset_snapshot_json)
        snapshot["workflow_input"] = truth.workflow_input
        db.add(BaselineSetItem(baseline_set_id=target.id, asset_id=item.asset_id,
            source_package_id=item.source_package_id, expected_level=truth.expected_level,
            expected_result_json=canonical_json(truth.expected_result), truth_contract_key=truth.truth_contract_key,
            asset_snapshot_json=canonical_json(snapshot)))
    receipt = {"baseline_set_id": target.id, "total": len(items), "request_hash": digest}
    record(db, "sample_set_cloned", key, receipt)
    return receipt


def import_baseline(db, *, category_key, name, items, request_id):
    """Import reference annotations for existing assets, never mutate asset data."""
    from .models import Asset
    from .audit import canonical_json
    from .prompt_lab import allowed_categories
    from .baseline_truth import resolve_effective_baseline_truth, BaselineTruthError
    digest = _digest({"category_key": category_key, "name": name, "items": items})
    key = f"stock-lab:import:{request_id}"
    prior = event(db, key)
    if prior:
        if prior["request_hash"] != digest:
            raise PromptLabError("request_conflict", "导入请求内容已变化", status_code=409)
        return prior
    if category_key not in allowed_categories(db):
        raise PromptLabError("category_not_allowed", "类目不存在或已退役")
    if db.scalar(select(BaselineSet.id).where(BaselineSet.name == name)):
        raise PromptLabError("sample_name_exists", "样本集名称已存在", status_code=409)
    if len({i["asset_id"] for i in items}) != len(items):
        raise PromptLabError("duplicate_asset", "同一素材不能重复导入")
    kinds = {"level" if i.get("expected_level") else "structured" for i in items}
    if len(kinds) != 1:
        raise PromptLabError("mixed_truth_kind", "一个样本集不能混合等级与结构化参考答案")
    target = BaselineSet(category_key=category_key, name=name, description="存量API导入已有素材参考答案",
                         default_expected_level="L1", fingerprint=digest, created_by=LAB_ACTOR)
    db.add(target); db.flush()
    for value in items:
        asset = db.get(Asset, value["asset_id"])
        if asset is None or asset.category_key != category_key or asset.status == "deleted":
            raise PromptLabError("asset_scope_mismatch", "素材不存在、已删除或不属于所选类目")
        level = value.get("expected_level")
        result = value.get("expected_result")
        if (level is not None and result is not None) or (level is None and (result is None or not value.get("truth_contract_key"))):
            raise PromptLabError("truth_shape_invalid", "每条只可提供等级或带合同的结构化参考答案")
        row = BaselineSetItem(baseline_set_id=target.id, asset_id=asset.id,
            expected_level=level, expected_result_json=canonical_json(result) if result is not None else None,
            truth_contract_key=value.get("truth_contract_key"),
            asset_snapshot_json=canonical_json({"schema_version": "baseline-asset-v1", "asset_id": asset.id,
                "category_key": category_key, "name": asset.original_name, "sha256": asset.sha256,
                "workflow_input": value.get("workflow_input") or {}}))
        db.add(row); db.flush()
        try:
            resolve_effective_baseline_truth(db, row)
        except BaselineTruthError as exc:
            raise PromptLabError(exc.code, str(exc), status_code=exc.status_code) from exc
    receipt = {"baseline_set_id": target.id, "total": len(items), "request_hash": digest}
    record(db, "sample_set_imported", key, receipt)
    return receipt


def control_items(db, *, run_id, item_ids, action, request_id):
    """Bounded, atomic control. Busy paid work is never interrupted or replayed."""
    from .models import BaselineRegressionRun, BaselineRegressionItem, ProductionRun, ProductionStepAttempt, WorkflowModelInvocation
    from .workflow_snapshot_store import read_run_snapshot_document
    from .workflow_runtime import resume_from_checkpoint, retry_failed_run_from_checkpoint, terminalize_runtime_run, WorkflowRuntimeError
    from .native_baseline_regression import finalize_failed_native_baseline_item
    key = f"stock-lab:control:{run_id}:{request_id}"
    digest = _digest({"items": sorted(set(item_ids)), "action": action})
    prior = event(db, key)
    if prior:
        if prior["request_hash"] != digest:
            raise PromptLabError("request_conflict", "控制请求幂等键已用于其他操作", status_code=409)
        return prior
    aggregate = db.get(BaselineRegressionRun, run_id)
    if aggregate is None or aggregate.execution_engine != "native_workflow":
        raise PromptLabError("native_stock_required", "仅支持原生存量回归；旧引擎需迁移后操作", status_code=409)
    items = db.scalars(select(BaselineRegressionItem).where(
        BaselineRegressionItem.run_id == run_id, BaselineRegressionItem.id.in_(item_ids)).order_by(BaselineRegressionItem.production_run_id)).all()
    if len(items) != len(set(item_ids)):
        raise PromptLabError("item_scope_mismatch", "存在不属于本轮的条目")
    children = []
    for item in items:
        child = db.scalar(select(ProductionRun).where(ProductionRun.id == item.production_run_id)
                          .with_for_update().execution_options(populate_existing=True))
        context = read_run_snapshot_document(child).get("runtime_context", {}) if child else {}
        if (child is None or child.source_type != "baseline_regression_item" or child.source_id != item.id
                or child.environment != "dry_run" or context.get("run_mode") != "stock_regression"
                or context.get("baseline_run_id") != aggregate.id or context.get("baseline_item_id") != item.id):
            raise PromptLabError("stock_scope_mismatch", "运行归属或隔离环境不一致，未执行操作", status_code=409)
        busy = db.scalar(select(ProductionStepAttempt.id).where(
            ProductionStepAttempt.run_id == child.id, ProductionStepAttempt.status.in_(["running", "leased"])).limit(1))
        reserved = db.scalar(select(WorkflowModelInvocation.id).where(
            WorkflowModelInvocation.production_run_id == child.id, WorkflowModelInvocation.status == "reserved").limit(1))
        if busy or reserved:
            raise PromptLabError("stock_item_in_flight", "所选条目仍有在途调用，请等待其完成后重试；本次整批未执行", status_code=409)
        children.append((item, child))
    changed = []
    before = [{"item_id": item.id, "status": item.status, "error_message": item.error_message,
               "result_snapshot_json": item.result_snapshot_json, "runtime_status": child.status}
              for item, child in children]
    try:
        for item, child in children:
            if action == "pause":
                if child.status not in {"queued", "running", "retryable", "paused"}:
                    raise PromptLabError("pause_state_invalid", "所选条目不能暂停", status_code=409)
                if child.status != "paused":
                    child.status = "paused"
            elif action == "resume":
                if child.status != "paused":
                    raise PromptLabError("resume_state_invalid", "只能继续已暂停条目", status_code=409)
                resume_from_checkpoint(db, child.id)
            elif action == "retry":
                retry_failed_run_from_checkpoint(db, child.id, actor=LAB_ACTOR, restart_transport_window=False)
            elif action == "cancel":
                if child.status not in {"succeeded", "failed", "canceled"}:
                    terminalize_runtime_run(db, child, status="canceled", error_code="operator_canceled", error_message="存量API取消")
            else:
                raise PromptLabError("action_invalid", "不支持的操作")
            db.flush()
            if action == "cancel" and child.status == "canceled":
                finalize_failed_native_baseline_item(db, child)
            changed.append({"item_id": item.id, "status": child.status})
    except WorkflowRuntimeError as exc:
        raise PromptLabError(exc.code, str(exc), status_code=exc.status_code) from exc
    receipt = {"request_hash": digest, "items": changed, "action": action}
    record(db, "stock_control_previous_results", key + ":before", {"items": before})
    record(db, "stock_items_controlled", key, receipt)
    return receipt

"""Explicit stock-only endpoints; never mount production APIs with a machine user."""
from typing import Any
from typing import Literal

from fastapi import Depends, Query
from pydantic import BaseModel, ConfigDict, Field
from sqlalchemy import select
from sqlalchemy.orm import Session

from .database import get_db
from . import stock_lab


class StrictRequest(BaseModel):
    model_config = ConfigDict(extra="forbid")


class DraftRequest(StrictRequest):
    source_version_id: int = Field(gt=0)
    expected_source_hash: str = Field(pattern="^[0-9a-f]{64}$")
    expected_revision: int = Field(ge=0)
    manifest: dict[str, Any]
    note: str = Field(default="", max_length=2000)


class FreezeRequest(StrictRequest):
    revision: int = Field(gt=0)


class SampleCloneRequest(StrictRequest):
    source_set_id: int = Field(gt=0)
    item_ids: list[int] = Field(min_length=1, max_length=10000)
    name: str = Field(min_length=1, max_length=160)
    request_id: str = Field(pattern="^[A-Za-z0-9.-]{8,80}$")


class TruthRequest(StrictRequest):
    expected_hash: str = Field(pattern="^[0-9a-f]{64}$")
    expected_result: dict[str, Any] | None = None
    expected_level: str | None = Field(default=None, pattern="^L[1-5]$")
    reason: str = Field(min_length=1, max_length=2000)
    request_id: str = Field(pattern="^[A-Za-z0-9.-]{8,80}$")
    confirmed: bool = Field(strict=True)


class ControlRequest(StrictRequest):
    item_ids: list[int] = Field(min_length=1, max_length=200)
    action: Literal["pause", "resume", "retry", "cancel"]
    request_id: str = Field(pattern="^[A-Za-z0-9.-]{8,80}$")


class ComponentRequest(StrictRequest):
    source_asset_id: int = Field(gt=0)
    content: dict[str, Any]
    request_id: str = Field(pattern="^[A-Za-z0-9.-]{8,80}$")


class ImportItem(StrictRequest):
    asset_id: int = Field(gt=0)
    expected_level: str | None = Field(default=None, pattern="^L[1-5]$")
    expected_result: dict[str, Any] | None = None
    truth_contract_key: str | None = Field(default=None, max_length=120)
    workflow_input: dict[str, Any] = Field(default_factory=dict)


class ImportRequest(StrictRequest):
    category_key: str = Field(min_length=1, max_length=40)
    name: str = Field(min_length=1, max_length=160)
    items: list[ImportItem] = Field(min_length=1, max_length=10000)
    request_id: str = Field(pattern="^[A-Za-z0-9.-]{8,80}$")

    def component(payload: ComponentRequest, _client=Depends(lab_client), db: Session = Depends(get_db)):

        from .prompt_lab import PromptLabError, _json

            try:
                row = require_component_asset(db, asset_id)
                working = get_component_working_copy(db, asset_id)

            "component_limit": 500,

def _lock_engine(business_engine: Any) -> Any:
    """命名锁专用连接通道（ls-review-r24 F1）。

    锁连接若从业务连接池借，持锁者做事务还要再借一条，等锁的请求也各占一条等着，
    池一小（合法配置 pool_size=1/2、无溢出）就互相卡死。这里用同一个库地址另建不入池的
    NullPool 引擎：等锁只占自己的物理连接，关闭即断开，MySQL 随之释放会话级命名锁。
    """
    key = business_engine.url.render_as_string(hide_password=False)
    engine = _LOCK_ENGINES.get(key)
    if engine is None:
        from sqlalchemy import create_engine
        from sqlalchemy.pool import NullPool

        engine = create_engine(business_engine.url, poolclass=NullPool, connect_args={"connect_timeout": 10})
        _LOCK_ENGINES[key] = engine
    return engine


class LabWriteLock:
    """实验室写操作（建候选、开回归、取消）的全局串行锁。

    ls-review-r23 R1/S1：并发请求各自读到"没有在跑、额度够"，就能越过一轮上限与日额度；
    同一候选首次并发创建会撞唯一键成 500。所以必须在本请求第一次读库之前拿锁、提交之后才放：
    MySQL 用独立连接上的命名锁（会话连接提交后会还回连接池，锁不能挂在它上面），
    锁前没有读过库也就没有旧的可重复读视图；SQLite 在本会话 BEGIN IMMEDIATE。
    """

    def __init__(self, db: Session) -> None:
        self._db = db
        self._conn = None

    def __enter__(self) -> "LabWriteLock":
        if self._db.in_transaction():
            raise RuntimeError("prompt lab lock must be taken before the first read of this session")
        engine = self._db.get_bind()
        if engine.dialect.name == "mysql":
            conn = _lock_engine(engine).connect()
            try:
                got = conn.execute(
                    text("SELECT GET_LOCK(:name, :wait)"), {"name": LAB_LOCK_NAME, "wait": LAB_LOCK_WAIT_SECONDS}
                ).scalar()
            except BaseException:
                # 服务端可能已拿到锁而客户端没读到结果（ls-review-r24 F3）：作废并物理断开，锁随会话释放。
                conn.invalidate()
                conn.close()
                raise
            if got != 1:
                conn.close()
                raise PromptLabError("lab_busy", "实验室正在处理另一个写请求，请稍后重试", status_code=409)
            self._conn = conn
        else:
            from .database_capabilities import DatabaseCapabilities

            DatabaseCapabilities.for_dialect(engine.dialect.name).reserve_claim_transaction(self._db)
        return self

    def __exit__(self, *exc: Any) -> None:
        if self._conn is None:
            return
        conn, self._conn = self._conn, None
        try:
            conn.execute(text("SELECT RELEASE_LOCK(:name)"), {"name": LAB_LOCK_NAME})
        finally:
            conn.close()  # NullPool：物理断开，即使 RELEASE_LOCK 失败锁也随会话释放

def _digest(value: Any) -> str:
    return hashlib.sha256(
        json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"), default=str).encode()
    ).hexdigest()


def _json(value: str | None) -> Any:
    try:
        return json.loads(value or "{}")
    except json.JSONDecodeError:
        return None


def _iso(value: datetime | None) -> str | None:
    if value is None:
        return None
    if value.tzinfo is None:
        value = value.replace(tzinfo=timezone.utc)
    return value.astimezone(timezone(timedelta(hours=8))).isoformat()

def _comparable(manifest: Mapping[str, Any]) -> dict[str, Any]:
    """比对"候选只差提示词"时去掉模型计费价目表：调价只影响预算，不改模型行为。

    .137 实测：价目表 v2→v3 之后重新冻结老机制，模型快照只有 pricing_contract 与随之变化的
    snapshot_hash 不同，原来会把所有老机制都判成"除提示词外不一致"。模型身份、参数、输出合同照样严格比对。
    """
    copy = deepcopy(dict(manifest))
    for node in copy.get("nodes") or []:
        snapshot = (node.get("config") or {}).get("model_snapshot") if isinstance(node, dict) else None
        if isinstance(snapshot, dict):
            snapshot.pop("snapshot_hash", None)
            runtime = snapshot.get("runtime_config")
            if isinstance(runtime, dict):
                runtime.pop("pricing_contract", None)
    return copy

MAX_MECHANISMS_SCANNED = 500


def list_mechanisms_page(
    db: Session,
    *,
    limit: int = 30,
    before_id: int | None = None,
    scan_cap: int = MAX_MECHANISMS_SCANNED,
) -> dict[str, Any]:
    """有界分页列出实验室可用的评测机制版本（按 id 倒序）。

    Astra 全仓复审 F1：原实现不带 LIMIT 把全部 workflow-v3 版本连同大正文载入再在 Python 里过滤。
    现在先用流程定义的 allowed_categories_json 在 SQL 里缩小范围，按 id 分页只取 id，
    每条再单独载入核对冻结类目并立即移出会话，内存上界按页大小而不是机制总量。

    复审 N1：一个流程定义可能同时允许多个类目，冻结类目不在开放范围的版本会被跳过；一次调用最多检查
    ``scan_cap`` 个版本（每页取 min(limit, 剩余额度)，上限在页内检查），扫满还没凑够 ``limit`` 条时
    返回 ``scan_capped=True``，不再静默返回空列表或残缺列表。还有更早的版本时给出 ``next_before_id``，
    调用方带上 ``before_id`` 接着翻。
    """
    from .models import WorkflowDefinition

    categories = allowed_categories(db)
    definition_ids = [
        definition_id
        for definition_id, allowed in db.execute(
            select(WorkflowDefinition.id, WorkflowDefinition.allowed_categories_json)
        )
        if isinstance(parsed := _json(allowed), list) and set(parsed) & categories
    ]
    out: list[dict[str, Any]] = []
    page_info: dict[str, Any] = {"items": out, "next_before_id": None, "scan_capped": False, "scanned": 0}
    if not definition_ids:
        return page_info
    base = select(WorkflowVersion.id).where(
        WorkflowVersion.workflow_schema_version == "workflow-v3",
        WorkflowVersion.workflow_definition_id.in_(definition_ids),
    )
    last_id = before_id
    scanned = 0
    exhausted = False
    while len(out) < limit and scanned < scan_cap:
        query = base if last_id is None else base.where(WorkflowVersion.id < last_id)
        page = db.scalars(query.order_by(WorkflowVersion.id.desc()).limit(min(limit, scan_cap - scanned))).all()
        if not page:
            exhausted = True
            break
        for version_id in page:
            last_id = version_id
            scanned += 1
            version = db.get(WorkflowVersion, version_id)
            category = _manifest(version).get("category_key")
            if category in categories:
                out.append({
                    "id": version.id,
                    "version": version.version,
                    "canonical_hash": version.canonical_hash,
                    "category_key": category,
                    "status": version.status,
                    "created_by": version.created_by,
                    "created_at": _iso(version.created_at),
                    "is_lab_candidate": is_lab_version(version),
                })
            db.expunge(version)
            if len(out) >= limit:
                break
    if not exhausted and last_id is not None:
        # 只查一个 id 判断后面还有没有，避免"刚好扫完"时误报不完整。
        exhausted = db.scalar(base.where(WorkflowVersion.id < last_id).limit(1)) is None
    page_info["scanned"] = scanned
    if not exhausted:
        page_info["next_before_id"] = last_id
        page_info["scan_capped"] = len(out) < limit
    return page_info


def list_mechanisms(db: Session, *, limit: int = 30) -> list[dict[str, Any]]:
    """兼容旧调用：只要第一页的条目。需要判断是否扫满上限、继续翻页时用 list_mechanisms_page。"""
    return list_mechanisms_page(db, limit=limit)["items"]


@dataclass(frozen=True)
class PromptEdit:
    system_prompt: str | None = None
    user_prompt: str | None = None


def _checked_text(value: str, *, field: str, node_id: str, allowed: set[str], allow_empty: bool = False) -> str:
    if not isinstance(value, str) or (not value.strip() and not allow_empty):
        raise PromptLabError("prompt_text_empty", f"{node_id}.{field} 不能为空")
    if len(value) > MAX_PROMPT_CHARS:
        raise PromptLabError("prompt_text_too_long", f"{node_id}.{field} 超过 {MAX_PROMPT_CHARS} 字")
    unknown = sorted(set(_PLACEHOLDER.findall(value)) - allowed)
    if unknown:
        raise PromptLabError(
            "prompt_placeholder_not_allowed",
            f"{node_id}.{field} 出现源提示词里没有的占位符：" + "、".join(unknown),
        )
    return value


def create_candidate(
    db: Session,
    *,
    source_version_id: int,
    expected_source_hash: str,
    edits: Mapping[str, PromptEdit],
    output_schemas: Mapping[str, dict[str, Any]] | None = None,
    note: str = "",
) -> dict[str, Any]:
    """在源机制上修改提示词和输出字段定义，冻结成一个新的评测机制版本（同样的输入得到同一个版本）。"""

    from .workflow_working_copies import WorkflowWorkingCopyError, create_immutable_evaluation_mechanism_version

    source, before = _require_version(db, source_version_id)
    if source.canonical_hash != expected_source_hash:
        raise PromptLabError("source_hash_mismatch", "源机制的 canonical_hash 与请求不一致，请重新读取机制", status_code=409)
    if len(note) > MAX_NOTE_CHARS:
        raise PromptLabError("note_too_long", f"说明不能超过 {MAX_NOTE_CHARS} 字")
    nodes = {str(node.get("node_instance_id")): node for node in _prompt_nodes(before)}
    output_schemas = output_schemas or {}
    unknown = sorted((set(edits) | set(output_schemas)) - set(nodes))
    if unknown:
        raise PromptLabError("prompt_node_unknown", "源机制里没有这些提示词节点：" + "、".join(unknown))
    planned: dict[str, dict[str, Any]] = {}
    for node_id, edit in sorted(edits.items()):
        content = _prompt_content(nodes[node_id])
        old_system = str(content.get("system_prompt") or "")
        old_user = str(content.get("user_prompt") or "")
        allowed = set(_PLACEHOLDER.findall(old_system + "\n" + old_user))
        system = old_system if edit.system_prompt is None else _checked_text(edit.system_prompt, field="system_prompt", node_id=node_id, allowed=allowed)
        user = old_user if edit.user_prompt is None else _checked_text(edit.user_prompt, field="user_prompt", node_id=node_id, allowed=allowed, allow_empty=True)
        if system == old_system and user == old_user:
            continue
        planned[node_id] = {"system": system, "user": user, "content": content, "cfg": nodes[node_id]["config"]}
    from .knowledge_graph_contracts import extend_knowledge_graph_contract, KnowledgeGraphContractError, knowledge_graph_strict_output_contract
    schemas = {}
    for node_id, schema in sorted(output_schemas.items()):
        try:
            contract = extend_knowledge_graph_contract(nodes[node_id]["config"].get("output_contract_key"), schema)
        except KnowledgeGraphContractError as exc:
            raise PromptLabError(exc.code, str(exc)) from exc
        if contract.snapshot() != nodes[node_id]["config"].get("output_contract_snapshot"):
            schemas[node_id] = contract
    if not planned and not schemas:
        raise PromptLabError("prompt_unchanged", "提示词没有任何变化")
    patch_hash = _digest({
        "source": source.canonical_hash,
        "prompts": {node_id: [plan["system"], plan["user"]] for node_id, plan in planned.items()},
        **({"output_schemas": {k: v.snapshot() for k, v in schemas.items()}} if schemas else {}),
    })
    version_name = f"{LAB_VERSION_PREFIX}{source.id}-{patch_hash[:16]}"
    candidate = db.scalar(
        select(WorkflowVersion).where(
            WorkflowVersion.workflow_definition_id == source.workflow_definition_id,
            WorkflowVersion.version == version_name,
        )
    )
    created = candidate is None
    expected_manifest = deepcopy(before)
    schema_nodes = set()
    for node_id, contract in schemas.items():
        old_key = nodes[node_id]["config"]["output_contract_key"]
        # A route's prompts and projections must share one field definition.
        peers = [n for n in _prompt_nodes(before) if n["config"].get("output_contract_key") == old_key]
        if len(peers) != 1:
            raise PromptLabError("schema_route_ambiguous", "同一路线有多个提示词节点，无法自动确定投影字段定义")
        for node in expected_manifest["nodes"]:
            cfg = node.get("config") or {}
            if node.get("node_instance_id") == node_id:
                cfg["output_contract_snapshot"] = contract.snapshot()
                cfg["output_contract"] = knowledge_graph_strict_output_contract(old_key, contract.snapshot())
                node["output_schema"] = contract.output_schema
                schema_nodes.add(node_id)
            elif node.get("kind") == "downstream_projection" and cfg.get("field_contract_key") == old_key:
                cfg["field_contract_snapshot"] = contract.snapshot()
                schema_nodes.add(node["node_instance_id"])
    if candidate is None:
        category = before["category_key"]
        manifest = deepcopy(expected_manifest)
        for node_id, plan in planned.items():
            cfg = plan["cfg"]
            old = db.get(PromptVersion, cfg["prompt_version_id"])
            if old is None or old.category_key != category:
                raise PromptLabError("mechanism_prompt_missing", f"{node_id} 引用的提示词版本已不可用", status_code=409)
            prompt_version = f"{version_name}-{node_id}"[:120]
            new_prompt = db.scalar(
                select(PromptVersion).where(
                    PromptVersion.category_key == category,
                    PromptVersion.stage == old.stage,
                    PromptVersion.version == prompt_version,
                )
            )
            if new_prompt is None:
                new_prompt = PromptVersion(
                    category_key=category,
                    pipeline_scope=old.pipeline_scope,
                    stage=old.stage,
                    name=f"{old.name[:100]} · 实验室",
                    version=prompt_version,
                    system_prompt=plan["system"],
                    user_prompt=plan["user"],
                    rubric_version=str(plan["content"].get("rubric_version") or old.rubric_version),
                    status="draft",
                    source="manual",
                    created_by=LAB_ACTOR,
                    change_note=("提示词实验室候选（仅评测，不会自动上线）。" + (note[:500] if note else "")),
                )
                db.add(new_prompt)
                db.flush()
            if new_prompt.system_prompt != plan["system"] or new_prompt.user_prompt != plan["user"]:
                raise PromptLabError("prompt_version_conflict", f"{node_id} 同名实验提示词内容不一致", status_code=409)
            asset = db.scalar(
                select(WorkflowNodeAsset).where(
                    WorkflowNodeAsset.category_key == category,
                    WorkflowNodeAsset.asset_key == cfg["asset_key"],
                    WorkflowNodeAsset.kind == "prompt",
                )
            )
            if asset is None:
                raise PromptLabError("prompt_asset_missing", f"{node_id} 的提示词组件不存在", status_code=409)
            if db.scalar(
                select(WorkflowNodeAssetBinding.id).where(
                    WorkflowNodeAssetBinding.asset_id == asset.id,
                    WorkflowNodeAssetBinding.prompt_version_id == new_prompt.id,
                )
            ) is None:
                db.add(WorkflowNodeAssetBinding(asset_id=asset.id, prompt_version_id=new_prompt.id, created_by=LAB_ACTOR))
                db.flush()
            node_cfg = next(n for n in _prompt_nodes(manifest) if n.get("node_instance_id") == node_id)["config"]
            node_cfg["prompt_version_id"] = new_prompt.id
            node_cfg.pop("component_snapshot", None)
        try:
            candidate = create_immutable_evaluation_mechanism_version(
                db,
                definition=source.definition,
                version=version_name,
                manifest=manifest,
                created_by=LAB_ACTOR,
            )
        except WorkflowWorkingCopyError as exc:
            # 源机制引用的模型/组件现在冻结不了（例如模型配置被改成未登记能力档案的型号）。
            raise PromptLabError(exc.code, f"候选机制无法冻结：{exc}", status_code=409) from exc
    elif not is_lab_version(candidate):
        raise PromptLabError("candidate_name_conflict", "候选版本名已被非实验室版本占用", status_code=409)
    # 逐项核对：只允许请求中的提示词/字段定义和同路线投影同步，其他内容不变。
    frozen = _manifest(candidate)
    restored = deepcopy(frozen)
    before_nodes = {str(n.get("node_instance_id")): n for n in _prompt_nodes(before)}
    for node in _prompt_nodes(restored):
        node_id = str(node.get("node_instance_id"))
        content = _prompt_content(node)
        expected = planned.get(node_id)
        if expected is not None:
            if content.get("system_prompt") != expected["system"] or (content.get("user_prompt") or "") != expected["user"]:
                raise PromptLabError("candidate_differs_beyond_prompts", f"{node_id} 冻结后的提示词与请求不一致", status_code=409)
            for key in ("prompt_version_id", "component_snapshot"):
                node["config"][key] = deepcopy(before_nodes[node_id]["config"][key])
    if _comparable(restored) != _comparable(expected_manifest):
        # 源机制冻结之后平台组件（模型登记、输出合同等）又变过，重新冻结会带进别的差异，
        # 候选与源机制就不再只差提示词，对比失去意义。
        raise PromptLabError(
            "candidate_differs_beyond_prompts",
            "候选机制存在请求之外的差异（源机制冻结后平台组件有变化），请换用最新的源机制",
            status_code=409,
        )
    receipt = {
        "source_version_id": source.id,
        "source_hash": source.canonical_hash,
        "candidate_version_id": candidate.id,
        "candidate_hash": candidate.canonical_hash,
        "candidate_version": candidate.version,
        "changed_nodes": sorted(planned),
        "schema_changed_nodes": sorted(schema_nodes),
        "output_schema_changes": {node_id: {
            "previous_contract_hash": nodes[node_id]["config"].get("output_contract_snapshot", {}).get("contract_hash"),
            "contract_hash": contract.snapshot()["contract_hash"],
            "fields": list(contract.field_order),
            "added_fields": sorted(set(contract.field_order) - set(nodes[node_id]["config"].get("output_contract_snapshot", {}).get("field_order", []))),
            "removed_fields": sorted(set(nodes[node_id]["config"].get("output_contract_snapshot", {}).get("field_order", [])) - set(contract.field_order)),
        } for node_id, contract in schemas.items()},
        "patch_hash": patch_hash,
        "note": note,
        "created": created,
    }
    if created:
        append_audit_event(
            db,
            category="prompt_lab",
            action="candidate_created",
            subject_type="workflow_version",
            subject_id=candidate.id,
            actor=LAB_ACTOR,
            payload=receipt,
            event_key=f"prompt-lab:candidate:{candidate.id}",
        )
    return receipt

def start_run(
    db: Session,
    *,
    workflow_version_id: int,
    baseline_set_id: int,
    baseline_item_ids: Sequence[int] | None = None,
    client_request_id: str | None = None,
    note: str = "",
    now: datetime | None = None,
) -> dict[str, Any]:
    """在已有基准集上用冻结机制跑真实回归，验证队列、业务写入隔离。"""

    from .native_baseline_regression import (
        NativeBaselineRegressionError,
        create_native_baseline_run,
    )

    version, manifest = _require_version(db, workflow_version_id)
    if version.status != "active":
        raise PromptLabError("mechanism_not_active", "只能用可执行（active）的评测机制版本跑回归", status_code=409)
    baseline_set = db.get(BaselineSet, baseline_set_id)
    if baseline_set is None:
        raise PromptLabError("baseline_set_not_found", "基准集不存在", status_code=404)
    if baseline_set.category_key != manifest.get("category_key"):
        raise PromptLabError("baseline_set_category_mismatch", "基准集与评测机制的类目不一致")
    if len(note) > MAX_NOTE_CHARS:
        raise PromptLabError("note_too_long", f"说明不能超过 {MAX_NOTE_CHARS} 字")
    items = list(baseline_set.items)
    if baseline_item_ids is not None:
        wanted = list(dict.fromkeys(int(i) for i in baseline_item_ids))
        by_id = {item.id: item for item in items}
        missing = [i for i in wanted if i not in by_id]
        if missing:
            raise PromptLabError("baseline_items_invalid", f"有 {len(missing)} 个条目不属于这个基准集：{missing[:20]}")
        items = [by_id[i] for i in wanted]
    if not items:
        raise PromptLabError("baseline_items_empty", "没有可跑的条目")
    # ls-review-r23 R2：同一个 client_request_id 只认同一份请求（机制、集合、实际条目），
    # 请求记录落在唯一键的审计事件上；内容不同一律 409，不能把没跑过的样本说成已跑。
    request_digest = _digest({
        "workflow_version_id": version.id,
        "workflow_hash": version.canonical_hash,
        "baseline_set_id": baseline_set.id,
        "baseline_item_ids": sorted(item.id for item in items),
    })
    request_key = f"prompt-lab:request:{client_request_id}" if client_request_id is not None else None
    if request_key is not None:
        prior = db.scalar(select(AuditEvent).where(AuditEvent.event_key == request_key))
        if prior is not None:
            recorded = _json(prior.payload_json)
            recorded = recorded if isinstance(recorded, dict) else {}
            existing = db.get(BaselineRegressionRun, recorded.get("run_id")) if recorded.get("run_id") else None
            if recorded.get("request_digest") != request_digest or existing is None:
                raise PromptLabError("client_request_id_reused", "client_request_id 已用于另一份请求（机制、基准集或条目不同）", status_code=409)
            return {**run_summary(existing), "created": False}
    key = f"prompt-lab:{client_request_id or uuid.uuid4().hex}:{request_digest[:16]}"
    active = db.scalar(
        select(func.count()).select_from(BaselineRegressionRun).where(
            BaselineRegressionRun.created_by == LAB_ACTOR,
            BaselineRegressionRun.status == "running",
        )
    ) or 0
    if active >= MAX_ACTIVE_RUNS:
        raise PromptLabError("lab_run_in_progress", "实验室已有一轮回归在跑，等它结束或先取消", status_code=409)
    now = now or datetime.now(timezone.utc)
    day_start = now.astimezone(timezone(timedelta(hours=8))).replace(hour=0, minute=0, second=0, microsecond=0)
    used = db.scalar(
        select(func.coalesce(func.sum(BaselineRegressionRun.total), 0)).where(
            BaselineRegressionRun.created_by == LAB_ACTOR,
            BaselineRegressionRun.created_at >= day_start.astimezone(timezone.utc).replace(tzinfo=None),
        )
    ) or 0
    cap = daily_item_cap()
    if used + len(items) > cap:
        raise PromptLabError(
            "lab_daily_cap_reached",
            f"今天（东八区）实验室已跑 {used} 条，本轮 {len(items)} 条会超过每日上限 {cap} 条",
            status_code=429,
        )
    try:
        run = create_native_baseline_run(
            db,
            baseline_set=baseline_set,
            workflow_version=version,
            actor=LAB_ACTOR,
            idempotency_key=key,
            items=items,
            model_execution_mode=MODEL_EXECUTION_MODE,
            real_execution_confirmed=MODEL_EXECUTION_MODE == "real",
            inference_mode="online",
            evaluation_purpose="golden_calibration",
        )
    except NativeBaselineRegressionError as exc:
        raise PromptLabError(exc.code, str(exc), status_code=exc.status_code) from exc
    append_audit_event(
        db,
        category="prompt_lab",
        action="run_created",
        subject_type="baseline_regression_run",
        subject_id=run.id,
        actor=LAB_ACTOR,
        payload={
            "baseline_set_id": baseline_set.id,
            "workflow_version_id": version.id,
            "workflow_canonical_hash": version.canonical_hash,
            "total": run.total,
            "partial": baseline_item_ids is not None,
            "note": note,
            "daily_used_before": int(used),
            "daily_cap": cap,
        },
        event_key=f"prompt-lab:run:{run.id}",
    )
    if request_key is not None:
        append_audit_event(
            db,
            category="prompt_lab",
            action="run_request",
            subject_type="baseline_regression_run",
            subject_id=run.id,
            actor=LAB_ACTOR,
            payload={"request_digest": request_digest, "run_id": run.id, "client_request_id": client_request_id},
            event_key=request_key,
        )
    return {**run_summary(run), "created": True}


def _require_run(db: Session, run_id: int) -> BaselineRegressionRun:
    run = db.get(BaselineRegressionRun, run_id)
    if run is None:
        raise PromptLabError("run_not_found", "回归不存在", status_code=404)
    if run.category_key not in allowed_categories(db):
        raise PromptLabError("category_not_allowed", "这个回归不在实验室开放的类目内", status_code=403)
    return run


def run_summary(run: BaselineRegressionRun) -> dict[str, Any]:
    return {
        "id": run.id,
        "baseline_set_id": run.baseline_set_id,
        "workflow_version_id": run.workflow_version_id,
        "category_key": run.category_key,
        "status": run.status,
        "total": run.total,
        "completed": run.completed,
        "failed": run.failed,
        "valid_predictions": run.valid_predictions,
        "created_by": run.created_by,
        "is_lab_run": run.created_by == LAB_ACTOR,
        "created_at": _iso(run.created_at),
        "finished_at": _iso(run.finished_at),
    }


def get_run(db: Session, run_id: int) -> dict[str, Any]:
    return run_summary(_require_run(db, run_id))


def list_runs(db: Session, *, limit: int = 30, lab_only: bool = False) -> list[dict[str, Any]]:
    query = select(BaselineRegressionRun).where(BaselineRegressionRun.category_key.in_(sorted(allowed_categories(db))))
    if lab_only:
        query = query.where(BaselineRegressionRun.created_by == LAB_ACTOR)
    rows = db.scalars(query.order_by(BaselineRegressionRun.id.desc()).limit(max(1, min(limit, 200))))
    return [run_summary(run) for run in rows]


def list_baseline_sets(db: Session) -> list[dict[str, Any]]:
    rows = db.execute(
        select(BaselineSet, func.count(BaselineSetItem.id))
        .join(BaselineSetItem, BaselineSetItem.baseline_set_id == BaselineSet.id, isouter=True)
        .where(BaselineSet.category_key.in_(sorted(allowed_categories(db))))
        .group_by(BaselineSet.id)
        .order_by(BaselineSet.id)
    ).all()
    return [
        {"id": row.id, "name": row.name, "category_key": row.category_key, "items": int(count)}
        for row, count in rows
    ]


def run_items(db: Session, run_id: int, *, offset: int = 0, limit: int = 200) -> dict[str, Any]:
    """逐条：模型输出、原始输出、跑时冻结的真值、当前最新真值。怎么算准确率由调用方自己定。"""

    from .baseline_truth import BaselineTruthError, effective_truth_from_latest

    run = _require_run(db, run_id)
    limit = max(1, min(limit, MAX_ITEMS_PAGE))
    offset = max(0, offset)
    total = db.scalar(select(func.count()).select_from(BaselineRegressionItem).where(BaselineRegressionItem.run_id == run.id)) or 0
    # ls-review-r23 S2：按页批量取基准条目、图片与最新真值修订，不逐条懒加载。
    rows = db.scalars(
        select(BaselineRegressionItem)
        .where(BaselineRegressionItem.run_id == run.id)
        .options(selectinload(BaselineRegressionItem.baseline_set_item), selectinload(BaselineRegressionItem.asset))
        .order_by(BaselineRegressionItem.id)
        .offset(offset)
        .limit(limit)
    ).all()
    set_item_ids = sorted({row.baseline_set_item_id for row in rows})
    latest_by_item: dict[int, BaselineTruthRevision] = {}
    if set_item_ids:
        newest = (
            select(
                BaselineTruthRevision.baseline_set_item_id.label("item_id"),
                func.max(BaselineTruthRevision.revision).label("revision"),
            )
            .where(BaselineTruthRevision.baseline_set_item_id.in_(set_item_ids))
            .group_by(BaselineTruthRevision.baseline_set_item_id)
            .subquery()
        )
        for revision in db.scalars(
            select(BaselineTruthRevision).join(
                newest,
                and_(
                    BaselineTruthRevision.baseline_set_item_id == newest.c.item_id,
                    BaselineTruthRevision.revision == newest.c.revision,
                ),
            )
        ):
            latest_by_item[revision.baseline_set_item_id] = revision
    items = []
    for row in rows:
        snapshot = _json(row.result_snapshot_json)
        snapshot = snapshot if isinstance(snapshot, dict) else {}
        set_item = row.baseline_set_item
        try:
            truth = effective_truth_from_latest(set_item, latest_by_item.get(set_item.id))
            current_truth = {
                "revision": truth.revision,
                "source": truth.source,
                "truth_contract_key": truth.truth_contract_key,
                "expected_result": truth.expected_result,
            }
        except BaselineTruthError as exc:
            current_truth = {"error": exc.code}
        items.append({
            "item_id": row.id,
            "baseline_set_item_id": row.baseline_set_item_id,
            "asset_id": row.asset_id,
            "file_name": row.asset.original_name if row.asset is not None else None,
            "truth_contract_key": row.truth_contract_key,
            "status": row.status,
            "error_message": row.error_message or None,
            "prediction": snapshot.get("normalized_result"),
            "raw_output": snapshot.get("stage_a"),
            "output_contract_versions": [
                {"node_id": node_id, "contract_key": result.get("output_contract_key"),
                 "contract_hash": result.get("output_contract_hash")}
                for node_id, result in (snapshot.get("workflow_node_outputs") or {}).items()
                if isinstance(result, dict) and result.get("output_contract_key")
            ],
            "truth_at_run": _json(row.expected_result_json),
            "current_truth": current_truth,
        })
    return {"run": run_summary(run), "offset": offset, "limit": limit, "total": int(total), "items": items}


def cancel_run(db: Session, run_id: int) -> dict[str, Any]:
    from .native_baseline_regression import cancel_native_baseline_runs

    run = _require_run(db, run_id)
    if run.created_by != LAB_ACTOR:
        raise PromptLabError("run_not_owned", "只能取消实验室自己发起的回归", status_code=403)
    if run.status != "running":
        return {**run_summary(run), "cancelled_items": 0}
    affected = cancel_native_baseline_runs(db, baseline_run_ids=[run.id])
    append_audit_event(
        db,
        category="prompt_lab",
        action="run_cancelled",
        subject_type="baseline_regression_run",
        subject_id=run.id,
        actor=LAB_ACTOR,
        payload={"cancelled_items": affected},
        event_key=f"prompt-lab:run:{run.id}:cancel",
    )
    db.flush()
    db.refresh(run)
    return {**run_summary(run), "cancelled_items": affected}


def require_asset(db: Session, asset_id: int) -> Asset:
    """只放出开放类目基准集里的图。"""

    asset = db.get(Asset, asset_id)
    if asset is None:
        raise PromptLabError("asset_not_found", "图片不存在", status_code=404)
    in_allowed_set = db.scalar(
        select(BaselineSetItem.id)
        .join(BaselineSet, BaselineSet.id == BaselineSetItem.baseline_set_id)
        .where(
            BaselineSetItem.asset_id == asset_id,
            BaselineSet.category_key.in_(sorted(allowed_categories(db))),
        )
        .limit(1)
    )
    if in_allowed_set is None:
        raise PromptLabError("asset_not_allowed", "这张图不在实验室开放类目的基准集里", status_code=403)
    return asset

    def lab_client(authorization: str | None = Header(default=None)) -> str:
        configured = settings_provider().prompt_lab_token
        if not configured or len(configured.encode("utf-8")) < MIN_TOKEN_BYTES:
            raise HTTPException(status_code=503, detail="提示词实验室未开通")
        prefix = "Bearer "
        supplied = authorization[len(prefix):] if authorization and authorization.startswith(prefix) else ""
        if not supplied or not hmac.compare_digest(supplied.encode("utf-8"), configured.encode("utf-8")):
            raise HTTPException(status_code=401, detail="提示词实验室认证失败", headers={"WWW-Authenticate": "Bearer"})
        return "prompt-lab"

    def call(db: Session, action: Callable[[], Any], *, write: bool = False) -> Any:
        if not write:
            try:
                return action()
            except PromptLabError as exc:
                raise _http(exc) from exc
        try:
            # 写操作全局串行：锁在本请求第一次读库之前拿，提交之后才放（见 LabWriteLock）。
            with LabWriteLock(db):
                try:
                    result = action()
                    db.commit()
                except BaseException:
                    db.rollback()
                    raise
        except PromptLabError as exc:
            raise _http(exc) from exc
        return result

def writable_categories(user: User) -> list[str] | None:
    role = effective_role(user)
    if role in {"admin", "product_rd"}:
        return None
    if role != "category_manager":
        return []
    try:
        values = json.loads(user.category_keys_json or "[]")
    except (ValueError, TypeError):
        return []
    return sorted(set(item for item in values if isinstance(item, str))) if isinstance(values, list) else []

    if effective_role(user) == "admin":
        return
    if not categories or None in categories or "" in categories:

def lock_access_users(db: Session, user_ids: list[int]) -> None:
    reserve_sqlite_immediate_transaction(db)
    db.execute(select(User.id).where(User.id.in_(sorted(set(user_ids))))

def protect_write_transaction(db: Session, user: User) -> None:
    """Recheck after legacy service commits; serialize with account revocation.

    Revocation locks the same actor row: it either precedes this transaction
    (rejected) or follows its commit. Already approved queued work continues.
    """
    db.info["employee_write_guard"] = (user.id, user.permission_revision)
    _check_write_transaction(db)


def _check_write_transaction(db: Session) -> None:
    identity = db.info.get("employee_write_guard")
    if not identity:
        return
    if db.get_transaction() is not None and db.info.get("employee_guard_transaction") is db.get_transaction():
        return
    user_id, revision = identity
    lock_access_users(db, [user_id])
    row = db.execute(select(User.is_active, User.permission_revision).where(User.id == user_id)).one_or_none()
    if row is None or not row.is_active or row.permission_revision != revision:
        raise HTTPException(403, "权限已变更，请刷新后重试")
    from .employee_access import recheck_employee_objects
    recheck_employee_objects(db, db.get(User, user_id))
    db.info["employee_guard_transaction"] = db.get_transaction()

def _before_employee_commit(db: Session) -> None:
    _check_write_transaction(db)

def check_origin(request: Request) -> None:
    """Cookie writes cannot originate from another site, including login CSRF."""
    if request.method in {"GET", "HEAD", "OPTIONS"}:
        return
    if request.headers.get("sec-fetch-site") == "cross-site":
        raise HTTPException(403, "不接受跨站操作")
    source = request.headers.get("origin") or request.headers.get("referer")
    if source:
        supplied, actual = urlsplit(source), urlsplit(str(request.base_url))
        from .dingtalk_identity import identity_settings
        config = identity_settings()
        canonical = urlsplit(config.redirect_uri)
        # A configured HTTPS public origin also works behind the private HTTP
        # gateway; arbitrary forwarded headers never supply this authority.
        if config.ready and actual.netloc == canonical.netloc:
            actual = canonical
        def origin(value):
            try:
                return value.scheme, value.hostname, value.port or (443 if value.scheme == "https" else 80)
            except ValueError:
                return None
        if origin(supplied) is None or origin(supplied) != origin(actual):
            raise HTTPException(403, "操作来源不匹配")


def _row(db: Session, model, identity):
    if identity is None:
        raise HTTPException(422, "缺少业务对象标识")
    # Pin ownership until this business transaction commits. A concurrent
    # administrator category change must order before or after the write.
    row = db.scalar(select(model).where(model.id == identity).with_for_update()
        .execution_options(populate_existing=True))
    if row is None:
        raise HTTPException(404, "业务对象不存在")
    return row


def _asset_categories(db: Session, ids: list[int]) -> set[str | None]:
    if not ids or len(ids) > 5000 or any(type(i) is not int or i <= 0 for i in ids):
        raise HTTPException(422, "素材范围无效")
    rows = db.execute(select(m.Asset.id, m.Asset.category_key).where(m.Asset.id.in_(ids))
        .order_by(m.Asset.id).with_for_update()).all()
    if len(rows) != len(set(ids)):
        raise HTTPException(404, "包含不存在的素材")
    return {r.category_key for r in rows}


def _evaluation(db: Session, identity) -> set[str | None]:
    row = _row(db, m.EvaluationResult, identity)
    categories = {_row(db, m.Asset, row.asset_id).category_key}
    if row.production_run_id:
        categories.add(_row(db, m.ProductionRun, row.production_run_id).category_key)
    return categories


def object_categories(db: Session, resolver: str, params: dict, body: dict) -> set[str | None]:
    params = {key: int(value) if key.endswith("_id") and str(value).isdigit() else value for key, value in params.items()}
    if resolver in {"cover_run", "cover_item"}:
        from .cover_regression import CoverRegressionRun, CoverRegressionItem
        run = _row(db, CoverRegressionRun, str(params.get("run_id")))
        if resolver == "cover_item":
            item = _row(db, CoverRegressionItem, str(params.get("item_id")))
            if item.run_id != run.id:
                raise HTTPException(404, "样本不属于本轮")
            return {run.category_key} | _asset_categories(db, [item.asset_id])
        return {run.category_key}
    if resolver == "evaluation":
        return _evaluation(db, params.get("evaluation_id"))
    if resolver == "evaluation_batch":
        items = body.get("items")
        if not isinstance(items, list) or not 1 <= len(items) <= 200:
            raise HTTPException(422, "批量对象范围无效")
        result: set[str | None] = set()
        for item in items:
            if not isinstance(item, dict) or type(item.get("evaluation_id")) is not int:
                raise HTTPException(422, "批量对象无效")
        for item in sorted(items, key=lambda item: item["evaluation_id"]):
            result |= _evaluation(db, item.get("evaluation_id"))
        return result
    if resolver in {"baseline_item", "baseline_set_item"}:
        model = m.BaselineRegressionItem if resolver == "baseline_item" else m.BaselineSetItem
        row = _row(db, model, params.get("item_id"))
        parent_key = "run_id" if resolver == "baseline_item" else "baseline_set_id"
        if getattr(row, parent_key) != params.get(parent_key):
            raise HTTPException(404, "条目不属于指定集合")
        parent = _row(db, m.BaselineRegressionRun if resolver == "baseline_item" else m.BaselineSet, getattr(row, parent_key))
        return {parent.category_key} | _asset_categories(db, [row.asset_id])
    if resolver in {"baseline_set", "baseline_run"}:
        parent = _row(db, m.BaselineSet if resolver == "baseline_set" else m.BaselineRegressionRun,
                      params.get("baseline_set_id" if resolver == "baseline_set" else "run_id"))
        model, key = (m.BaselineSetItem, "baseline_set_id") if resolver == "baseline_set" else (m.BaselineRegressionItem, "run_id")
        categories = set(db.scalars(select(m.Asset.category_key).join(model, model.asset_id == m.Asset.id)
            .where(getattr(model, key) == parent.id).order_by(m.Asset.id).with_for_update()).all())
        return categories | {parent.category_key}
    if resolver == "baseline_correction":
        return {_row(db, m.BaselineCorrectionRun, params.get("correction_id")).category_key}
    if resolver == "sample_set":
        row = _row(db, m.SampleSet, body.get("sample_set_id"))
        return {row.category_key} | set(db.scalars(select(m.Asset.category_key).join(m.SampleSetItem)
            .where(m.SampleSetItem.sample_set_id == row.id).order_by(m.Asset.id).with_for_update()).all())
    if resolver in {"prompt_item", "paired_item", "migration_item"}:
        row = _row(db, m.MigrationItem if resolver == "migration_item" else m.PromptRegressionItem, params.get("item_id"))
        if row.run_id != params.get("run_id"):
            raise HTTPException(404, "条目不属于指定运行")
        if resolver == "migration_item":
            return _asset_categories(db, [row.asset_id])
        sample = _row(db, m.SampleSetItem, row.sample_item_id)
        categories = _asset_categories(db, [sample.asset_id]) | {_row(db, m.SampleSet, sample.sample_set_id).category_key}
        if resolver == "paired_item":
            categories |= _evaluation(db, body.get("baseline_evaluation_id")) | _evaluation(db, body.get("candidate_evaluation_id"))
        return categories
    if resolver in {"runtime_id", "runtime_key"}:
        row = (_row(db, m.ProductionRun, params.get("run_id")) if resolver == "runtime_id" else
               db.scalar(select(m.ProductionRun).where(m.ProductionRun.run_key == params.get("run_key"))
                   .with_for_update().execution_options(populate_existing=True)))
        if row is None:
            raise HTTPException(404, "运行不存在")
        return {row.category_key}
    if resolver == "material_job":
        from .incremental_material_models import IncrementalMaterialJob
        return {_row(db, IncrementalMaterialJob, params.get("job_id")).category_key}
    if resolver == "assets":
        return _asset_categories(db, body.get("asset_ids", []))
    if resolver == "package":
        row = _row(db, m.MaterialPackage, body.get("material_package_id"))
        return {row.category_key} | set(db.scalars(select(m.Asset.category_key).join(m.MaterialPackageItem)
            .where(m.MaterialPackageItem.package_id == row.id).order_by(m.Asset.id).with_for_update()).all())
    if resolver in {"production_run", "production_evaluation"}:
        row = _row(db, m.EvaluationProductionRun, params.get("run_id"))
        categories = {row.category_key}
        if resolver == "production_evaluation":
            categories |= _evaluation(db, params.get("evaluation_id"))
        return categories
    raise HTTPException(403, "该操作仅系统管理员可执行")


def authorize_employee_write(request: Request, db: Session, user: m.User, body: dict[str, Any]) -> None:
    if request.method in {"GET", "HEAD", "OPTIONS"}:
        return
    check_origin(request)
    if effective_role(user) == "admin":
        return
    path = getattr(request.scope.get("route"), "path", "").rstrip("/")
    policy = WRITE_POLICIES.get((request.method, path))
    if policy is None or not has_permission(user, policy[0]):
        raise HTTPException(403, "当前角色不能执行此操作")
    resolver = policy[1]
    if resolver in {"self", "read"}:
        return
    if resolver == "draft":
        if body.get("status", "draft") not in {"draft", "candidate"} or body.get("active") is True:
            raise HTTPException(403, "产研只能编辑草稿，正式启用由系统管理员执行")
        return
    categories = object_categories(db, resolver, request.path_params, body)
    require_categories(user, categories)
    # A supplied category is never authoritative, but must also agree with the
    # source objects so it cannot redirect a write into another category.
    if body.get("category_key") and body["category_key"] not in categories:
        raise HTTPException(403, "请求类目与实际对象不一致")
    db.info["employee_object_guard"] = (resolver, dict(request.path_params), body)


def recheck_employee_objects(db: Session, user: m.User) -> None:
    guard = db.info.get("employee_object_guard")
    if guard:
        resolver, params, body = guard
        categories = object_categories(db, resolver, params, body)
        require_categories(user, categories)
        if body.get("category_key") and body["category_key"] not in categories:
            raise HTTPException(403, "对象类目已变化，请刷新后重试")

_BINDING_CHANGED = "员工绑定刚刚变更，请重试"


def aware(value: datetime) -> datetime:
    return value if value.tzinfo else value.replace(tzinfo=timezone.utc)


def verified_identity(db: Session, employee: VerifiedEmployee) -> EmployeeIdentity:
    if employee.corp_id != identity_settings().corp_id:
        raise HTTPException(403, "仅限本公司员工登录")
    row = db.scalar(select(EmployeeIdentity).where(EmployeeIdentity.corp_id == employee.corp_id,
        EmployeeIdentity.employee_id == employee.employee_id))
    if row is None:
        row = EmployeeIdentity(corp_id=employee.corp_id, employee_id=employee.employee_id,
            union_id=employee.union_id, display_name=employee.display_name, status="active")
        db.add(row)
    else:
        expected_user_id = row.user_id
        if expected_user_id:
            lock_access_users(db, [expected_user_id])
        row = db.scalar(select(EmployeeIdentity).where(EmployeeIdentity.id == row.id)
                        .with_for_update().execution_options(populate_existing=True))
        if row.user_id != expected_user_id:
            raise HTTPException(409, _BINDING_CHANGED)
        if row.union_id != employee.union_id:
            raise HTTPException(409, "员工身份绑定冲突，请联系系统管理员")
        if row.status == "removed":
            # Successful re-login/reconciliation never revives a departed account.
            raise HTTPException(403, "账号已停用，重新入职请联系系统管理员开通")
    row.display_name = employee.display_name
    row.verified_at = datetime.now(timezone.utc)
    db.flush()
    return row


def employee_login_user(db: Session, employee: VerifiedEmployee) -> User:
    """First verified company login creates only a reader, in the login transaction.

    Existing bindings/disabled accounts are authoritative. Never match accounts
    by name, phone or username. A competing first login may win the unique
    identity insert; retry once in a fresh transaction and use its binding.
    """
    for attempt in range(2):
        try:
            identity = verified_identity(db, employee)
            if identity.status != "active":
                raise HTTPException(403, "账号已停用，请联系系统管理员")
            if identity.user_id is not None:
                user = db.get(User, identity.user_id)
                if user is None or not user.is_active:
                    raise HTTPException(403, "账号已停用，请联系系统管理员")
                return user
            # Stable, opaque account name; a collision fails closed instead of
            # attaching this identity to an unrelated pre-existing account.
            key = hashlib.sha256((employee.corp_id + "\0" + employee.employee_id).encode()).hexdigest()[:32]
            user = User(username="dd_" + key, display_name=employee.display_name[:80],
                password_hash="!uninitialized", password_initialized=False,
                is_active=True, is_admin=False, role="viewer", permissions_json="[]",
                category_keys_json="[]", permission_revision=1)
            db.add(user)
            db.flush()
            identity.user_id = user.id
            append_audit_event(db, category="auth", action="employee_self_registered",
                subject_type="user", subject_id=str(user.id), actor="dingtalk",
                event_key=f"employee-self-register:{identity.id}",
                payload={"employee_identity_id": identity.id, "reason": "公司员工首次钉钉登录",
                    "after": {"role": "viewer", "is_admin": False, "is_active": True,
                              "password_initialized": False, "category_keys": []}})
            return user
        except IntegrityError:
            db.rollback()
            if attempt:
                raise HTTPException(409, "员工身份或账号冲突，请联系系统管理员") from None
        except HTTPException as exc:
            if exc.status_code != 409 or exc.detail != _BINDING_CHANGED:
                raise
            db.rollback()
            if attempt:
                raise
    raise AssertionError("bounded registration retry exhausted")


def check_employee_access(db: Session, user: User) -> None:
    config = identity_settings()
    row = db.scalar(select(EmployeeIdentity).where(EmployeeIdentity.user_id == user.id))
    if row is None:
        if config.required:
            raise HTTPException(403, "账号尚未绑定公司钉钉身份，请联系系统管理员")
        return  # Explicit migration period only; cutover requires two bound admins.
    if row.status != "active":
        raise HTTPException(401, "员工账号已停用")
    if not config.corp_id or row.corp_id != config.corp_id:
        raise HTTPException(503, "公司身份配置不匹配，请联系系统管理员")
    if row.verified_at and (datetime.now(timezone.utc) - aware(row.verified_at)).total_seconds() < config.freshness_seconds:
        return
    try:
        employee = identity_client().employee(row.employee_id)
    except (IdentityUnavailable, NotCompanyEmployee):
        raise HTTPException(503, "公司身份验证暂不可用，请稍后重试") from None
    verified_identity(db, employee)
    # Persist freshness before the request starts its protected write transaction.
    db.commit()


def process_identity_event(db: Session, *, event_id: str, corp_id: str, event_type: str,
                           occurred_at: datetime, employee_ids: list[str]) -> bool:
    if corp_id != identity_settings().corp_id or not event_id or len(event_id) > 160:
        raise ValueError("identity event envelope invalid")
    if event_type not in {"user_leave_org", "user_add_org", "user_modify_org"}:
        return False
    if not employee_ids or len(employee_ids) > 1000 or any(not isinstance(v, str) or not v or len(v) > 100 for v in employee_ids):
        raise ValueError("identity event members invalid")
    bindings = dict(db.execute(select(EmployeeIdentity.id, EmployeeIdentity.user_id).where(EmployeeIdentity.corp_id == corp_id,
        EmployeeIdentity.employee_id.in_(employee_ids))).all())
    # Same global ordering as user management; no actor/target deadlock cycle.
    lock_access_users(db, [user_id for user_id in bindings.values() if user_id])
    if db.get(IdentityEvent, event_id):
        return False
    rows = db.scalars(select(EmployeeIdentity).where(EmployeeIdentity.corp_id == corp_id,
        EmployeeIdentity.employee_id.in_(employee_ids)).order_by(EmployeeIdentity.id).with_for_update()
        .execution_options(populate_existing=True)).all()
    if any(row.id not in bindings or row.user_id != bindings[row.id] for row in rows):
        # A concurrent binding won before identity-row locking. Retry the event
        # with the complete actor set, never take locks in the reverse order.
        raise HTTPException(409, "员工绑定刚刚变更，请重试")
    for row in rows:
        if event_type == "user_leave_org" and row.reopened_at and aware(row.reopened_at) >= aware(occurred_at):
            continue
        if event_type != "user_leave_org" and row.event_at and aware(row.event_at) > aware(occurred_at):
            continue
        # Equal timestamps: removal wins. No event grants or restores access.
        if row.event_at is None or aware(occurred_at) > aware(row.event_at):
            row.event_at = occurred_at
        row.verified_at = None
        if event_type == "user_leave_org":
            row.status = "removed"
            if row.user_id:
                user = db.get(User, row.user_id)
                before = {"is_active": user.is_active, "permission_revision": user.permission_revision}
                user.is_active = False
                user.permission_revision += 1
                db.execute(delete(SessionToken).where(SessionToken.user_id == user.id))
                append_audit_event(db, category="auth", action="employee_removed", subject_type="user",
                    subject_id=str(user.id), actor="dingtalk", event_key=f"identity:{event_id}:{user.id}",
                    payload={"before": before, "after": {"is_active": False, "permission_revision": user.permission_revision},
                             "reason": "员工离职或移出组织", "source_event_id": event_id})
    db.add(IdentityEvent(event_id=event_id, corp_id=corp_id, event_type=event_type, occurred_at=occurred_at))
    db.flush()
    return True

import { useState } from "react"
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query"
import { toast } from "sonner"

import { api } from "@/lib/api"

type Status = { enabled: boolean; path: string; last_used_at: string | null; last_tool: string | null }
export function PersonalMcpSettings({ userId }: { userId: number }) {
  const client = useQueryClient()
  const [token, setToken] = useState("")
  const [confirmReset, setConfirmReset] = useState(false)
  const status = useQuery({ queryKey: ["personal-mcp", userId], queryFn: () => api<Status>("/api/me/mcp") })
  const url = `${window.location.origin}/api/mcp`
  const config = JSON.stringify({ mcpServers: { "label-platform": { url, headers: { Authorization: `Bearer ${token || "<个人令牌>"}` } } } }, null, 2)
  const rotate = useMutation({ mutationFn: () => api<{ token: string }>("/api/me/mcp/rotate", { method: "POST" }),
    onSuccess: data => { setToken(data.token); setConfirmReset(false); client.invalidateQueries({ queryKey: ["personal-mcp"] }); toast.success("个人MCP已开通，请保存本次配置") },
    onError: error => toast.error(error.message) })
  const disable = useMutation({ mutationFn: () => api("/api/me/mcp", { method: "DELETE" }),
    onSuccess: () => { setToken(""); setConfirmReset(false); client.invalidateQueries({ queryKey: ["personal-mcp"] }); toast.success("MCP已停用，旧令牌已失效") }, onError: error => toast.error(error.message) })
  const download = () => {
    const blob = new Blob([config], { type: "application/json" })
    const objectUrl = URL.createObjectURL(blob)
    const a = document.createElement("a"); a.href = objectUrl; a.download = "label-platform-mcp.json"; a.click()
    setTimeout(() => URL.revokeObjectURL(objectUrl), 1000)
  }
  return <section className="employee-settings-section rounded-lg border border-line bg-white px-card py-card">
    <h2 className="visual-card-title">MCP 客户端接入</h2>
    <p className="mt-2 text-sm text-muted-foreground">服务由中台统一运行，无需在电脑上部署服务。客户端使用你的权限访问存量工具，不能操作增量开关、生产绑定或业务投递。</p>
    {status.isLoading ? <p className="mt-5">正在读取接入状态…</p> : status.isError ? <p role="alert" className="mt-5 text-danger">{status.error.message}</p> : <>
      <dl className="mt-5 grid gap-3 text-sm"><div><dt className="font-bold">状态</dt><dd>{status.data?.enabled ? "已开通" : "未开通"}</dd></div><div><dt className="font-bold">连接地址</dt><dd className="break-all">{url}</dd></div><div><dt className="font-bold">最近成功调用</dt><dd>{status.data?.last_used_at ? `${new Date(status.data.last_used_at).toLocaleString()} · ${status.data.last_tool}` : "暂无记录"}</dd></div></dl>
      <div className="mt-5 flex flex-wrap gap-3">
        {!status.data?.enabled ? <Button disabled={rotate.isPending} onClick={() => rotate.mutate()}>开通并生成配置</Button> : <><Button variant="secondary" disabled={rotate.isPending || disable.isPending} onClick={() => setConfirmReset(true)}>重置个人令牌</Button><Button variant="secondary" disabled={disable.isPending || rotate.isPending} onClick={() => disable.mutate()}>停用 MCP</Button></>}
      </div>
      {confirmReset && <div className="mt-5 rounded-lg border border-line p-4"><p>重置后，现有客户端配置立即失效，需要使用新配置重新连接。</p><div className="mt-3 flex gap-3"><Button disabled={rotate.isPending} onClick={() => rotate.mutate()}>确认重置</Button><Button variant="secondary" onClick={() => setConfirmReset(false)}>取消</Button></div></div>}
