def _cover_rows(db,run_id):
    from .cover_regression import CoverRegressionRun as Run,CoverRegressionItem as Item,CoverRegressionReview as Review
    from .cover_attempts import CoverGenerationAttempt as Attempt
    latest=(select(Attempt.sample_key,func.max(Attempt.sequence_no).label('seq')).join(Item,Item.id==Attempt.sample_key)
            .where(Item.run_id==run_id).group_by(Attempt.sample_key).subquery())
    success=(select(Attempt.sample_key,func.max(Attempt.sequence_no).label('seq')).join(Item,Item.id==Attempt.sample_key)
             .where(Item.run_id==run_id,Attempt.state=='succeeded').group_by(Attempt.sample_key).subquery())
    revisions=(select(Review.item_id,func.max(Review.revision).label('rev')).join(Item,Item.id==Review.item_id)
               .where(Item.run_id==run_id).group_by(Review.item_id).subquery())
    current=aliased(Attempt);raw=aliased(Attempt);human_image=aliased(Attempt)
    q=(select(Item,Run,current,raw,Review,human_image).join(Run,Run.id==Item.run_id)
       .outerjoin(latest,latest.c.sample_key==Item.id).outerjoin(current,and_(current.sample_key==Item.id,current.sequence_no==latest.c.seq))
       .outerjoin(success,success.c.sample_key==Item.id).outerjoin(raw,and_(raw.sample_key==Item.id,raw.sequence_no==success.c.seq))
       .outerjoin(revisions,revisions.c.item_id==Item.id).outerjoin(Review,and_(Review.item_id==Item.id,Review.revision==revisions.c.rev))
       .outerjoin(human_image,human_image.id==Review.attempt_id).where(Run.id==run_id).order_by(Item.id))

        def result(a):return {'attempt_id':a.id,**json.loads(a.result_json)} if a and a.result_json else None

def freeze_results(db:Session,*,source_kind:str,run_id:str,selection:dict,result_mode:str,actor:str):
    if result_mode not in {'effective','raw'}:raise ExportError('invalid_result_mode')
    if set(selection)-{'status','reviewed','q'}:raise ExportError('unknown_filter')
    if not isinstance(selection.get('q',''),str) or len(selection.get('q',''))>200:raise ExportError('invalid_filter')
    if 'reviewed' in selection and type(selection['reviewed']) is not bool:raise ExportError('invalid_filter')
    if selection.get('status') is not None and not isinstance(selection['status'],str):raise ExportError('invalid_filter')
    if source_kind not in {'cover','baseline'}:raise ExportError('unsupported_source')
    from .cover_regression import CoverRegressionRun
    from .models import BaselineRegressionRun
    try:run=db.get(CoverRegressionRun if source_kind=='cover' else BaselineRegressionRun,run_id if source_kind=='cover' else int(run_id))
    except ValueError:raise ExportError('invalid_run_id')
    if run is None:raise ExportError('run_not_found')
    config=({'baseline_set_id':run.baseline_set_id,'generation':json.loads(run.config_json)} if source_kind=='cover' else
            {'baseline_set_id':run.baseline_set_id,'workflow_version_id':run.workflow_version_id,'execution_engine':run.execution_engine,
             'execution_hash':hashlib.sha256(run.execution_snapshot_json.encode()).hexdigest()})
    snap=StockExportSnapshot(id=uuid.uuid4().hex,source_kind=source_kind,run_id=str(run_id),category_key=run.category_key,
        selection_json=canonical(selection),result_mode=result_mode,config_json=canonical(config),digest='',total=0,created_by=actor,created_at=now())
    db.add(snap);db.flush();digest=hashlib.sha256()
    rows=_cover_rows(db,run_id) if source_kind=='cover' else _baseline_rows(db,run_id)
    for _,doc in rows:
        if selection.get('status') and doc['status']!=selection['status']:continue
        if 'reviewed' in selection and (doc['human'] is not None)!=selection['reviewed']:continue
        query=selection.get('q','').casefold()
        if query and query not in (doc['name']+' '+doc['item_id']+' '+str(doc['asset_id'])).casefold():continue
        if result_mode=='raw':doc['effective']=doc['raw']
        document=canonical(doc);digest.update(document.encode());digest.update(b'\n');snap.total+=1
        db.add(StockExportItem(snapshot_id=snap.id,ordinal=snap.total,document_json=document))
        if snap.total%100==0:db.flush()
    if snap.total==0:raise ExportError('no_matching_results')
    snap.digest=digest.hexdigest();db.flush();return snap

def create_export(db,*,snapshot_id,format,actor):
    snap=db.get(StockExportSnapshot,snapshot_id)
    if snap is None:raise ExportError('snapshot_not_found')
    if format not in {'json','xlsx','images_zip'}:raise ExportError('unsupported_format')
    if format=='images_zip' and snap.source_kind!='cover':raise ExportError('images_require_generation')
    job=StockExportJob(id=uuid.uuid4().hex,snapshot_id=snapshot_id,format=format,status='queued',created_by=actor,created_at=now())
    db.add(job);db.flush();return job

def documents(db,snapshot_id):
    cursor=0
    while True:
        rows=db.scalars(select(StockExportItem).where(StockExportItem.snapshot_id==snapshot_id,StockExportItem.ordinal>cursor)
            .order_by(StockExportItem.ordinal).limit(100)).all()
        if not rows:return
        for row in rows:
            cursor=row.ordinal;yield json.loads(row.document_json)

def json_file(file,db,snap):

    file.write((canonical({'snapshot':header})[:-1]+',"items":[').encode())
    for i,doc in enumerate(documents(db,snap.id)):

def execute_export(db,job_id,*,settings):
    """Caller holds the export scheduler lock; restart reuses frozen rows only."""
    job=db.get(StockExportJob,job_id)
    if not job or job.status not in {'queued','running'}:return job
    snap=db.get(StockExportSnapshot,job.snapshot_id)
    job.status='running';job.error_code=None;db.commit()
    root=Path(settings.data_dir)/'stock-exports';root.mkdir(parents=True,exist_ok=True)
    ext={'json':'json','xlsx':'xlsx','images_zip':'zip'}[job.format]
    name=f'{job.id}.{ext}';target=root/name;temporary=root/(name+'.part')
    missing=0
    try:
        import shutil
        required=(db.scalar(select(func.sum(func.length(StockExportItem.document_json))).where(StockExportItem.snapshot_id==snap.id)) or 0)*4
        if job.format=='images_zip':
            for doc in documents(db,snap.id):
                digest=(doc['effective'] or {}).get('sha256','')
                if re.fullmatch('[0-9a-f]{64}',digest):
                    path=Path(settings.data_dir)/'stock-images'/'outputs'/(digest+'.png')
                    if path.is_file():required+=path.stat().st_size
        if shutil.disk_usage(root).free < required+512*1024*1024:raise ExportError('insufficient_export_space')
        if job.format=='json':
            with temporary.open('wb') as file:json_file(file,db,snap)
        elif job.format=='xlsx':
            from .stock_export_xlsx import write_xlsx
            write_xlsx(temporary,lambda:documents(db,snap.id),metadata=snapshot_payload(snap))
        else:
            with zipfile.ZipFile(temporary,'w',compression=zipfile.ZIP_DEFLATED,allowZip64=True) as z:
                with z.open('results.json','w',force_zip64=True) as file:json_file(file,db,snap)
                import tempfile,shutil
                with tempfile.SpooledTemporaryFile(max_size=2*1024*1024) as file:
                    file.write(b'[')
                    for i,doc in enumerate(documents(db,snap.id)):
                        record={'item_id':doc['item_id'],'file':None,'error':None}
                        result=doc['effective'] or {};digest=result.get('sha256','')
                        path=Path(settings.data_dir)/'stock-images'/'outputs'/(digest+'.png')
                        if not re.fullmatch('[0-9a-f]{64}',digest) or not path.is_file():record['error']='image_missing';missing+=1
                        elif _file_hash(path)!=digest:record['error']='image_checksum_changed';missing+=1
                        else:
                            record['file']=f'images/{i+1}-{digest}.png';z.write(path,record['file'])
                        if i:file.write(b',')
                        file.write(canonical(record).encode())
                    file.write(b']')
                    file.seek(0)
                    with z.open('files.json','w',force_zip64=True) as destination:shutil.copyfileobj(file,destination)
        with temporary.open('r+b') as file:os.fsync(file.fileno());checksum=hashlib.file_digest(file,'sha256').hexdigest()
        temporary.replace(target)
        job.file_name=name;job.file_bytes=target.stat().st_size;job.file_sha256=checksum;job.missing_files=missing
        job.status='succeeded';job.expires_at=(datetime.now(timezone.utc)+timedelta(days=30)).isoformat()
    except Exception as exc:
        job.status='failed';job.error_code=str(exc) if isinstance(exc,ExportError) else 'export_failed'
        import logging
        logging.getLogger(__name__).exception('Stock export failed: %s',job.id)
    finally:temporary.unlink(missing_ok=True)
    db.commit();return job

def _file_hash(path):
    with path.open('rb') as file:return hashlib.file_digest(file,'sha256').hexdigest()

def expire_packages(db,*,settings):
    count=0;root=(Path(settings.data_dir)/'stock-exports').resolve()
    for job in db.scalars(select(StockExportJob).where(StockExportJob.status=='succeeded',StockExportJob.expires_at<=now())):

def job_payload(db,job):
    snap=db.get(StockExportSnapshot,job.snapshot_id)
    available=job.status=='succeeded' and bool(job.expires_at and job.expires_at>now())
    return {'id':job.id,'snapshot_id':job.snapshot_id,'format':job.format,'status':'expired' if job.status=='succeeded' and not available else job.status,
        'total':snap.total,'result_mode':snap.result_mode,'selection':json.loads(snap.selection_json),
        'frozen_at':snap.created_at,'digest':snap.digest,'file_bytes':job.file_bytes,'missing_files':job.missing_files,
        'error_code':job.error_code,'created_at':job.created_at,'expires_at':job.expires_at,
        'download_url':f'/api/stock-exports/{job.id}/file' if available else None}

def build_stock_export_router(current_user):
    router=APIRouter()
    def permission(user):
        if not has_permission(user,'results:export'):raise HTTPException(403,'没有导出权限')
    def enabled():
        if os.environ.get('STOCK_EXPORT_ENABLED')!='1':raise HTTPException(503,'通用导出后台尚未启用')
    def job_row(db,id):
        row=db.get(StockExportJob,id)
        if row is None:raise HTTPException(404,'导出任务不存在')
        return row

    @router.get('/api/stock-exports/capabilities')
    def capabilities(user=Depends(current_user)):
        return {'enabled':os.environ.get('STOCK_EXPORT_ENABLED')=='1','formats':['xlsx','json','images_zip'],'retention_days':30}

    @router.post('/api/stock-exports')
    def launch(payload:ExportRequest,user=Depends(current_user),db:Session=Depends(get_db)):
        permission(user);enabled()
        try:
            snap=freeze_results(db,source_kind=payload.source_kind,run_id=payload.run_id,selection=payload.selection,
                result_mode=payload.result_mode,actor=user.username)
            job=create_export(db,snapshot_id=snap.id,format=payload.format,actor=user.username)
            from .audit import append_audit_event
            append_audit_event(db,category='baseline',action='stock_export_requested',subject_type='stock_export_snapshot',
                subject_id=snap.id,actor=user.username,payload={'source_kind':snap.source_kind,'run_id':snap.run_id,'total':snap.total,'digest':snap.digest,'format':job.format})
            db.commit();return job_payload(db,job)
        except ExportError as exc:
            db.rollback();raise HTTPException(422,str(exc)) from exc

    @router.get('/api/stock-exports')
    def listing(source_kind:Literal['cover','baseline'],run_id:str,user=Depends(current_user),db:Session=Depends(get_db)):
        rows=db.scalars(select(StockExportJob).join(StockExportSnapshot).where(StockExportSnapshot.source_kind==source_kind,
            StockExportSnapshot.run_id==run_id).order_by(StockExportJob.created_at.desc()).limit(30)).all()
        return {'items':[job_payload(db,row) for row in rows]}

    @router.get('/api/stock-exports/{job_id}')
    def detail(job_id:str,user=Depends(current_user),db:Session=Depends(get_db)):
        return job_payload(db,job_row(db,job_id))

    @router.post('/api/stock-exports/{job_id}/rebuild')
    def rebuild(job_id:str,payload:RebuildRequest,user=Depends(current_user),db:Session=Depends(get_db)):
        permission(user);enabled()
        try:
            previous=job_row(db,job_id)
            job=create_export(db,snapshot_id=previous.snapshot_id,format=payload.format,actor=user.username)
            db.commit();return job_payload(db,job)
        except ExportError as exc:
            db.rollback();raise HTTPException(422,str(exc)) from exc

    @router.get('/api/stock-exports/{job_id}/file')
    def download(job_id:str,user=Depends(current_user),db:Session=Depends(get_db)):
        job=job_row(db,job_id)
        if job.status=='expired' or (job.expires_at and job.expires_at<=now()):raise HTTPException(410,'下载包已过期，可按原版本重建')
        if job.status!='succeeded' or not job.file_name:raise HTTPException(409,'下载包尚未生成')
        if not re.fullmatch(r'[0-9a-f]{32}\.(json|xlsx|zip)',job.file_name):raise HTTPException(409,'文件记录异常')
        path=Path(get_settings().data_dir)/'stock-exports'/job.file_name
        if not path.is_file():raise HTTPException(410,'下载包不存在，可按原版本重建')
        media={'json':'application/json','xlsx':'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet','images_zip':'application/zip'}[job.format]
        return FileResponse(path,filename=f'{job.snapshot_id}.{path.suffix.lstrip(".")}',media_type=media,headers={'Cache-Control':'private, no-store'})
    return router

@contextmanager
def export_lock(engine,root):
    if engine.dialect.name=='mysql':
        lock_engine=create_engine(engine.url,poolclass=NullPool)
        try:
            with lock_engine.connect() as c:
                acquired=c.execute(text("SELECT GET_LOCK('stock-export-worker',0)")).scalar()==1
                try:yield acquired
                finally:
                    if acquired:c.execute(text("SELECT RELEASE_LOCK('stock-export-worker')"))
        finally:lock_engine.dispose()
    elif engine.dialect.name=='sqlite':
        # The existing SQLite helper locks its supplied directory. Its MySQL
        # branch is deliberately not reused, so long exports never block GPU work.
        from .cover_worker import scheduler_lock
        with scheduler_lock(engine,root/'export-worker') as acquired:yield acquired
    else:raise RuntimeError('unverified export locking backend')

def tick(engine,*,settings):
    with export_lock(engine,Path(settings.data_dir)) as acquired:
        if not acquired:return False
        with Session(engine) as db:
            expire_packages(db,settings=settings)
            job=db.scalar(select(StockExportJob).where(StockExportJob.status.in_(['queued','running']))
                .order_by(StockExportJob.created_at,StockExportJob.id).limit(1))
            if job is None:return False
            execute_export(db,job.id,settings=settings);return True

if __name__=='__main__':

"""Streaming OOXML export. Large JSON is split losslessly into numbered cells."""
import json,re,zipfile
from xml.sax.saxutils import escape

NS='http://schemas.openxmlformats.org/spreadsheetml/2006/main'
BAD=re.compile(r'[\x00-\x08\x0b\x0c\x0e-\x1f]')

def chunks(text,limit=30000):
    # Excel counts UTF-16 units, not Python Unicode code points.
    current=[];size=0
    for ch in text:
        n=2 if ord(ch)>0xffff else 1
        if size+n>limit:
            yield ''.join(current);current=[];size=0
        current.append(ch);size+=n
    if current or not text:yield ''.join(current)

def cell(value):
    # Explicit inline strings never execute formulas, including chunk boundaries
    # starting with '='. Prefixing a quote would corrupt lossless JSON reassembly.
    value=BAD.sub('',str(value))
    return '<c t="inlineStr"><is><t xml:space="preserve">'+escape(value)+'</t></is></c>'

def write_xlsx(path,documents,metadata=None):
    """documents is a repeatable iterable factory; only one row stays in memory."""
    fields=set()
    for d in documents():
        if isinstance(d['effective'],dict):fields.update(d['effective'])
    fields=sorted(fields)
    expanded=len(fields)<=16380
    names=[]
    with zipfile.ZipFile(path,'w',compression=zipfile.ZIP_DEFLATED,allowZip64=True) as z:
        def sheets(title,headers,rows):
            stream=None;number=0;part=0
            try:
                for row in rows:
                    if stream is None or number>=1_000_000:
                        if stream:stream.write(b'</sheetData></worksheet>');stream.close()
                        part+=1;names.append(title if part==1 else title+str(part))
                        stream=z.open(f'xl/worksheets/sheet{len(names)}.xml','w',force_zip64=True)
                        stream.write((f'<?xml version="1.0" encoding="UTF-8"?><worksheet xmlns="{NS}"><sheetViews><sheetView workbookViewId="0"><pane ySplit="1" topLeftCell="A2" state="frozen"/></sheetView></sheetViews><sheetData><row>'+''.join(cell(x) for x in headers)+'</row>').encode())
                        number=1
                    stream.write(('<row>'+''.join(cell(x) for x in row)+'</row>').encode('utf-8'));number+=1
            finally:
                if stream:stream.write(b'</sheetData></worksheet>');stream.close()
        def summary():
            for d in documents():
                def preview(value):
                    value=value if isinstance(value,str) else json.dumps(value,ensure_ascii=False,separators=(',',':'))
                    return value if len(value.encode('utf-16-le'))<=60000 else '完整内容见“完整字段”表'
                values=[(d['effective'] or {}).get(k) for k in fields] if expanded else [d['effective']]
                yield (d['item_id'],d['name'],d['status'],d['human'] is not None,*(preview(v) for v in values))
        def detail():
            for d in documents():
                # ensure_ascii also preserves XML-disallowed control characters.
                parts=list(chunks(json.dumps(d,ensure_ascii=True,separators=(',',':'))))
                for i,p in enumerate(parts):yield (d['item_id'],i+1,len(parts),p)
        sheets('结果',['样本ID','素材名称','执行状态','有人工修订',*(['结果.'+k for k in fields] if expanded else ['当前结果（完整字段见明细）'])],summary())
        sheets('完整字段',['样本ID','片段序号','片段总数','完整JSON片段（按序拼接）'],detail())
        if metadata:
            sheets('导出版本',['项目','内容'],((k,json.dumps(v,ensure_ascii=True)) for k,v in metadata.items()))
        if not names:raise ValueError('empty workbook')
        z.writestr('[Content_Types].xml','<?xml version="1.0"?><Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types"><Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/><Default Extension="xml" ContentType="application/xml"/><Override PartName="/xl/workbook.xml" ContentType="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet.main+xml"/>'+''.join(f'<Override PartName="/xl/worksheets/sheet{i}.xml" ContentType="application/vnd.openxmlformats-officedocument.spreadsheetml.worksheet+xml"/>' for i in range(1,len(names)+1))+'</Types>')
        z.writestr('_rels/.rels','<?xml version="1.0"?><Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships"><Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/officeDocument" Target="xl/workbook.xml"/></Relationships>')
        z.writestr('xl/workbook.xml',f'<?xml version="1.0"?><workbook xmlns="{NS}" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships"><sheets>'+''.join(f'<sheet name="{n}" sheetId="{i}" r:id="rId{i}"/>' for i,n in enumerate(names,1))+'</sheets></workbook>')
        z.writestr('xl/_rels/workbook.xml.rels','<?xml version="1.0"?><Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">'+''.join(f'<Relationship Id="rId{i}" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/worksheet" Target="worksheets/sheet{i}.xml"/>' for i in range(1,len(names)+1))+'</Relationships>')

def test_real_cookie_export_is_read_access_and_remains_stock_only(context,monkeypatch):

        asset=db.scalar(select(Asset).where(Asset.category_key=='model_3d_su'))
        b=BaselineSet(name='export-api',category_key=asset.category_key,default_expected_level='L1',fingerprint='a'*64)
