===== backend/app/workflow_inventory.py =====
"""Versioned descriptive inventory. This module never operates a workflow or host.

Lock order: catalog -> draft -> affected records. Autosave locks only its draft.
Drafts are private; a published record and its history are readable by employees.
"""
from __future__ import annotations

import hashlib
import json
import uuid
from datetime import datetime, timezone
from typing import Literal

from pydantic import BaseModel, ConfigDict, Field, ValidationError
from sqlalchemy import ForeignKey, Integer, String, Text, UniqueConstraint, select
from sqlalchemy.orm import Mapped, mapped_column

from .database import Base, reserve_sqlite_immediate_transaction


def now():
    return datetime.now(timezone.utc).isoformat()


def encoded(value):
    return json.dumps(value, ensure_ascii=True, sort_keys=True, separators=(",", ":"), allow_nan=False)


def digest(value):
    return hashlib.sha256(encoded(value).encode()).hexdigest()


class InventoryError(ValueError):
    def __init__(self, code, message):
        self.code = code
        super().__init__(message)


class InventoryState(Base):
    __tablename__ = "workflow_inventory_states"
    key: Mapped[str] = mapped_column(String(30), primary_key=True)
    revision: Mapped[int] = mapped_column(Integer, default=0)


class InventoryObject(Base):
    __tablename__ = "workflow_inventory_objects"
    key: Mapped[str] = mapped_column(String(120), primary_key=True)
    kind: Mapped[str] = mapped_column(String(30), index=True)
    name: Mapped[str] = mapped_column(String(200), index=True)
    status: Mapped[str] = mapped_column(String(20), index=True)
    revision: Mapped[int] = mapped_column(Integer)
    # Derived from the descriptive payload; one shared record per executable flow.
    platform_workflow_key: Mapped[str | None] = mapped_column(String(160), nullable=True, unique=True)
    payload_json: Mapped[str] = mapped_column(Text)
    updated_by: Mapped[str] = mapped_column(String(100))
    updated_at: Mapped[str] = mapped_column(String(40), default=now)


class InventoryRelation(Base):
    __tablename__ = "workflow_inventory_relations"
    key: Mapped[str] = mapped_column(String(120), primary_key=True)
    source: Mapped[str] = mapped_column(ForeignKey("workflow_inventory_objects.key", ondelete="RESTRICT"), index=True)
    target: Mapped[str] = mapped_column(ForeignKey("workflow_inventory_objects.key", ondelete="RESTRICT"), index=True)
    status: Mapped[str] = mapped_column(String(20), index=True)
    revision: Mapped[int] = mapped_column(Integer)
    payload_json: Mapped[str] = mapped_column(Text)
    updated_by: Mapped[str] = mapped_column(String(100))
    updated_at: Mapped[str] = mapped_column(String(40), default=now)


class InventoryDraft(Base):
    __tablename__ = "workflow_inventory_drafts"
    id: Mapped[str] = mapped_column(String(36), primary_key=True)
    actor: Mapped[str] = mapped_column(String(100), index=True)
    title: Mapped[str] = mapped_column(String(200))
    revision: Mapped[int] = mapped_column(Integer, default=1)
    state: Mapped[str] = mapped_column(String(20), default="draft", index=True)
    payload_json: Mapped[str] = mapped_column(Text)
    preview_json: Mapped[str | None] = mapped_column(Text, nullable=True)
    preview_hash: Mapped[str | None] = mapped_column(String(64), nullable=True)
    # Immutable provenance: restoring history creates a new draft, never edits it.
    restored_from: Mapped[int | None] = mapped_column(Integer, nullable=True)
    created_at: Mapped[str] = mapped_column(String(40), default=now)
    updated_at: Mapped[str] = mapped_column(String(40), default=now)
    published_at: Mapped[str | None] = mapped_column(String(40), nullable=True)


class InventoryHistory(Base):
    __tablename__ = "workflow_inventory_history"
    __table_args__ = (UniqueConstraint("record_type", "record_key", "revision", name="uq_inventory_history_revision"),)
    id: Mapped[int] = mapped_column(primary_key=True)
    record_type: Mapped[str] = mapped_column(String(20))
    record_key: Mapped[str] = mapped_column(String(120), index=True)
    revision: Mapped[int] = mapped_column(Integer)
    draft_id: Mapped[str] = mapped_column(ForeignKey("workflow_inventory_drafts.id", ondelete="RESTRICT"), index=True)
    snapshot_json: Mapped[str] = mapped_column(Text)
    actor: Mapped[str] = mapped_column(String(100))
    created_at: Mapped[str] = mapped_column(String(40), default=now)


Key = str
KEY_PATTERN = r"^[a-z0-9][a-z0-9._:-]{0,119}$"
KINDS = ("workflow", "lifecycle", "rule", "script", "machine", "table", "field", "interface", "business", "deployment")
STAGES = ("接入", "治理加工", "评测审核", "发布交付", "业务使用", "反馈纠偏", "更新与退役")
DETAIL_KEYS = {
    "workflow": {"platform_workflow_key", "trigger", "completion", "limitations", "version"},
    "lifecycle": {"stage", "scope", "node_code", "applicability", "completion", "limitations", "gaps", "raw_row"},
    "rule": {"version", "code_reference", "logic", "limitations"},
    "script": {"repository", "path", "version", "entrypoint", "limitations"},
    "machine": {"location", "environment", "limitations"},
    "table": {"database", "table_name", "primary_key", "retention", "limitations"},
    "field": {"field_name", "data_type", "meaning", "mapping", "limitations"},
    "interface": {"address", "method", "version", "direction", "limitations"},
    "business": {"scope"},
    "deployment": {"workflow_key", "location", "environment", "version", "run_evidence", "evidence_at", "limitations", "runtime_source"},
}

class ChangeSet(BaseModel):
    model_config = ConfigDict(extra="forbid", strict=True)
    objects: list[ObjectEdit] = Field(default_factory=list, max_length=500)
    relations: list[RelationEdit] = Field(default_factory=list, max_length=1000)


def record(row):
    value = json.loads(row.payload_json)
    value.update(revision=row.revision, updated_by=row.updated_by, updated_at=row.updated_at)
    return value


def _current(db, model, key):
    return db.scalar(select(model).where(model.key == key).with_for_update().execution_options(populate_existing=True))


def _catalog(db):
    reserve_sqlite_immediate_transaction(db)
    state = _current(db, InventoryState, "catalog")
    if state is None:
        raise InventoryError("not_initialized", "登记库尚未完成升级")
    return state


def _draft(db, draft_id, actor):
    draft = db.scalar(select(InventoryDraft).where(InventoryDraft.id == draft_id, InventoryDraft.actor == actor)
                      .with_for_update().execution_options(populate_existing=True))
    if draft is None:
        raise InventoryError("not_found", "草稿不存在或不属于当前账号")
    return draft


def _bounded(value):
    if not isinstance(value, dict) or len(encoded(value).encode()) > 2 * 1024 * 1024:
        raise InventoryError("too_large", "草稿必须是对象，且不能超过2MiB")


def create_draft(db, *, actor, title, payload=None, restored_from=None):
    payload = payload if payload is not None else {"objects": [], "relations": []}
    _bounded(payload)
    row = InventoryDraft(id=str(uuid.uuid4()), actor=actor, title=title, payload_json=encoded(payload), restored_from=restored_from)
    db.add(row); db.flush()
    return row


def save_draft(db, draft_id, *, actor, expected_revision, title, payload):
    _bounded(payload)
    reserve_sqlite_immediate_transaction(db)
    row = _draft(db, draft_id, actor)
    if row.state != "draft" or row.revision != expected_revision:
        raise InventoryError("draft_conflict", "草稿已变化，请重新打开；当前输入不会覆盖已保存版本")
    row.payload_json = encoded(payload); row.title = title
    row.revision += 1; row.updated_at = now(); row.preview_json = row.preview_hash = None
    db.flush()
    return row


def _validate(db, payload):
    """Validation has no writes, network requests or execution side effects."""
    try:
        parsed = ChangeSet.model_validate(payload)
    except ValidationError as exc:
        return [], [{"location": ".".join(map(str, x["loc"])), "message": x["msg"]} for x in exc.errors(include_input=False, include_url=False)]
    errors, changes, pending = [], [], {}
    for kind, items, model in (("object", parsed.objects, InventoryObject), ("relation", parsed.relations, InventoryRelation)):
        seen = set()
        for item in items:
            def error(message): errors.append({"location": f"{kind}:{item.key}", "message": message})
            if item.key in seen:
                error("同一批次编号重复"); continue
            seen.add(item.key)
            old = _current(db, model, item.key)
            if (old.revision if old else 0) != item.expected_revision:
                error("编号已存在或资料版本已变化，请核对当前版本"); continue
            value = item.model_dump(exclude={"expected_revision"})
            if kind == "object":
                if not item.name.strip(): error("名称不能为空")
                if old and old.kind != item.kind: error("已有对象不能改换类型；请创建正确的对象并重新关联")
                if set(item.details) - DETAIL_KEYS[item.kind] or any(len(v) > 16000 for v in item.details.values()):
                    error("包含该类型不支持的资料字段，或单项资料超过16000字")
                if item.position is not None and (set(item.position) != {"x", "y"} or any(v < 0 or v > 20000 for v in item.position.values())):
                    error("图上坐标需要包含0至20000的整数x和y")
                if item.verification == "confirmed" and not item.evidence.strip(): error("已核实对象需要填写依据")
                if item.verification == "not_applicable" and not item.description.strip(): error("不适用对象需要说明原因")
                if item.kind == "lifecycle" and item.details.get("stage", "") not in STAGES: error("请选择统一生命周期阶段")
                pending[item.key] = value
            else:
                if not item.label.strip(): error("关系名称不能为空")
                if item.source == item.target: error("关系不能指向自身；返工请连接对应节点")
                if item.verification == "confirmed" and not item.evidence.strip(): error("已核实关系需要填写依据")
                if old and (old.source != item.source or old.target != item.target):
                    error("已有连线不能改换两端，请停用旧连线并创建新连线")
            before = json.loads(old.payload_json) if old else None
            if before != value:
                changes.append({"type": kind, "key": item.key, "expected_revision": item.expected_revision, "before": before, "after": value})
    for change in changes:
        value = change["after"]
        if change["type"] == "relation":
            for endpoint in (value["source"], value["target"]):
                existing = pending.get(endpoint)
                if existing is None:
                    row = _current(db, InventoryObject, endpoint)
                    existing = json.loads(row.payload_json) if row else None
                if existing is None:
                    errors.append({"location": "relation:" + value["key"], "message": "关系端点不存在：" + endpoint})
        elif value["kind"] == "deployment":
            if value["details"].get("runtime_source", "") not in ("", "local_platform"):
                errors.append({"location": "object:" + value["key"], "message": "运行来源仅支持当前平台；其他部署保持未知"})
            key = value["details"].get("workflow_key", "")
            workflow = pending.get(key)
            if workflow is None:
                row = _current(db, InventoryObject, key)
                workflow = json.loads(row.payload_json) if row else None
            if workflow is None or workflow["kind"] != "workflow":
                errors.append({"location": "object:" + value["key"], "message": "部署必须关联已登记的流程编号"})
        elif value["kind"] == "workflow" and value["details"].get("platform_workflow_key"):
            from .models import WorkflowDefinition
            key = value["details"]["platform_workflow_key"]
            if db.scalar(select(WorkflowDefinition.id).where(WorkflowDefinition.workflow_key == key)) is None:
                errors.append({"location": "object:" + value["key"], "message": "平台流程编号不存在"})
            duplicate = db.scalar(select(InventoryObject.key).where(InventoryObject.platform_workflow_key == key, InventoryObject.key != value["key"]))
            if duplicate or sum(x["kind"] == "workflow" and x["details"].get("platform_workflow_key") == key for x in pending.values()) > 1:
                errors.append({"location": "object:" + value["key"], "message": "平台流程已有共用登记资料，请引用已有编号"})
    return changes, errors


def preview(db, draft_id, *, actor, expected_revision):
    state = _catalog(db)
    draft = _draft(db, draft_id, actor)
    if draft.state != "draft" or draft.revision != expected_revision:
        raise InventoryError("draft_conflict", "草稿已变化，请先保存当前输入并重新检查")
    changes, errors = _validate(db, json.loads(draft.payload_json))
    value = {"catalog_revision": state.revision, "draft_revision": draft.revision, "changes": changes, "errors": errors}
    draft.preview_json = encoded(value)
    draft.preview_hash = digest({"id": draft.id, "payload": draft.payload_json, "preview": value})
    db.flush()
    return value


def publish(db, draft_id, *, actor, preview_hash):
    state = _catalog(db)
    draft = _draft(db, draft_id, actor)
    if not draft.preview_hash or draft.preview_hash != preview_hash:
        raise InventoryError("preview_changed", "检查结果已变化，请重新检查并确认")
    if draft.state == "published": return draft
    value = json.loads(draft.preview_json)
    if value["errors"] or not value["changes"]:
        raise InventoryError("invalid", "须全部检查通过且有实际修改，才能整批生效")
    if state.revision != value["catalog_revision"] or draft.revision != value["draft_revision"]:
        raise InventoryError("catalog_conflict", "登记资料已更新；草稿保留，请核对最新资料后重新检查")
    changes, errors = _validate(db, json.loads(draft.payload_json))
    if errors or changes != value["changes"]:
        raise InventoryError("catalog_conflict", "对象或关系已变化，请重新检查；草稿保留")
    for kind in ("object", "relation"):
        for change in changes:
            if change["type"] != kind: continue
            model = InventoryObject if kind == "object" else InventoryRelation
            row = _current(db, model, change["key"])
            if row is None:
                row = model(key=change["key"]); db.add(row)
            data = change["after"]
            for key in (("kind", "name", "status") if kind == "object" else ("source", "target", "status")):
                setattr(row, key, data[key])
            if kind == "object":
                row.platform_workflow_key = data["details"].get("platform_workflow_key") or None if data["kind"] == "workflow" else None
            row.revision = change["expected_revision"] + 1
            row.payload_json = encoded(data); row.updated_by = actor; row.updated_at = now()
            db.add(InventoryHistory(record_type=kind, record_key=row.key, revision=row.revision, draft_id=draft.id,
                                    snapshot_json=row.payload_json, actor=actor))
        db.flush()  # objects must exist before inserting foreign-key edges
    state.revision += 1
    draft.state = "published"; draft.published_at = draft.updated_at = now()
    db.flush()
    return draft


def restore_draft(db, history_id, *, actor):
    _catalog(db)
    history = db.get(InventoryHistory, history_id)
    if history is None: raise InventoryError("not_found", "历史记录不存在")
    model = InventoryObject if history.record_type == "object" else InventoryRelation
    row = _current(db, model, history.record_key)
    if row is None: raise InventoryError("not_found", "原对象不存在")
    data = json.loads(history.snapshot_json)
    data["expected_revision"] = row.revision
    payload = {"objects": [], "relations": []}
    payload["objects" if history.record_type == "object" else "relations"] = [data]
    return create_draft(db, actor=actor, title="恢复 " + history.record_key + " 的修订 " + str(history.revision),
                        payload=payload, restored_from=history.id)

"""Employee-only descriptive workflow inventory; no execution or network IO."""
from __future__ import annotations

import csv
import io

from typing import Literal

from fastapi import APIRouter, Depends, HTTPException, Query
from fastapi.responses import Response

from sqlalchemy import func, or_, select

from .authz import has_permission
from .database import get_db
from . import workflow_inventory as w
from .workflow_inventory_runtime import observations, platform_draft


class DraftInput(BaseModel):
    model_config = ConfigDict(extra="forbid", strict=True)

class DraftSave(DraftInput):

class PlatformImport(BaseModel):
    model_config = ConfigDict(extra="forbid", strict=True)
    workflow_keys: list[str] = Field(min_length=1, max_length=100)


class PreviewInput(BaseModel):
    model_config = ConfigDict(extra="forbid", strict=True)
    expected_revision: int = Field(ge=1)

    model_config = ConfigDict(extra="forbid", strict=True)
    preview_hash: str = Field(pattern=r"^[0-9a-f]{64}$")
    confirmed: Literal[True]


def draft_payload(row):
    return {"id": row.id, "title": row.title, "revision": row.revision, "state": row.state,
            "payload": json.loads(row.payload_json), "preview": json.loads(row.preview_json) if row.preview_json else None,
            "preview_hash": row.preview_hash, "restored_from": row.restored_from, "updated_at": row.updated_at}


def build_workflow_inventory_router(current_user):
    router = APIRouter(tags=["workflow-inventory"])

    def writer(user):
        if not has_permission(user, "drafts:write"):
            raise HTTPException(403, "需要系统管理员或产研权限")
        return "user:" + str(user.id)

    def execute(db, fn):
        try:
            result = fn(); db.commit(); return result
        except w.InventoryError as exc:
            db.rollback()
            raise HTTPException(404 if exc.code == "not_found" else 409, {"code": exc.code, "message": str(exc)}) from None
        except IntegrityError:
            db.rollback()
            raise HTTPException(409, "编号或版本已变化，请重新检查；草稿保留") from None

    def filtered(kind, q, status):
        query = select(w.InventoryObject)
        if kind: query = query.where(w.InventoryObject.kind == kind)
        if status: query = query.where(w.InventoryObject.status == status)
        if q: query = query.where(or_(w.InventoryObject.name.contains(q, autoescape=True), w.InventoryObject.key.contains(q, autoescape=True)))
        return query

    @router.get("/api/workflow-inventory/config")
    def config(user=Depends(current_user)):
        return {"can_edit": has_permission(user, "drafts:write"), "kinds": w.KINDS, "stages": w.STAGES,
                "detail_keys": {k: sorted(v) for k, v in w.DETAIL_KEYS.items()}, "runtime_observation": "local_platform"}

    @router.get("/api/workflow-inventory/objects")
    def listing(kind: str = Query(default="workflow", max_length=30), q: str = Query(default="", max_length=200),
                status: Literal["active", "inactive", ""] = "", offset: int = Query(default=0, ge=0), limit: int = Query(default=24, ge=1, le=100),
                user=Depends(current_user), db: Session = Depends(get_db)):
        query = filtered(kind, q, status)
        total = db.scalar(select(func.count()).select_from(query.subquery()))
        items = [w.record(r) for r in db.scalars(query.order_by(w.InventoryObject.key).offset(offset).limit(limit))]
        return {"items": items, "runtime": observations(db, items), "total": total, "offset": offset, "limit": limit, "has_more": offset + limit < total}

    @router.get("/api/workflow-inventory/platform-workflows")
    def platform_workflows(q: str = Query(default="", max_length=200), offset: int = Query(default=0, ge=0),
                           user=Depends(current_user), db: Session = Depends(get_db)):
        from .models import WorkflowDefinition as Definition
        query = select(Definition).where(or_(Definition.name.contains(q, autoescape=True), Definition.workflow_key.contains(q, autoescape=True)))
        total = db.scalar(select(func.count()).select_from(query.subquery()))
        rows = list(db.scalars(query.order_by(Definition.id).offset(offset).limit(24)))
        registered = dict(db.execute(select(w.InventoryObject.platform_workflow_key, w.InventoryObject.key)
            .where(w.InventoryObject.platform_workflow_key.in_([r.workflow_key for r in rows]))).all())
        return {"total": total, "offset": offset, "limit": 24, "has_more": offset + 24 < total,
                "items": [{"workflow_key": r.workflow_key, "name": r.name, "owner": r.owner,
                           "status": r.status, "registered_key": registered.get(r.workflow_key)} for r in rows]}

    @router.post("/api/workflow-inventory/imports/platform")
    def import_platform(body: PlatformImport, user=Depends(current_user), db: Session = Depends(get_db)):
        actor = writer(user)
        return execute(db, lambda: draft_payload(platform_draft(db, actor=actor, workflow_keys=body.workflow_keys)))

    @router.get("/api/workflow-inventory/objects/{key}")
    def detail(key: str, user=Depends(current_user), db: Session = Depends(get_db)):
        row = db.get(w.InventoryObject, key)
        if row is None: raise HTTPException(404, "登记对象不存在")
        return w.record(row)

    @router.get("/api/workflow-inventory/objects/{key}/runtime")
    def runtime(key: str, user=Depends(current_user), db: Session = Depends(get_db)):
        row = db.get(w.InventoryObject, key)
        if row is None: raise HTTPException(404, "登记对象不存在")
        return observations(db, [w.record(row)]).get(key)

    @router.get("/api/workflow-inventory/graph/{key}")
    def graph(key: str, depth: int = Query(default=1, ge=1, le=3), user=Depends(current_user), db: Session = Depends(get_db)):
        if db.get(w.InventoryObject, key) is None: raise HTTPException(404, "登记对象不存在")
        keys, frontier, edges, truncated = {key}, {key}, {}, False
        for _ in range(depth):
            if not frontier: break
            rows = list(db.scalars(select(w.InventoryRelation).where(
                or_(w.InventoryRelation.source.in_(frontier), w.InventoryRelation.target.in_(frontier)),
                w.InventoryRelation.status == "active").order_by(w.InventoryRelation.key).limit(601)))
            if len(rows) > 600: truncated = True
            new_keys = set()

"""Import the agreed two-sheet lifecycle inventory without inferring relationships."""
import csv
import hashlib
import io
import json
import re

from sqlalchemy import select

from . import workflow_inventory as w

HEADERS = (
    "生命周期阶段", "业务／类目范围", "节点编号", "节点名称与用途", "路径类型", "上游输入与触发", "工作流／任务",
    "业务规则与版本", "脚本与执行入口", "机器与运行方式", "读写表与主键", "关键字段来源与去向", "下游接口／接收对象",
    "完成判定", "下一节点与条件", "频率、耗时与限制", "存放位置与留存", "最近运行与证据", "适用与核实情况", "缺口与改造事项", "梳理人／更新日期",
)


def draft_from_lifecycle(db, *, actor, table, text, source):
    if table not in ("3D-SU", "采集"): raise w.InventoryError("format", "请选择3D-SU或采集表")
    header_index, rows = None, []
    for delimiter in (",", "\t"):
        try:
            candidate = list(csv.reader(io.StringIO(text.lstrip("\ufeff")), delimiter=delimiter, strict=True))
        except csv.Error:
            continue
        for index, row in enumerate(candidate[:5]):
            normalized = tuple(re.sub(r"^【(?:产品|研发|共同)】\s*", "", c.strip()).replace("/", "／") for c in row)
            if normalized == HEADERS:
                header_index, rows = index, candidate; break
        if header_index is not None: break
    if header_index is None:
        raise w.InventoryError("format", "未找到现行21列表头；请导出钉钉梳理表CSV，或从表头开始复制表格")
    if len(rows) - header_index - 1 > 500: raise w.InventoryError("format", "每次最多导入500行资料")
    groups = {}
    for row_no, cells in enumerate(rows[header_index + 1:], header_index + 2):
        if not any(c.strip() for c in cells): continue
        if len(cells) != len(HEADERS): raise w.InventoryError("format", f"第{row_no}行列数不一致")
        row = dict(zip(HEADERS, cells))
        code, name, stage = row["节点编号"].strip(), row["节点名称与用途"].strip(), row["生命周期阶段"].strip()
        if not code or not name or stage not in w.STAGES:
            raise w.InventoryError("format", f"第{row_no}行需要节点编号、名称和有效的生命周期阶段")
        if code.upper().startswith("EX"):
            raise w.InventoryError("format", f"第{row_no}行仍是EX示例，请删除示例后再导入")
        if row["路径类型"].strip() not in ("正常", "异常", "返工"):
            raise w.InventoryError("format", f"第{row_no}行路径类型须为正常、异常或返工")
        key = "lifecycle:" + ("3dsu:" if table == "3D-SU" else "collection:") + hashlib.sha256(code.casefold().encode()).hexdigest()[:32]
        if key in groups and (groups[key]["code"] != code or groups[key]["name"] != name or groups[key]["stage"] != stage):
            raise w.InventoryError("format", f"第{row_no}行重复编号的名称或阶段不一致，请先合并核对")
        group = groups.setdefault(key, {"code": code, "name": name, "stage": stage, "rows": []})
        group["rows"].append({"row_number": row_no, "values": row})
    if not groups: raise w.InventoryError("format", "表格没有可导入的资料行")
    objects = []
    for key, group in groups.items():
        existing = db.scalar(select(w.InventoryObject).where(w.InventoryObject.key == key))
        first = group["rows"][0]["values"]
        raw = json.dumps(group["rows"], ensure_ascii=False)
        if len(raw) > 16000: raise w.InventoryError("format", f"节点{group['code']}资料过长，请把详细内容放到引用文档")
        # Matching is confined to stable sheet+node number. Never infer script,
        # host, table or edge IDs from a prose cell or mark evidence as verified.
        value = w.ObjectEdit(key=key, expected_revision=existing.revision if existing else 0, kind="lifecycle",
            name=group["name"], description=f"{table}表节点 {group['code']}，含{len(group['rows'])}条路径；原始资料待核对。",
            source=source, evidence=first["最近运行与证据"], verification="unverified",
            details={"stage": group["stage"], "scope": first["业务／类目范围"], "node_code": group["code"],
                "applicability": first["适用与核实情况"], "completion": first["完成判定"], "limitations": first["频率、耗时与限制"],
                "gaps": first["缺口与改造事项"], "raw_row": raw},
            status=existing.status if existing else "active",
            owner=json.loads(existing.payload_json).get("owner", "") if existing else "",
            position=json.loads(existing.payload_json).get("position") if existing else None).model_dump()
        objects.append(value)
    draft = w.create_draft(db, actor=actor, title=f"{table}生命周期资料导入", payload={"objects": objects, "relations": []})
    w.preview(db, draft.id, actor=actor, expected_revision=1)
    return draft

"""Read local runtime facts without loading execution snapshots or changing runs."""
from __future__ import annotations

import hashlib
import json
from datetime import datetime, timezone

from sqlalchemy import Index, select

from . import workflow_inventory as w
from .models import ProductionRun, WorkflowDefinition


RUNTIME_INDEX = Index("ix_production_runs_inventory_latest", ProductionRun.workflow_definition_id,
                      ProductionRun.run_mode, ProductionRun.model_execution_mode, ProductionRun.id)


def observations(db, objects):
    """A latest-run fact is not a health verdict for every deployment or task."""
    flow_keys = {o["details"].get("workflow_key") for o in objects
                 if o["kind"] == "deployment" and o["details"].get("runtime_source") == "local_platform"}
    referenced = {r.key: json.loads(r.payload_json) for r in db.scalars(
        select(w.InventoryObject).where(w.InventoryObject.key.in_(flow_keys)))} if flow_keys else {}
    bindings = {}
    for obj in objects:
        if obj["kind"] == "deployment" and obj["details"].get("runtime_source") != "local_platform":
            continue
        flow = obj if obj["kind"] == "workflow" else referenced.get(obj["details"].get("workflow_key"))
        if flow and flow["kind"] == "workflow":
            bindings[obj["key"]] = flow["details"].get("platform_workflow_key")
    keys = sorted({v for v in bindings.values() if v})
    facts = {}
    if keys:
        latest_id = select(ProductionRun.id).where(
            ProductionRun.workflow_definition_id == WorkflowDefinition.id,
            ProductionRun.run_mode == "incremental", ProductionRun.model_execution_mode == "real"
        ).order_by(ProductionRun.id.desc()).limit(1).correlate(WorkflowDefinition).scalar_subquery()
        latest = db.execute(select(WorkflowDefinition.workflow_key, latest_id.label("run_id"))
                            .where(WorkflowDefinition.workflow_key.in_(keys))).all()
        ids = [r.run_id for r in latest if r.run_id is not None]
        runs = {r.id: r for r in db.execute(select(ProductionRun.id, ProductionRun.status,
                    ProductionRun.workflow_version_id, ProductionRun.created_at, ProductionRun.updated_at,
                    ProductionRun.started_at, ProductionRun.finished_at, ProductionRun.error_code)
                    .where(ProductionRun.id.in_(ids)))} if ids else {}
        for row in latest:
            run = runs.get(row.run_id)
            facts[row.workflow_key] = {
                "state": "observed" if run else "unknown", "reason": "" if run else "no_incremental_run",
                "source": "local_platform", "scope": "latest_created_real_incremental_run",
                "observed_at": w.now(), "run": {key: (value.replace(tzinfo=timezone.utc) if isinstance(value, datetime) and value.tzinfo is None else value)
                    for key, value in run._mapping.items()} if run else None,
            }
    result = {}
    for obj in objects:
        if obj["kind"] not in ("workflow", "deployment"): continue
        fact = facts.get(bindings.get(obj["key"]))
        result[obj["key"]] = fact or {"state": "unknown", "reason": "not_connected", "source": "none", "run": None}
    return result


def platform_draft(db, *, actor, workflow_keys):
    definitions = list(db.scalars(select(WorkflowDefinition).where(WorkflowDefinition.workflow_key.in_(workflow_keys))
                                 .order_by(WorkflowDefinition.id)))
    if len(definitions) != len(workflow_keys):
        raise w.InventoryError("invalid", "平台流程不存在或选择重复，请重新选择")
    registered = set(db.scalars(select(w.InventoryObject.platform_workflow_key)
                               .where(w.InventoryObject.platform_workflow_key.in_(workflow_keys))))
    if registered:
        raise w.InventoryError("already_registered", "所选流程已有共用资料，请从流程清单编辑或关联")
    objects = []
    for definition in definitions:
        key = "platform:" + hashlib.sha256(definition.workflow_key.encode()).hexdigest()[:32]
        objects.append(w.ObjectEdit(key=key, expected_revision=0, kind="workflow", name=definition.name,
            description=definition.description[:10000], owner=definition.owner,
            status="inactive" if definition.status == "retired" else "active", source="平台工作流登记",
            details={"platform_workflow_key": definition.workflow_key}).model_dump())
    return w.create_draft(db, actor=actor, title="纳入已迁移流程", payload={"objects": objects, "relations": []})

export function editable<T extends InventoryObject | InventoryRelation>(value: T): T {

/** Only the exported schema is imported here; free-text lifecycle rows need explicit mapping. */
export function parseInventoryImport(raw: string): InventoryChangeSet {
  const text = raw.replace(/^\uFEFF/, "").trim()
  if (text.startsWith("{")) {
    const value = JSON.parse(text)
    if (value.schema_version && value.schema_version !== "workflow-inventory-v1") throw new Error("导入文件格式版本不受支持")
    if (!Array.isArray(value.objects) || !Array.isArray(value.relations)) throw new Error("JSON需要objects和relations两个数组")
    return { objects: value.objects.map(editable), relations: value.relations.map(editable) }
  }
  const rows: string[][] = []; let row: string[] = [], field = "", quoted = false, closed = false
  for (let i = 0; i < text.length; i++) {
    const c = text[i]
    if (c === '"') {
      if (quoted && text[i + 1] === '"') { field += '"'; i++ }
      else if (quoted) { quoted = false; closed = true }
      else if (!field && !closed) quoted = true
      else throw new Error("CSV引号位置无效")
    } else if (c === "," && !quoted) { row.push(field); field = ""; closed = false }
    else if ((c === "\n" || c === "\r") && !quoted) {
      if (c === "\r" && text[i + 1] === "\n") i++
      row.push(field); rows.push(row); row = []; field = ""; closed = false
    } else { if (closed) throw new Error("CSV结束引号后有多余字符"); field += c }
  }
  if (quoted) throw new Error("CSV引号未闭合")
  if (field || row.length) { row.push(field); rows.push(row) }
  const headers = rows.shift()?.map(h => h.trim()) || []
  const allowed = new Set([...Object.keys(newInventoryObject()), "_text_encoding"])
  if (!headers.length || new Set(headers).size !== headers.length || headers.some(h => !allowed.has(h))) throw new Error("请使用本页导出的表格列；生命周期原表需要先明确对象对应关系")
  const objects = rows.filter(r => r.some(Boolean)).map((r, i) => {
    if (r.length !== headers.length) throw new Error(`第${i + 2}行列数不一致`)
    const encoded = headers.indexOf("_text_encoding")
    if (encoded >= 0 && r[encoded] !== "apostrophe-v1") throw new Error("导出转义格式无效")
    const parsed = Object.fromEntries(headers.flatMap<[string, unknown]>((key, n) => {
      if (key === "_text_encoding") return []
      let value = r[n]; if (encoded >= 0 && /^'['=+@\-\t\r]/.test(value)) value = value.slice(1)
      if (key === "expected_revision") {
        if (!/^\d+$/.test(value) || !Number.isSafeInteger(Number(value))) throw new Error(`第${i + 2}行版本须为非负整数`)
        return [[key, Number(value)]]
      }
      if (key === "details" || key === "position") return [[key, JSON.parse(value || (key === "details" ? "{}" : "null"))]]
      return [[key, value]]
    }))
    return parsed as InventoryObject
  })
  return { objects, relations: [] }
}

# 工作流登记与运行配置分离

登记资料独立于现有可执行WorkflowDefinition及脚本执行注册。工作流、生命周期节点、业务、规则、脚本、机器、表、字段、接口、部署分别有稳定编号，关系引用同一对象，不因服务多个业务复制流程。部署资料单独引用流程编号；登记模块不运行脚本、不访问任意接口、不写生产绑定和开关。

所有编辑进入个人草稿。自动保存使用草稿修订号，检查和生效使用资料库修订、草稿修订及检查摘要。锁序为资料库、草稿、对象；自动保存只锁草稿。任何一条资料错误都不写正式资料；并发修改冲突时保留草稿，要求核对后重新检查。已生效请求可按相同摘要幂等返回。历史修订只追加，恢复先创建新草稿并保留原历史引用。

关系区分正常、异常、返工，允许不同对象形成返工回路；登记图不是执行引擎DAG。图查询限制最多300对象、600关系，明确标出截断；清单分页，导出当前筛选的全部对象分页，不在导出时持有写锁。CSV的表格公式转义有显式格式标记，避免吞掉原文的单引号。

庆枝团队的3D-SU和采集两张21列原表按“表名+节点编号”匹配节点，同节点的正常/异常/返工行完整留存。导入不会从自由文字推断脚本、机器、表和连线，更不会自动标记已核实。EX示例、错列、重复编号含义冲突等整批拒绝。收录后可在图和表单逐项核对并关联共享对象；既有对象的编号及历史不重生。

权限沿用管理员及产研可编辑、其他员工只读；个人草稿仅本人可读写。接口遵守员工Cookie/同源边界，未增加服务令牌权限。实际部署状态需明确采集来源，旧部署未知不冒充正常；平台流程可明确选择纳入草稿；运行观察只覆盖当前平台最近创建的真实增量任务。负责人异常归并和钉钉通知仍为下一实施部分，尚未实现。

草稿上下文只读补齐现有端点，不改变保存时的修订；已停用连线仍可查看和恢复。平台编号的唯一列由发布代码从资料字段派生，不允许通过API单独修改。增量查询使用workflow_definition_id/run_mode/model_execution_mode/id索引，排除模拟及存量记录；不同部署必须明确观测来源。JSON导出在单个只读事务快照中读取对象和两端都匹配的关系，不持有登记写锁。
